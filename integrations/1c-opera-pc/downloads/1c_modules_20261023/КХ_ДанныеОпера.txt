// Снимок только переданной номенклатуры. Цена и остаток относятся к указанной СЕ.
// Удаление - пометка удаления. Физическое удаление без сохранённого внешнего ID
// здесь не превращается в фиктивную карточку или молчаливое удаление.
Функция СформироватьСостояниеПоНоменклатурам(СтруктурнаяЕдиница, Ссылки) Экспорт
	Результат = Новый Структура("products,deleted_nomenclature_ids", Новый Массив, Новый Массив);
	Если Ссылки.Количество() = 0 Тогда Возврат Результат; КонецЕсли;
	Запрос = Новый Запрос;
	Запрос.Текст =
	"ВЫБРАТЬ Номенклатура.Ссылка КАК Ссылка,
	| Номенклатура.СтроковыйИдентификатор КАК Идентификатор,
	| Номенклатура.ПометкаУдаления КАК ПометкаУдаления,
	| Номенклатура.ЭтоГруппа КАК ЭтоГруппа
	|ИЗ Справочник.Номенклатура КАК Номенклатура
	|ГДЕ Номенклатура.Ссылка В (&Ссылки)";
	Запрос.УстановитьПараметр("Ссылки", Ссылки);
	Таблица = Запрос.Выполнить().Выгрузить();
	Уникальные = Новый Соответствие;
	Для Каждого Ссылка Из Ссылки Цикл Уникальные.Вставить(Ссылка, Истина); КонецЦикла;
	Если Таблица.Количество() <> Уникальные.Количество() Тогда
		ВызватьИсключение "Часть номенклатуры отсутствует в базе; проверьте физическое удаление и очередь изменений";
	КонецЕсли;
	Действующие = Новый Массив;
	Для Каждого СтрокаНоменклатуры Из Таблица Цикл
		Если СтрокаНоменклатуры.ЭтоГруппа Тогда Продолжить; КонецЕсли;
		Если ПустаяСтрока(СокрЛП(Строка(СтрокаНоменклатуры.Идентификатор))) Тогда
			ВызватьИсключение "Не заполнен строковый идентификатор товара: " + Строка(СтрокаНоменклатуры.Ссылка);
		КонецЕсли;
		Если СтрокаНоменклатуры.ПометкаУдаления Тогда
			Результат.deleted_nomenclature_ids.Добавить(Строка(СтрокаНоменклатуры.Идентификатор));
		Иначе
			Действующие.Добавить(СтрокаНоменклатуры.Ссылка);
		КонецЕсли;
	КонецЦикла;
	Если Действующие.Количество() > 0 Тогда
		Данные = ПолучитьДанныеДляВыгрузки(СтруктурнаяЕдиница, ТекущаяДатаСеанса(), Действующие);
		Результат.products = СформироватьКарточкиТоваров(Данные);
		Если Результат.products.Количество() <> Действующие.Количество() Тогда
			ВызватьИсключение "Не все карточки сформированы однозначно: проверьте наименования, идентификаторы и свойства товаров";
		КонецЕсли;
	КонецЕсли;
	Возврат Результат;
КонецФункции

Функция ПолучитьКарточкуПоШтрихкоду(СтруктурнаяЕдиница, Штрихкод) Экспорт

	КонецЕсли;

	Товар.Вставить("nomenclature_id", Идентификатор);
	Товар.Вставить("name", Наименование);

	Если НЕ ПустаяСтрока(СокрЛП(Строка(СтрокаНоменклатуры.НаименованиеБазовойЕдиницыИзмерения))) Тогда

	КонецЕсли;

	Если НЕ ПустаяСтрока(СокрЛП(Строка(СтрокаНоменклатуры.НаименованиеСтраныПроисхождения))) Тогда
		Товар.Вставить("country", Строка(СтрокаНоменклатуры.НаименованиеСтраныПроисхождения));
	КонецЕсли;

	Если ЕстьЗначениеДляJSON(СтрокаНоменклатуры.Производитель) Тогда
		Товар.Вставить("manufacturer", Строка(СтрокаНоменклатуры.Производитель));
	КонецЕсли;

	Если НЕ ПустаяСтрока(СокрЛП(Строка(СтрокаНоменклатуры.НаименованиеГруппыНоменклатуры))) Тогда
		Товар.Вставить("category", Строка(СтрокаНоменклатуры.НаименованиеГруппыНоменклатуры));
	КонецЕсли;

	Если ЕстьЗначениеДляJSON(СтрокаНоменклатуры.Состав) Тогда
		Товар.Вставить("composition", Строка(СтрокаНоменклатуры.Состав));
	КонецЕсли;

	Товар.Вставить("nutrition", СформироватьКБЖУ(СтрокаНоменклатуры.Ккал, СтрокаНоменклатуры.Белки, СтрокаНоменклатуры.Жиры, СтрокаНоменклатуры.Углеводы));

	ДанныеЦены = ЦеныПоНоменклатуре.Получить(СтрокаНоменклатуры.Номенклатура);

	Если ДанныеЦены <> Неопределено Тогда
		Товар.Вставить("price", СформироватьДанныеЦены(ДанныеЦены));
	КонецЕсли;

	ШтрихКоды = ШтрихКодыПоНоменклатуре.Получить(СтрокаНоменклатуры.Номенклатура);

	Если ШтрихКоды = Неопределено Тогда
		ШтрихКоды = Новый Массив;
	КонецЕсли;

	Товар.Вставить("barcodes", ШтрихКоды);

	Акции = АкцииПоНоменклатуре.Получить(СтрокаНоменклатуры.Номенклатура);

	Если Акции = Неопределено Тогда
		Акции = Новый Массив;
	КонецЕсли;

	Товар.Вставить("promos", Акции);

	Количество = ОстаткиПоНоменклатуре.Получить(СтрокаНоменклатуры.Номенклатура);

	Если Количество = Неопределено Тогда
		Количество = 0;
	КонецЕсли;

	Товар.Вставить("stock", СформироватьДанныеОстатка(Количество, МоментВыгрузки));

	Возврат Товар;

Функция СформироватьАкцию(СтрокаАкции)

	Если НЕ ЗначениеЗаполнено(СтрокаАкции.ДатаНачала) Или НЕ ЗначениеЗаполнено(СтрокаАкции.ДатаОкончания) Тогда
		Возврат Неопределено;
	КонецЕсли;

	Если СтрокаАкции.ДатаОкончания < СтрокаАкции.ДатаНачала Тогда
		Возврат Неопределено;
	КонецЕсли;

	Акция = Новый Структура;

	Если СтрокаАкции.ВидРекламнойАкции = Перечисления.ВидыРекламныхАкций.СпецЦена Тогда

		Если НЕ ЕстьЗначениеДляJSON(СтрокаАкции.ЦенаАкции) Тогда
			Возврат Неопределено;
		КонецЕсли;

		Акция.Вставить("promo_type", "special_price");
		Акция.Вставить("promo_price", ФорматDecimal(СтрокаАкции.ЦенаАкции, 2));

	Иначе

		ПредставлениеВидаАкции = НРег(СокрЛП(Строка(СтрокаАкции.ВидРекламнойАкции)));

		Если ПредставлениеВидаАкции = "m по n" Или ПредставлениеВидаАкции = "м по n" Или ПредставлениеВидаАкции = "м по н" Тогда

			Если НЕ ЕстьЗначениеДляJSON(СтрокаАкции.ЦенаАкции) Или СтрокаАкции.КоличествоМ <= 0 Тогда
				Возврат Неопределено;
			КонецЕсли;

			Акция.Вставить("promo_type", "m_for_n");
			Акция.Вставить("promo_price", ФорматDecimal(СтрокаАкции.ЦенаАкции, 2));
			Акция.Вставить("required_quantity", СтрокаАкции.КоличествоМ);

		ИначеЕсли ПредставлениеВидаАкции = "скидка %" Тогда

			Если НЕ ЕстьЗначениеДляJSON(СтрокаАкции.Скидка) Тогда
				Возврат Неопределено;
			КонецЕсли;

			Акция.Вставить("promo_type", "percent_discount");
			Акция.Вставить("discount_percent", ФорматDecimal(СтрокаАкции.Скидка, 2));

		ИначеЕсли ПредставлениеВидаАкции = "цена на товары индикаторы" Тогда

			Если НЕ ЕстьЗначениеДляJSON(СтрокаАкции.ЦенаАкции) Тогда
				Возврат Неопределено;
			КонецЕсли;

			Акция.Вставить("promo_type", "indicator_price");
			Акция.Вставить("promo_price", ФорматDecimal(СтрокаАкции.ЦенаАкции, 2));

		Иначе
			Возврат Неопределено;
		КонецЕсли;

	КонецЕсли;

	Акция.Вставить("starts_at", ФорматДатаJSON(СтрокаАкции.ДатаНачала));
	Акция.Вставить("ends_at", ФорматДатаJSON(СтрокаАкции.ДатаОкончания));

	Возврат Акция;

Функция ЕстьЗначениеДляJSON(Значение)

	Если Значение = Неопределено Тогда
		Возврат Ложь;
	КонецЕсли;

	Если ТипЗнч(Значение) = Тип("Null") Тогда
		Возврат Ложь;
	КонецЕсли;

	Если ТипЗнч(Значение) = Тип("Строка") Тогда
		Возврат НЕ ПустаяСтрока(СокрЛП(Значение));
	КонецЕсли;

	Возврат Истина;

Функция ПолучитьЧисловоеЗначение(Значение)

	Если Значение = Неопределено Тогда
		Возврат Неопределено;
	КонецЕсли;

	Если ТипЗнч(Значение) = Тип("Null") Тогда
		Возврат Неопределено;
	КонецЕсли;

	Если ТипЗнч(Значение) = Тип("Число") Тогда
		Возврат Значение;
	КонецЕсли;

	СтрокаЗначения = СокрЛП(Строка(Значение));

	Если ПустаяСтрока(СтрокаЗначения) Тогда
		Возврат Неопределено;
	КонецЕсли;

	СтрокаЗначения = СтрЗаменить(СтрокаЗначения, " ", "");
	СтрокаЗначения = СтрЗаменить(СтрокаЗначения, ".", ",");

	Попытка
		Возврат Число(СтрокаЗначения);
	Исключение
		Возврат Неопределено;
	КонецПопытки;

Функция ФорматDecimal(Значение, КоличествоЗнаков)

	ЧисловоеЗначение = ПолучитьЧисловоеЗначение(Значение);

	Если ЧисловоеЗначение = Неопределено Тогда
		ВызватьИсключение "Не удалось преобразовать значение в число: " + Строка(Значение);
	КонецЕсли;

	ЧисловоеЗначение = Окр(ЧисловоеЗначение, КоличествоЗнаков);
	Результат = Формат(ЧисловоеЗначение, "ЧДЦ=" + Строка(КоличествоЗнаков) + ";ЧГ=0");
	Результат = СтрЗаменить(Результат, ",", ".");
	Результат = СокрЛП(Результат);

	Если КоличествоЗнаков = 0 Тогда

		Если ПустаяСтрока(Результат) Тогда
			Возврат "0";
		КонецЕсли;

		Возврат Результат;

	КонецЕсли;

	Если ПустаяСтрока(Результат) Тогда
		Результат = "0";
	КонецЕсли;

	ПозицияТочки = Найти(Результат, ".");

	Если ПозицияТочки = 0 Тогда
		Результат = Результат + "." + Лев("000000000000000", КоличествоЗнаков);
	Иначе
		КоличествоПослеТочки = СтрДлина(Результат) - ПозицияТочки;

		Если КоличествоПослеТочки < КоличествоЗнаков Тогда
			Результат = Результат + Лев("000000000000000", КоличествоЗнаков - КоличествоПослеТочки);
		КонецЕсли;

	КонецЕсли;

	Возврат Результат;

КонецФункции

Функция ФорматDecimalБезНулей(Значение)

	ЧисловоеЗначение = ПолучитьЧисловоеЗначение(Значение);

	Если ЧисловоеЗначение = Неопределено Тогда
		Возврат "";
	КонецЕсли;

	Результат = Формат(ЧисловоеЗначение, "ЧДЦ=6;ЧГ=0");
	Результат = СтрЗаменить(Результат, ",", ".");
	Результат = СокрЛП(Результат);

	Пока Найти(Результат, ".") > 0 И Прав(Результат, 1) = "0" Цикл
		Результат = Лев(Результат, СтрДлина(Результат) - 1);
	КонецЦикла;

	Если Прав(Результат, 1) = "." Тогда
		Результат = Лев(Результат, СтрДлина(Результат) - 1);
	КонецЕсли;

	Если ПустаяСтрока(Результат) Тогда
		Результат = "0";
	КонецЕсли;

	Возврат Результат;
