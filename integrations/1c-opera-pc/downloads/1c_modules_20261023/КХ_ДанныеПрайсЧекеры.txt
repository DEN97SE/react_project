// Данные собираются в своей магазинной базе. Обращения в ЦО нет.
// Общая карточка используется двумя проектами без копирования запросов.

Функция ПолучитьКарточкуМагазина(ИдентификаторМагазина, Штрихкод) Экспорт
	СЕ = КХ_ОбменШиныОпера.ТекущаяСЕ();
	СвойИдентификатор = КХ_ДанныеОпера.ПолучитьИдентификаторМагазина(СЕ);
	Если ТипЗнч(ИдентификаторМагазина) <> Тип("Строка")
		Или ИдентификаторМагазина <> СвойИдентификатор Тогда
		ВызватьИсключение "Запрос адресован другой базе магазина";
	КонецЕсли;
	Если ТипЗнч(Штрихкод) <> Тип("Строка") Или ПустаяСтрока(СокрЛП(Штрихкод)) Тогда
		ВызватьИсключение "Штрихкод должен быть непустой строкой";
	КонецЕсли;
	Возврат КХ_ДанныеОпера.ПолучитьКарточкуПоШтрихкоду(СЕ, СокрЛП(Штрихкод));
КонецФункции
