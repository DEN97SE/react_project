    Подготовленные = ПодготовитьПлан(План);
    Результат = Новый Массив;

// {{ semidockij - 28.10.2026 - Программная настройка подключений своей базы
// Назначение: сохраняет индивидуальные адреса и ключи наших сервисов, при необходимости задаёт активность.
// Как работает: сначала проверяет весь план, затем последовательно вызывает УстановитьНастройки и УстановитьАктивность.
// Поле ИмяПользователя сохраняется. Возвращает этапы без секретов; ошибка останавливает дальнейшее применение.
// Изменения двух сервисов не объявляются атомарными. Успех записи не означает успешную авторизацию или получение ACK.

Функция ПрименитьПлан(План) Экспорт
    Подготовленные = ПодготовитьПлан(План);
    Результат = Новый Массив;
    Для Каждого Подключение Из Подготовленные Цикл
        Итог = Новый Структура("Сервис,НастройкиЗаписаны,АктивностьУстановлена,Ошибка,Этап",
            Подключение.Сервис, Ложь, Ложь, Ложь, "Сохранение настроек");
        Попытка
            Настройки = Подключение.Настройки;
            Настройки.АдресВнешнегоСервисаИнтеграции = Подключение.Адрес;
            Настройки.ИмяПользователяВнешнегоСервисаИнтеграции = Подключение.ИдентификаторКлюча;
            Настройки.ПарольПользователяВнешнегоСервисаИнтеграции = Подключение.СекретКлиента;
            СервисыИнтеграции[Подключение.Сервис].УстановитьНастройки(Настройки);
            Итог.НастройкиЗаписаны = Истина;
            Если Подключение.Активность <> Неопределено Тогда
                Итог.Этап = "Установка активности";
                СервисыИнтеграции[Подключение.Сервис].УстановитьАктивность(Подключение.Активность);
                Итог.АктивностьУстановлена = Истина;
            КонецЕсли;
            Итог.Этап = "Вызовы платформы завершены";
        Исключение
            // Текст исключения платформы может содержать параметры подключения, поэтому наружу не передаётся.
            // Причина изучается администратором локально в отладчике, без публикации секретов.
            Итог.Ошибка = Истина;
        КонецПопытки;
        Результат.Добавить(Итог);
        Если Итог.Ошибка Тогда Прервать; КонецЕсли;
    КонецЦикла;
    Возврат Результат;
КонецФункции

// {{ semidockij - 28.10.2026 - Подготовка проверенных параметров без записи
// Назначение: проверяет весь набор до первого изменения, чтобы отсутствующее второе подключение выявилось заранее.
// Как работает: сравнивает магазин с текущей СЕ, допускает только два наших сервиса, читает их действующие настройки.
// Пустой Адрес сохраняет текущий адрес; если адреса нет нигде, выполнение прекращается. Настройки содержат секреты.
// }} semidockij - 28.10.2026
Функция ПодготовитьПлан(План)
    Если ТранзакцияАктивна() Тогда ВызватьИсключение "Настройку подключений выполняйте вне транзакции"; КонецЕсли;
    Если ТипЗнч(План) <> Тип("Структура") Тогда ВызватьИсключение "План должен быть структурой"; КонецЕсли;
    Магазин = ОбязательноеПоле(План, "Магазин");
    ПроверитьСтроку(Магазин, "Магазин");
    Если Магазин <> КХ_ОбменШиныОпера.ИдентификаторМагазина() Тогда
        ВызватьИсключение "План предназначен другой текущей СЕ. Настройки не записаны";
    КонецЕсли;
    Подключения = ОбязательноеПоле(План, "Подключения");
    Если ТипЗнч(Подключения) <> Тип("Массив") Тогда ВызватьИсключение "Подключения должны быть массивом"; КонецЕсли;
    Если Подключения.Количество() = 0 ИЛИ Подключения.Количество() > 2 Тогда
        ВызватьИсключение "План должен содержать одно или два подключения наших сервисов";
    КонецЕсли;
    Сервисы = Новый Соответствие;
    Результат = Новый Массив;
    Для Каждого Элемент Из Подключения Цикл
        Если ТипЗнч(Элемент) <> Тип("Структура") Тогда ВызватьИсключение "Подключение должно быть структурой"; КонецЕсли;
        ИмяСервиса = ОбязательноеПоле(Элемент, "Сервис");
        ПроверитьСтроку(ИмяСервиса, "Сервис");
        Если ИмяСервиса <> "ExchOperaPC" И ИмяСервиса <> "ExchPriceChecker" Тогда
            ВызватьИсключение "Разрешены только наши ExchOperaPC и ExchPriceChecker";
        КонецЕсли;
        Если Сервисы.Получить(ИмяСервиса) <> Неопределено Тогда ВызватьИсключение "Сервис повторяется в плане"; КонецЕсли;
        Сервисы.Вставить(ИмяСервиса, Истина);
        Если Метаданные.СервисыИнтеграции.Найти(ИмяСервиса) = Неопределено Тогда
            ВызватьИсключение "В конфигурации базы отсутствует сервис " + ИмяСервиса;
        КонецЕсли;
        ИдентификаторКлюча = ОбязательноеПоле(Элемент, "ИдентификаторКлюча");
        СекретКлиента = ОбязательноеПоле(Элемент, "СекретКлиента");
        ПроверитьСтроку(ИдентификаторКлюча, "ИдентификаторКлюча");
        ПроверитьСтроку(СекретКлиента, "СекретКлиента");
        Активность = Неопределено;
        Если Элемент.Свойство("Активность", Активность) Тогда
            Если ТипЗнч(Активность) <> Тип("Булево") Тогда ВызватьИсключение "Активность должна быть булевым значением"; КонецЕсли;
        КонецЕсли;
        Попытка
            Настройки = СервисыИнтеграции[ИмяСервиса].ПолучитьНастройки();
        Исключение
            ВызватьИсключение "Не удалось прочитать настройки " + ИмяСервиса + ". Проверьте права и область данных";
        КонецПопытки;
        Адрес = Настройки.АдресВнешнегоСервисаИнтеграции;
        НовыйАдрес = "";
        Если Элемент.Свойство("Адрес", НовыйАдрес) Тогда
            Если ТипЗнч(НовыйАдрес) <> Тип("Строка") Тогда ВызватьИсключение "Адрес должен быть строкой"; КонецЕсли;
            Если НЕ ПустаяСтрока(НовыйАдрес) Тогда Адрес = СокрЛП(НовыйАдрес); КонецЕсли;
        КонецЕсли;
        ПроверитьСтроку(Адрес, "АдресВнешнегоСервисаИнтеграции");
        Результат.Добавить(Новый Структура("Сервис,Настройки,Адрес,ИдентификаторКлюча,СекретКлиента,Активность",
            ИмяСервиса, Настройки, Адрес, ИдентификаторКлюча, СекретКлиента, Активность));
    КонецЦикла;
    Возврат Результат;
КонецФункции

// {{ semidockij - 28.10.2026 - Проверка обязательного поля плана
// Назначение: возвращает существующее поле структуры без подстановки реквизитов другого магазина.
// Как работает: при отсутствии поля выдаёт только его имя, не значения остальных полей и секретов.

Функция ОбязательноеПоле(Данные, ИмяПоля)
    Значение = Неопределено;
    Если НЕ Данные.Свойство(ИмяПоля, Значение) Тогда ВызватьИсключение "В плане отсутствует поле " + ИмяПоля; КонецЕсли;
    Возврат Значение;
КонецФункции

// {{ semidockij - 28.10.2026 - Проверка индивидуальных строк настройки
// Назначение: не допускает пустые значения и управляющие переводы строк в адресе и учётных данных.
// Как работает: проверяет тип и содержимое, сохраняет значимые символы секрета без сокращения.
// }} semidockij - 28.10.2026
Процедура ПроверитьСтроку(Значение, ИмяПоля)
    Если ТипЗнч(Значение) <> Тип("Строка") Тогда ВызватьИсключение "Поле должно быть строкой: " + ИмяПоля; КонецЕсли;
    Если ПустаяСтрока(Значение) Тогда ВызватьИсключение "Не заполнено поле " + ИмяПоля; КонецЕсли;
    Если СтрНайти(Значение, Символы.ПС) > 0 ИЛИ СтрНайти(Значение, Символы.ВК) > 0 Тогда
        ВызватьИсключение "В поле обнаружен перевод строки: " + ИмяПоля;
    КонецЕсли;
КонецПроцедуры
