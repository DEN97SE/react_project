// {{ semidockij - 27.10.2026 - Пояснения к методам
// Назначение: регистрирует создание или изменение карточки магазина после записи СЕ.
// Как работает: учитывает текущий и прежний вид СЕ, пропускает группы и создаёт маркер изменения.
// Выход из вида Магазин регистрирует как удаление внешней карточки; при выключенной публикации ничего не делает.
// }} semidockij - 27.10.2026

	Если Источник.ЭтоГруппа Тогда Возврат; КонецЕсли;

	Старые = Неопределено;
	Если Источник.ДополнительныеСвойства.Свойство("КХ_СЕСтарыеДанные", Старые) Тогда
		БылМагазином = Старые.was_store;

	Магазин = Источник.ВидСтруктурнойЕдиницы = Перечисления.ВидыСтруктурныхЕдиниц.Магазин;
	Если НЕ Магазин И НЕ БылМагазином Тогда Возврат; КонецЕсли;
	ЗарегистрироватьИзменение(Источник.Ссылка, СокрЛП(Строка(Источник.СтроковыйИдентификатор)), НЕ Магазин);

	Имя = Источник.Метаданные().Имя;
	Если Имя = "ФорматыМагазинов" Тогда Поле = "Формат";
	ИначеЕсли Имя = "ТерриториальныеОбразования" Тогда Поле = "ТерриториальноеОбразование";
	ИначеЕсли Имя = "СубъектыРФ" Тогда Поле = "СубъектРФ";

	КонецЕсли;
	Запрос = Новый Запрос;

// {{ semidockij - 27.10.2026 - Пояснения к методам

	Если ТипЗнч(Ссылка) <> Тип("СправочникСсылка.СтруктурныеЕдиницы") ИЛИ НЕ ЗначениеЗаполнено(Ссылка) Тогда

	Состояние.revision = Состояние.revision + 1;
	Маркер = Новый Структура("reference_guid,id,force_delete,revision",
		Строка(Ссылка.УникальныйИдентификатор()), ВнешнийID, ПринудительноеУдаление, Состояние.revision);
	ЗаблокироватьТехническую(Маркер.reference_guid, "КХ_СЕChange");
	ЗаписатьТехническую(Маркер.reference_guid, "КХ_СЕChange", Маркер);

// {{ semidockij - 27.10.2026 - Пояснения к методам
// Назначение: создаёт сохранённый снимок всего справочника магазинов для полной публикации.
// Как работает: резервирует revision, читает карточки и удаления, учитывает настроенные PC и сохраняет снимок сессии.
// Возвращает session_id; размер порции ограничен 1-500, существующая активная сессия продолжается, а не заменяется.
// }} semidockij - 27.10.2026
Функция НачатьПолнуюВыгрузку(РазмерПакета = 200) Экспорт
	Исходное = ЗарезервироватьВерсиюПолной();
	Если Исходное.full_active Тогда Возврат Исходное.full.session_id; КонецЕсли;
	РазмерПакета = Мин(500, Макс(1, Цел(РазмерПакета)));
	// Граница до чтения снимка. Последующие изменения имеют большую revision.
	Данные = КХ_ДанныеСтруктурныхЕдиниц.ПолучитьСостояниеМагазинов();
	Записи = ЗаписиИзДанных(Данные);
	ID = Новый Соответствие;
	Для Каждого Запись Из Записи Цикл ID.Вставить(Запись.id, Истина); КонецЦикла;
	// Локальному PC нужен пустой full и при исчезновении его магазина из общего списка.
	Для Каждого Маршрут Из Исходное.pc_routes Цикл
		Если ID.Получить(Маршрут.store_id) = Неопределено Тогда
			Записи.Добавить(Новый Структура("id,unit", Маршрут.store_id, Неопределено));
		КонецЕсли;
	КонецЦикла;
	СвояТранзакция = НЕ ТранзакцияАктивна();
	Если СвояТранзакция Тогда НачатьТранзакцию(); КонецЕсли;
	Попытка
		Состояние = СостояниеПодБлокировкой();
		ПроверитьПубликацию(Состояние);
		Если НЕ Состояние.full_active Тогда
			// Не допускаем, чтобы снимок со старыми маршрутами стал новой сессией после перенастройки.
			Если Состояние.settings_revision <> Исходное.settings_revision Тогда
				ВызватьИсключение "Маршруты изменились во время чтения справочника; повторите full";
			КонецЕсли;
			Полная = Новый Структура("session_id,batch_size,total_batches,next_sequence,source_revision",
				Строка(Новый УникальныйИдентификатор), РазмерПакета,
				Макс(1, Цел((Записи.Количество() + РазмерПакета - 1) / РазмерПакета)), 1, Исходное.revision);
			ЗаблокироватьТехническую(Полная.session_id, "КХ_СЕFull");
			ЗаписатьТехническую(Полная.session_id, "КХ_СЕFull", Новый Структура("entries", Записи));
			Состояние.full = Полная;
			Состояние.full_active = Истина;
			Состояние.full_requested = Ложь;
			ЗаписатьСостояние(Состояние);
		КонецЕсли;
		Результат = Состояние.full.session_id;
		Если СвояТранзакция Тогда ЗафиксироватьТранзакцию(); КонецЕсли;
	Исключение
		Если СвояТранзакция Тогда ОтменитьТранзакцию(); КонецЕсли;
		ВызватьИсключение;
	КонецПопытки;
	Возврат Результат;

// {{ semidockij - 27.10.2026 - Пояснения к методам
// Назначение: ставит в очередь очередную порцию сохранённого full справочника.
// Как работает: в транзакции читает снимок, регистрирует порцию и продвигает номер; после последней удаляет технический снимок.
// Возвращает признак подготовленной порции; готовность всех порций не означает их подтверждение бэками.
// }} semidockij - 27.10.2026
Функция ПодготовитьСледующийПолныйПакет() Экспорт
	// Всего сотни СЕ; чтение неизменяемого снимка и запись порции под одной блокировкой.
	СвояТранзакция = НЕ ТранзакцияАктивна();
	Если СвояТранзакция Тогда НачатьТранзакцию(); КонецЕсли;
	Попытка
		Состояние = СостояниеПодБлокировкой();
		ПроверитьПубликацию(Состояние);
		ЕстьПакет = Состояние.full_active;
		Если ЕстьПакет Тогда
			Полная = Состояние.full;
			Снимок = ПрочитатьТехническую(Полная.session_id, "КХ_СЕFull");
			Если Снимок = Неопределено Тогда ВызватьИсключение "Утрачен снимок full справочника СЕ"; КонецЕсли;
			Порция = Новый Массив;

// Назначение: распределяет одну порцию справочника по Opera и явно настроенным PC.
// Как работает: Opera получает общую порцию, каждый PC получает только запись своей СЕ в отдельном пакете.

Процедура ЗарегистрироватьПорцию(Состояние, Записи, Вид, Версия, Полная = Неопределено)

// Назначение: проверяет пакет справочника до его сохранения в исходящую очередь.
