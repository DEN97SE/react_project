Функция ПодготовитьШтрихКоды(ТаблицаШтрихКодов)

	Результат = Новый Соответствие;
	ГруппыЕдиниц = Новый Соответствие;
	ПорядокЕдиниц = Новый Массив;

	Для Каждого СтрокаШК Из ТаблицаШтрихКодов Цикл

		ШтрихКод = СокрЛП(Строка(СтрокаШК.ШтрихКод));
		Если ПустаяСтрока(ШтрихКод) Тогда
			Продолжить;
		КонецЕсли;
		Если НЕ ЗначениеЗаполнено(СтрокаШК.ЕдиницаИзмерения) Тогда
			ВызватьИсключение "ШК не связан с единицей измерения: " + ШтрихКод;
		КонецЕсли;

		ЕдиницаГруппировки = СтрокаШК.ЕдиницаИзмерения;
		Если СтрокаШК.ЭтоБазоваяЕдиница Тогда
			// Для базовой единицы товара выбирается один ШК из всех её вариантов.
			ЕдиницаГруппировки = "БазоваяЕдиница";
		КонецЕсли;
		КлючЕдиницы = ЗначениеВСтрокуВнутр(Новый Структура(
			"Номенклатура,Единица", СтрокаШК.Номенклатура, ЕдиницаГруппировки));
		Группа = ГруппыЕдиниц.Получить(КлючЕдиницы);
		Если Группа = Неопределено Тогда
			ОсновнойШК = СокрЛП(Строка(СтрокаШК.ОсновнойШтрихКод));
			// Отсутствие назначенного основного ШК не запрещает выгрузку товара.
			ТребуетсяОсновной = СтрокаШК.ЭтоБазоваяЕдиница И НЕ ПустаяСтрока(ОсновнойШК);
			Группа = Новый Структура(
				"Номенклатура,Единица,ТребуетсяОсновной,ОсновнойШК,Строка,ДатаУстановки,ШтрихКод,Неоднозначно",
				СтрокаШК.Номенклатура, СтрокаШК.ЕдиницаИзмерения, ТребуетсяОсновной,
				ОсновнойШК, Неопределено, Дата(1, 1, 1), "", Ложь);
			ГруппыЕдиниц.Вставить(КлючЕдиницы, Группа);
			ПорядокЕдиниц.Добавить(КлючЕдиницы);
		КонецЕсли;

		// У базовой единицы установленный основной код имеет приоритет над датой.
		Если Группа.ТребуетсяОсновной И ШтрихКод <> Группа.ОсновнойШК Тогда
			Продолжить;
		КонецЕсли;

		Если Группа.Строка = Неопределено Тогда
			Группа.Строка = СтрокаШК;
			Группа.ДатаУстановки = СтрокаШК.ДатаУстановки;
			Группа.ШтрихКод = ШтрихКод;
		ИначеЕсли СтрокаШК.ДатаУстановки > Группа.ДатаУстановки Тогда
			Группа.Строка = СтрокаШК;
			Группа.ДатаУстановки = СтрокаШК.ДатаУстановки;
			Группа.ШтрихКод = ШтрихКод;
			Группа.Неоднозначно = Ложь;
		ИначеЕсли СтрокаШК.ДатаУстановки = Группа.ДатаУстановки
			И (ШтрихКод <> Группа.ШтрихКод
				Или СтрокаШК.ЕдиницаИзмерения <> Группа.Строка.ЕдиницаИзмерения) Тогда
			Группа.Неоднозначно = Истина;
		КонецЕсли;

	КонецЦикла;

	Для Каждого КлючЕдиницы Из ПорядокЕдиниц Цикл
		Группа = ГруппыЕдиниц.Получить(КлючЕдиницы);
		Если Группа.Строка = Неопределено Тогда
			ВызватьИсключение "Основной ШК " + Группа.ОсновнойШК
				+ " не связан с базовой единицей товара " + Строка(Группа.Номенклатура);
		КонецЕсли;
		Если Группа.Неоднозначно Тогда
			ВызватьИсключение "У разных ШК единицы " + Строка(Группа.Единица)
				+ " товара " + Строка(Группа.Номенклатура)
				+ " одинаковая последняя дата записи; нужен однозначный последний код";
		КонецЕсли;

		МассивШтрихКодов = Результат.Получить(Группа.Номенклатура);
		Если МассивШтрихКодов = Неопределено Тогда
			МассивШтрихКодов = Новый Массив;
			Результат.Вставить(Группа.Номенклатура, МассивШтрихКодов);
		КонецЕсли;
		МассивШтрихКодов.Добавить(СформироватьШтрихкод(Группа.Строка));
	КонецЦикла;

	Возврат Результат;

КонецФункции
