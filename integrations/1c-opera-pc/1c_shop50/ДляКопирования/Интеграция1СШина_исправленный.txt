// Обработка очереди и отправка сообщений
Процедура ОтправитьСообщенияВШину(ВнутреннийКанал = Ложь) Экспорт
	
	ОчередьВыгрузки = ОчередьДляВыгрузки(ВнутреннийКанал);
	
	Для каждого Запись Из ОчередьВыгрузки Цикл
		
		Если ТипЗнч(Запись.Объект) = Тип("Строка") Тогда
			Если ПустаяСтрока(Запись.ВнешняяСистема) Тогда
				Параметры = ПереводВJSONДляШины.ЗаписьРегистраСведений_JSONСообщение(Запись.Объект);
			Иначе
				Объект = Запись.СтроковыйОбъект;
				
				Если ПустаяСтрока(Объект) Тогда
					Объект = Запись.Объект;
				КонецЕсли;
				
				Параметры = ПереводВJSONДляШины.ЗаписьВнешняяСистема_JSONСообщение(Объект, Запись.ВнешняяСистема);
			КонецЕсли;
		Иначе
			ОбъектМетаданных = Метаданные.НайтиПоТипу(ТипЗнч(Запись.Объект));
			
			Если ОбъектМетаданных = Неопределено Тогда
				Продолжить;
			КонецЕсли;
			
			ЭтоДокумент = Метаданные.Документы.Содержит(ОбъектМетаданных);
			ЭтоСправочник = Метаданные.Справочники.Содержит(ОбъектМетаданных);
			Если ЭтоДокумент Тогда
				Параметры = РегистрацияОбъектов1СШина.Документ_JSON_ТипСообщения(Запись.Объект);
			ИначеЕсли ЭтоСправочник Тогда
				Параметры = РегистрацияОбъектов1СШина.ВыгрузитьСправочникВ1сШину(Запись.Объект);
			КонецЕсли; 
		КонецЕсли;
		
		// отправим сообщение
		Если Параметры.Свойство("Ошибка") Тогда
			
			Статус = Новый Структура("ЕстьОшибки, ТекстОшибки", Истина, Параметры.Ошибка);
			
		Иначе
			
			Статус = Интеграция1СШинаПереопределяемый.ОтправитьСообщениеВШину(Параметры, Запись.УзелПолучатель, Запись.КаналСервисаИнтеграции);     
			
		КонецЕсли;
		
		ЗафиксироватьИзмененияВОчередиШины(Запись, Статус);	
		
	КонецЦикла;
	
КонецПроцедуры

Процедура ОтправитьСообщенияВШинуВнутреннийКанал() Экспорт
	
	ОтправитьСообщенияВШину(Истина);

Функция ОтправитьИсходящееСообщениеВШину(СервисИнтеграции, КаналИнтеграции, ТипСообщения, 
										 КодОтправителя, ТекстСообщения, Узлы = "", ДополнительныеПараметрыЗаголовкаСообщения = "") Экспорт  

	Статус = Новый Структура;
	Попытка
	
		ИсходящееСообщение = СервисыИнтеграции[СервисИнтеграции].СоздатьСообщение();
		ИсходящееСообщение.КодОтправителя = КодОтправителя;
		ИсходящееСообщение.КодПолучателя = Узлы;
		ИсходящееСообщение.Параметры.Вставить("ТипСообщения", ТипСообщения);

		// добавим дополнительные параметры если они есть
		Если ТипЗнч(ДополнительныеПараметрыЗаголовкаСообщения) = Тип("Структура") Тогда
			Для каждого КлючЗначение Из ДополнительныеПараметрыЗаголовкаСообщения Цикл
				ИсходящееСообщение.Параметры.Вставить(КлючЗначение.Ключ, КлючЗначение.Значение);
			КонецЦикла;	
		КонецЕсли;

		// Запишем тело сообщения
		Если СтрДлина(ТекстСообщения) > 0 Тогда
			ТелоСообщения = ИсходящееСообщение.ПолучитьТелоКакПоток();
			ИмяВременногоФайла = ПолучитьИмяВременногоФайла("tmp");
			ЗаписьТекста = Новый ЗаписьТекста(ИмяВременногоФайла, КодировкаТекста.ANSI);
			ЗаписьТекста.ЗаписатьСтроку(ТекстСообщения);
			ЗаписьТекста.Закрыть();
			
			ФайловыйПоток = ФайловыеПотоки.Открыть(ИмяВременногоФайла, РежимОткрытияФайла.Открыть, ДоступКФайлу.Чтение);
			ФайловыйПоток.КопироватьВ(ТелоСообщения);
			
			ТелоСообщения.СброситьБуферы();
			РазмерСообщения = ФайловыйПоток.Размер();
			ТелоСообщения.Закрыть();
			
			ИсходящееСообщение.Параметры.Вставить("РазмерСообщения", РазмерСообщения);
		КонецЕсли;
		СервисыИнтеграции[СервисИнтеграции][СокрЛП(КаналИнтеграции)].ОтправитьСообщение(ИсходящееСообщение);	

		СобытиеЖурналаРегистрации = "Сервисы интеграции.ОтправкаСообщения";
		ТекстСообщенияЖР = "Отправлено сообщение ID:" + ИсходящееСообщение.Идентификатор + Символы.ВК + "  для : "
			+ ИсходящееСообщение.КодПолучателя;
		ЗаписьЖурналаРегистрации(СобытиеЖурналаРегистрации, УровеньЖурналаРегистрации.Информация, , , ТекстСообщенияЖР);
		
		Статус.Вставить("ЕстьОшибки", Ложь);
		Статус.Вставить("ИдентификаторСообщения", ИсходящееСообщение.Идентификатор);
		
	Исключение

		СобытиеЖурналаРегистрации = "Сервисы интеграции.ОтправкаСообщения"; 
		ТекстОшибки = ПодробноеПредставлениеОшибки(ИнформацияОбОшибке());
		ТекстСообщенияЖР = ТекстОшибки + ". Не удалось отправить сообщение для узл(а/ов):" + Узлы;
		ЗаписьЖурналаРегистрации(СобытиеЖурналаРегистрации, УровеньЖурналаРегистрации.Ошибка, , , ТекстСообщенияЖР);
		Статус.Вставить("ЕстьОшибки", Истина);		
		Статус.Вставить("ТекстОшибки", ТекстОшибки);				
		
	КонецПопытки;
	
	Возврат Статус;

Функция СтруктуруЗаписиОчереди(Знач Источник, ИдентификаторОбъекта, Узел, КаналСервисаИнтеграции, ВнешняяСистема)
	
	СтруктураОчередиШины = Новый Структура;
	
	СтруктураОчередиШины.Вставить("ИдентификаторОбъекта", ИдентификаторОбъекта);
	СтруктураОчередиШины.Вставить("ИдентификаторСообщения", "");
	
	Если ТипЗнч(Источник) = Тип("Строка") И Не ПустаяСтрока(ВнешняяСистема) Тогда
		СтруктураОчередиШины.Вставить("Объект", ЗначениеВСтрокуВнутр(Новый Структура("ВнешняяСистема", ВнешняяСистема)));
		СтруктураОчередиШины.Вставить("СтроковыйОбъект", Источник);
	Иначе
		СтруктураОчередиШины.Вставить("Объект", Источник);
		СтруктураОчередиШины.Вставить("СтроковыйОбъект", "");
	КонецЕсли;
	
	СтруктураОчередиШины.Вставить("ДатаРегистрации", ТекущаяДатаСеанса());  
	СтруктураОчередиШины.Вставить("ДатаОтправки", Дата(1, 1, 1));
	СтруктураОчередиШины.Вставить("СтатусСообщения", Перечисления.СтатусыСообщенийШины.ОжидаетОтправки);
	СтруктураОчередиШины.Вставить("УзелПолучатель", Узел);
	СтруктураОчередиШины.Вставить("ТекстОшибки", "");
	СтруктураОчередиШины.Вставить("КаналСервисаИнтеграции", КаналСервисаИнтеграции);
	СтруктураОчередиШины.Вставить("ВнешняяСистема", ВнешняяСистема);	

	Возврат СтруктураОчередиШины;

	Если ЗаписьОчереди <> Неопределено Тогда
		
		НаборЗаписей = РегистрыСведений.ОчередьШины.СоздатьНаборЗаписей();
		НаборЗаписей.Отбор.ИдентификаторОбъекта.Установить(Запись.ИдентификаторОбъекта);
		НаборЗаписей.Отбор.УзелПолучатель.Установить(Запись.УзелПолучатель);
		
		НоваяСтрока = НаборЗаписей.Добавить();
		
		ЗаполнитьЗначенияСвойств(НоваяСтрока, ЗаписьОчереди);
		НоваяСтрока.ДатаОтправки = ТекущаяДатаСеанса();
		Если Статус.ЕстьОшибки Тогда
			НоваяСтрока.ТекстОшибки = Статус.ТекстОшибки;
			НоваяСтрока.СтатусСообщения = Перечисления.СтатусыСообщенийШины.Ошибка;
		Иначе
			НоваяСтрока.ИдентификаторСообщения = Статус.ИдентификаторСообщения;
			НоваяСтрока.СтатусСообщения = Перечисления.СтатусыСообщенийШины.Отправлено;	
			НоваяСтрока.ТекстОшибки = "";
		КонецЕсли;     
		НаборЗаписей.Записать();
		
	Иначе   
		
		СобытиеЖурналаРегистрации = "Сервисы интеграции.ИзмененийОчереди";
		ТекстОшибки = "Регистрация отправки запись, отсутствующей в регистре";
		ЗаписьЖурналаРегистрации(СобытиеЖурналаРегистрации,
			УровеньЖурналаРегистрации.Ошибка, , ,
			ТекстОшибки);
		
	КонецЕсли;
