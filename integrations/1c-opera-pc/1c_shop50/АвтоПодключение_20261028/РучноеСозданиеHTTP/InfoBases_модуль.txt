// HTTP-сервис находится в Основной и использует существующий справочник Exch.
// Все HTTP-методы доступны только администраторам. Второй справочник в проекте не создаётся.
// Ключи сохраняет центральный исполнитель в защищённом DPAPI-кэше; секрет в Exch не записывается.
// Массовый исполнитель работает последовательно. Параллельная регистрация одного GUID не поддерживается этим кандидатом.

структура ЗаявкаМагазина
    обз пер Код: Строка
    обз пер Наименование: Строка
    пер Описание: Строка = ""
;

структура ОтветРегистрации
    обз пер Код: Строка
    обз пер Наименование: Строка
    обз пер clientid: Строка
    обз пер clientsecret: Строка
    пер created: Булево = Ложь
    пер version: Строка = "opera-pc-enrollment-2"
;

структура ОшибкаРегистрации
    обз пер error: Строка
;

структура ОтветВерсии
    пер version: Строка = "opera-pc-enrollment-2"
;

// Назначение: позволяет проверить публикацию и доступ к нужной версии сервиса без выдачи секретов.
метод ПроверитьВерсию(Запрос: HttpСервисЗапрос)
    УстановитьЗаголовкиОтвета(Запрос)
    Запрос.Ответ.УстановитьТело(СериализацияJson.ЗаписатьОбъект(новый ОтветВерсии()))
;

// Назначение: проверяет GUID, не позволяя использовать произвольный код другого назначения.
// Пустая строка означает некорректный либо нулевой UUID; ничего не создаётся.
метод НормализоватьКод(Код: Строка): Строка
    попытка
        пер Результат = (новый Ууид(Код)).ВСтроку()
        если Результат == "00000000-0000-0000-0000-000000000000"
            возврат ""
        ;
        возврат Результат
    поймать Искл: Исключение
        возврат ""
    ;
;

// Назначение: формирует контролируемую ошибку без возврата токенов и внутреннего текста исключения.
метод ОтветитьОшибкой(Запрос: HttpСервисЗапрос, Статус: Число, КодОшибки: Строка)
    УстановитьЗаголовкиОтвета(Запрос)
    Запрос.Ответ.КодСтатуса = Статус
    Запрос.Ответ.УстановитьТело(СериализацияJson.ЗаписатьОбъект(новый ОшибкаРегистрации(КодОшибки)))
;

// Назначение: задаёт JSON/UTF-8 и исключает кэширование ответов с параметрами доступа.
метод УстановитьЗаголовкиОтвета(Запрос: HttpСервисЗапрос)
    Запрос.Ответ.Заголовки.Установить("Content-Type", "application/json; charset=utf-8")
    Запрос.Ответ.Заголовки.Установить("Cache-Control", "no-store")
;

// Назначение: не создаёт вторую ИС, если GUID уже занят существующим участником.
метод УчастникСуществует(Код: Строка): Булево
    пер Данные = Запрос{
        ВЫБРАТЬ Т.Код
        ИЗ Exch КАК Т
    }
    для СтрокаТ из Данные.Выполнить()
        если СтрокаТ.Код == Код
            возврат Истина
        ;
    ;
    возврат Ложь
;

// Назначение: один раз создаёт пользователя, ИС Exch и членство в двух процессах.
// Секрет возвращается только в ответе на создание. Повторный запрос не создаёт ключи и возвращает 409.
// При потере первого ответа до сохранения центрального кэша требуется отдельное восстановление доступа.
метод ЗарегистрироватьМагазин(Запрос: HttpСервисЗапрос)
    пер Вход = новый ЗаявкаМагазина("", "")
    попытка
        пер НастройкиJson = новый НастройкиЧтенияОбъектовJson()
        НастройкиJson.ИгнорироватьНеизвестныеСвойства = Ложь
        Вход = СериализацияJson.ПрочитатьОбъект(Запрос.Тело, Тип<ЗаявкаМагазина>, НастройкиJson)
    поймать Искл: Исключение
        ОтветитьОшибкой(Запрос, 400, "invalid_json")
        возврат
    ;
    пер Код = НормализоватьКод(Вход.Код)
    если Код.Пусто() или Вход.Наименование.Пусто()
        ОтветитьОшибкой(Запрос, 400, "invalid_store_guid_or_name")
        возврат
    ;

    исп Транзакция = Транзакции.Начать()
    пер Ответ = новый ОтветРегистрации("", "", "", "")
    попытка
        если УчастникСуществует(Код)
            Транзакция.Откатить()
            ОтветитьОшибкой(Запрос, 409, "existing_participant_use_central_cache_or_recover")
            возврат
        ;
        пер Список = СпискиПользователей.ПолучитьСписокПоУмолчанию().Ид
        пер ОписаниеПользователя = новый ОписаниеПользователяСервиса(Список, Вход.Наименование, Код)
        ПользователиСервиса.Создать(ОписаниеПользователя)
        пер ДанныеДоступа = ПользователиСервиса.СгенерироватьДанныеЗапросаТокенаДоступа(
            ОписаниеПользователя.Ид, "Регистрация базы магазина Opera и PC")
        пер Пользователь = Пользователи.Подключить(ОписаниеПользователя.Ид)

        пер ИС = новый Exch.Объект()
        ИС.Код = Код
        ИС.Наименование = Вход.Наименование
        ИС.Описание = Вход.Описание
        ИС.ПользовательПоУмолчанию = Пользователь.Ссылка
        ИС.Записать()

        Opera.ДобавитьУчастника(ИС, ИС.ПользовательПоУмолчанию)
        Opera.ДобавитьУчастниковВГруппы("MagazyniOpera", ИС)
        PC.ДобавитьУчастника(ИС, ИС.ПользовательПоУмолчанию)
        PC.ДобавитьУчастниковВГруппы("MagazinyPriceChecker", ИС)

        Ответ = новый ОтветРегистрации(Код, Вход.Наименование, ДанныеДоступа.Ид, ДанныеДоступа.Секрет)
        Ответ.created = Истина
        Транзакция.Фиксировать()
    поймать Искл: Исключение
        Транзакция.Откатить()
        ОтветитьОшибкой(Запрос, 500, "registration_failed_check_server_locally")
        возврат
    ;
    // Формирование HTTP-ответа вынесено за пределы отката уже зафиксированной транзакции.
    УстановитьЗаголовкиОтвета(Запрос)
    Запрос.Ответ.УстановитьТело(СериализацияJson.ЗаписатьОбъект(Ответ))
;
