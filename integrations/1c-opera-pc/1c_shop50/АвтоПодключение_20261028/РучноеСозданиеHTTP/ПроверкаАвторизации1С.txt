// Выполнить в консоли кода 1С «На сервере».
// API-ключ пользователя с административным доступом к Opera_PC.
ИдентификаторКлиента = "ВСТАВЬ_CLIENT_ID";
СекретКлиента = "REDACTED_CREDENTIAL_7";

Попытка
    Если ИдентификаторКлиента = "ВСТАВЬ_CLIENT_ID"
        Или СекретКлиента = "REDACTED_CREDENTIAL_7"
        Или ПустаяСтрока(ИдентификаторКлиента)
        Или ПустаяСтрока(СекретКлиента) Тогда
        ВызватьИсключение "Заполни идентификатор и секрет API-клиента";
    КонецЕсли;

    Соединение = Новый HTTPСоединение("program58", 9090, , , , 30);
    СтрокаBase64 = Base64Строка(ПолучитьДвоичныеДанныеИзСтроки(
        ИдентификаторКлиента + ":" + СекретКлиента));
    СтрокаBase64 = СтрЗаменить(СтрокаBase64, Символы.ПС, "");
    СтрокаBase64 = СтрЗаменить(СтрокаBase64, Символы.ВК, "");

    ЗапросАвторизации = Новый HTTPЗапрос("/auth/oidc/token");
    ЗапросАвторизации.Заголовки.Вставить(
        "Authorization", "Basic " + СтрокаBase64);
    ЗапросАвторизации.Заголовки.Вставить(
        "Content-Type", "application/x-www-form-urlencoded");
    ЗапросАвторизации.УстановитьТелоИзСтроки(
        "grant_type=client_credentials");

    ОтветАвторизации = Соединение.ОтправитьДляОбработки(ЗапросАвторизации);
    Сообщить("Авторизация HTTP: " + Строка(ОтветАвторизации.КодСостояния));
    Если ОтветАвторизации.КодСостояния <> 200 Тогда
        ВызватьИсключение "Получение токена не выполнено. Проверь API-ключ и адрес авторизации";
    КонецЕсли;

    ЧитательJSON = Новый ЧтениеJSON;
    ЧитательJSON.УстановитьСтроку(ОтветАвторизации.ПолучитьТелоКакСтроку());
    ДанныеАвторизации = ПрочитатьJSON(ЧитательJSON);
    ЧитательJSON.Закрыть();

    Токен = "";
    Если ТипЗнч(ДанныеАвторизации) <> Тип("Структура") Тогда
        ВызватьИсключение "Неожиданный формат ответа авторизации";
    КонецЕсли;
    Если Не ДанныеАвторизации.Свойство("id_token", Токен) Тогда
        ВызватьИсключение "В ответе отсутствует id_token";
    КонецЕсли;
    Если ТипЗнч(Токен) <> Тип("Строка") Тогда
        ВызватьИсключение "Неожиданный тип id_token";
    КонецЕсли;
    Если ПустаяСтрока(Токен) Тогда
        ВызватьИсключение "Сервер вернул пустой id_token";
    КонецЕсли;

    ЗапросПроверки = Новый HTTPЗапрос(
        "/applications/Opera-PC/api/infobases/health");
    ЗапросПроверки.Заголовки.Вставить("Authorization", "Bearer " + Токен);
    ЗапросПроверки.Заголовки.Вставить("Accept", "application/json");

    ОтветПроверки = Соединение.Получить(ЗапросПроверки);
    Сообщить("Health HTTP: " + Строка(ОтветПроверки.КодСостояния));
    Сообщить("Content-Type: "
        + Строка(ОтветПроверки.Заголовки.Получить("Content-Type")));
    Сообщить(Лев(ОтветПроверки.ПолучитьТелоКакСтроку(), 4000));
Исключение
    Сообщить("Ошибка проверки: " + ОписаниеОшибки());
КонецПопытки;
