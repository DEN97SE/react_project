Функция ПодготовитьШтрихКоды(ТаблицаШтрихКодов)
	
	Результат = Новый Соответствие;
	
	Для Каждого СтрокаШтрихКода Из ТаблицаШтрихКодов Цикл
		
		МассивШтрихКодов = Результат.Получить(
		СтрокаШтрихКода.Номенклатура);
		
		Если МассивШтрихКодов = Неопределено Тогда
			
			МассивШтрихКодов = Новый Массив;
			
			Результат.Вставить(
			СтрокаШтрихКода.Номенклатура,
			МассивШтрихКодов);
			
		КонецЕсли;
		
		ДанныеШтрихКода = СформироватьШтрихкод(
		СтрокаШтрихКода);
		
		МассивШтрихКодов.Добавить(
		ДанныеШтрихКода);
		
	КонецЦикла;
	
	Возврат Результат;

Функция СформироватьТовар(СтрокаНоменклатуры, ШтрихКодыПоНоменклатуре, ЦеныПоНоменклатуре, ОстаткиПоНоменклатуре) 
	
	Товар = Новый Структура;
	
	Товар.Вставить("nomenclature_id", СокрЛП(Строка(СтрокаНоменклатуры.Идентификатор)));
	
	Товар.Вставить("name", Строка(СтрокаНоменклатуры.Наименование));
	
	Если ЗначениеЗаполнено(СтрокаНоменклатуры.БазоваяЕдиницаИзмерения) Тогда
		
		Товар.Вставить("base_unit",	Строка(СтрокаНоменклатуры.БазоваяЕдиницаИзмерения));
		
	КонецЕсли;
	
	Если ЗначениеЗаполнено(СтрокаНоменклатуры.СтранаПроисхождения) Тогда
		
		Товар.Вставить("country", Строка(СтрокаНоменклатуры.СтранаПроисхождения));
		
	КонецЕсли;
	
	Если ЗначениеЗаполнено(СтрокаНоменклатуры.Производитель) Тогда
		
		Товар.Вставить("manufacturer", Строка(СтрокаНоменклатуры.Производитель));
		
	КонецЕсли;
	
	Если ЗначениеЗаполнено(СтрокаНоменклатуры.ГруппаНоменклатуры) Тогда
		
		Товар.Вставить("category", Строка(СтрокаНоменклатуры.ГруппаНоменклатуры));
		
	КонецЕсли;
	
	Товар.Вставить("is_own_production", СтрокаНоменклатуры.СобственноеПроизводство);
	
	Если ЗначениеЗаполнено(СтрокаНоменклатуры.Состав) Тогда
		
		Товар.Вставить("composition", Строка(СтрокаНоменклатуры.Состав));
		
	КонецЕсли;
	
	КБЖУ = СформироватьКБЖУ(
	СтрокаНоменклатуры.Ккал,
	СтрокаНоменклатуры.Белки,
	СтрокаНоменклатуры.Жиры,
	СтрокаНоменклатуры.Углеводы);
	
	Товар.Вставить("nutrition", КБЖУ);
	
	Цена = ЦеныПоНоменклатуре.Получить(СтрокаНоменклатуры.Номенклатура);
	
	Если Цена <> Неопределено Тогда
		
		Товар.Вставить("price",	СформироватьДанныеЦены(Цена));
		
	КонецЕсли;
	
	Количество = ОстаткиПоНоменклатуре.Получить(
	СтрокаНоменклатуры.Номенклатура);
	
	Если Количество = Неопределено Тогда
		Количество = 0;
	КонецЕсли;
	
	Товар.Вставить("stock", СформироватьДанныеОстатка(Количество));
	
	ШтрихКоды = ШтрихКодыПоНоменклатуре.Получить(СтрокаНоменклатуры.Номенклатура);
	
	Если ШтрихКоды = Неопределено Тогда
		ШтрихКоды = Новый Массив;
	КонецЕсли;
	
	Товар.Вставить("barcodes", ШтрихКоды);
	
	// Акции добавим следующим шагом.
	Товар.Вставить("promos", Новый Массив);
	
	Возврат Товар;
