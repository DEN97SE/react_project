Функция СформироватьПолнуюВыгрузку(СтруктурнаяЕдиница) Экспорт

	Если НЕ ЗначениеЗаполнено(СтруктурнаяЕдиница) Тогда
		Возврат Неопределено;
	КонецЕсли;

	МоментВыгрузки = ТекущаяДатаСеанса();

	Данные = ПолучитьДанныеДляВыгрузки(СтруктурнаяЕдиница, МоментВыгрузки);

	ШтрихКодыПоНоменклатуре = ПодготовитьШтрихКоды(Данные.ШтрихКоды);
	ЦеныПоНоменклатуре = ПодготовитьЦены(Данные.Цены);
	ОстаткиПоНоменклатуре = ПодготовитьОстатки(Данные.Остатки);
	
	ТаблицаАкций = ПолучитьАкции(СтруктурнаяЕдиница, МоментВыгрузки);
	АкцииПоНоменклатуре = ПодготовитьАкции(ТаблицаАкций);
	
	Товары = Новый Массив;
	
	Сообщить("Всего строк: " + Данные.Номенклатура.Количество());
	
	НоменклатураПроверка = Новый Соответствие;
	
	Для Каждого СтрокаНоменклатуры Из Данные.Номенклатура Цикл
		НоменклатураПроверка.Вставить(СтрокаНоменклатуры.Номенклатура, Истина);
	КонецЦикла;
	
	Сообщить("Уникальных товаров: " + НоменклатураПроверка.Количество());
	
	Для Каждого СтрокаНоменклатуры Из Данные.Номенклатура Цикл

		Товар = СформироватьТовар(СтрокаНоменклатуры,
		ШтрихКодыПоНоменклатуре,
		ЦеныПоНоменклатуре,
		ОстаткиПоНоменклатуре,
		АкцииПоНоменклатуре,
		МоментВыгрузки);

		Если Товар <> Неопределено Тогда
			Товары.Добавить(Товар);
		КонецЕсли;

	КонецЦикла;

	Возврат Товары;

Функция ПодготовитьШтрихКоды(ТаблицаШтрихКодов)

	Результат = Новый Соответствие;

	Для Каждого СтрокаШтрихКода Из ТаблицаШтрихКодов Цикл

		МассивШтрихКодов = Результат.Получить(СтрокаШтрихКода.Номенклатура);

		Если МассивШтрихКодов = Неопределено Тогда
			МассивШтрихКодов = Новый Массив;
			Результат.Вставить(СтрокаШтрихКода.Номенклатура, МассивШтрихКодов);
		КонецЕсли;

		ДанныеШтрихКода = СформироватьШтрихкод(СтрокаШтрихКода);
		МассивШтрихКодов.Добавить(ДанныеШтрихКода);

	КонецЦикла;

	Возврат Результат;

Функция ПодготовитьЦены(ТаблицаЦен)

	Результат = Новый Соответствие;

	Для Каждого СтрокаЦены Из ТаблицаЦен Цикл

		ТекущаяЦена = Результат.Получить(СтрокаЦены.Номенклатура);

		Если ТекущаяЦена <> Неопределено И ТекущаяЦена <> СтрокаЦены.Цена Тогда
			ВызватьИсключение "Получено несколько различных цен для номенклатуры: " + Строка(СтрокаЦены.Номенклатура);
		КонецЕсли;

		Результат.Вставить(СтрокаЦены.Номенклатура, СтрокаЦены.Цена);

	КонецЦикла;

	Возврат Результат;

		Если МассивАкций = Неопределено Тогда
			МассивАкций = Новый Массив;
			Результат.Вставить(СтрокаАкции.Номенклатура, МассивАкций);
		КонецЕсли;

		Акция = СформироватьАкцию(СтрокаАкции);

		Если Акция <> Неопределено Тогда
			МассивАкций.Добавить(Акция);
		КонецЕсли;

	КонецЦикла;

	Возврат Результат;

КонецФункции

Функция СформироватьТовар(СтрокаНоменклатуры, ШтрихКодыПоНоменклатуре, ЦеныПоНоменклатуре,
	ОстаткиПоНоменклатуре, АкцииПоНоменклатуре, МоментВыгрузки)

	Идентификатор = СокрЛП(Строка(СтрокаНоменклатуры.Идентификатор));
	Наименование = СокрЛП(Строка(СтрокаНоменклатуры.Наименование));

	Если ПустаяСтрока(Идентификатор) Или ПустаяСтрока(Наименование) Тогда
		Возврат Неопределено;
	КонецЕсли;

	Товар = Новый Структура;

	Товар.Вставить("nomenclature_id", Идентификатор);
	Товар.Вставить("name", Наименование);

	Если НЕ ПустаяСтрока(СокрЛП(Строка(СтрокаНоменклатуры.НаименованиеБазовойЕдиницыИзмерения))) Тогда
		Товар.Вставить("base_unit", Строка(СтрокаНоменклатуры.НаименованиеБазовойЕдиницыИзмерения));
	КонецЕсли;

	Если НЕ ПустаяСтрока(СокрЛП(Строка(СтрокаНоменклатуры.НаименованиеСтраныПроисхождения))) Тогда
		Товар.Вставить("country", Строка(СтрокаНоменклатуры.НаименованиеСтраныПроисхождения));
	КонецЕсли;

	Если ЕстьЗначениеДляJSON(СтрокаНоменклатуры.Производитель) Тогда
		Товар.Вставить("manufacturer", Строка(СтрокаНоменклатуры.Производитель));
	КонецЕсли;

	Если НЕ ПустаяСтрока(СокрЛП(Строка(СтрокаНоменклатуры.НаименованиеГруппыНоменклатуры))) Тогда
		Товар.Вставить("category", Строка(СтрокаНоменклатуры.НаименованиеГруппыНоменклатуры));
	КонецЕсли;

	Товар.Вставить("is_own_production", СтрокаНоменклатуры.СобственноеПроизводство);

	Если ЕстьЗначениеДляJSON(СтрокаНоменклатуры.Состав) Тогда
		Товар.Вставить("composition", Строка(СтрокаНоменклатуры.Состав));
	КонецЕсли;

	Товар.Вставить("nutrition", СформироватьКБЖУ(СтрокаНоменклатуры.Ккал, СтрокаНоменклатуры.Белки, СтрокаНоменклатуры.Жиры, СтрокаНоменклатуры.Углеводы));

	Цена = ЦеныПоНоменклатуре.Получить(СтрокаНоменклатуры.Номенклатура);

	Если Цена <> Неопределено Тогда
		Товар.Вставить("price", СформироватьДанныеЦены(Цена));
	КонецЕсли;

	Количество = ОстаткиПоНоменклатуре.Получить(СтрокаНоменклатуры.Номенклатура);

	Если Количество = Неопределено Тогда
		Количество = 0;
	КонецЕсли;

	Товар.Вставить("stock", СформироватьДанныеОстатка(Количество, МоментВыгрузки));

	ШтрихКоды = ШтрихКодыПоНоменклатуре.Получить(СтрокаНоменклатуры.Номенклатура);

	Если ШтрихКоды = Неопределено Тогда
		ШтрихКоды = Новый Массив;
	КонецЕсли;

	Товар.Вставить("barcodes", ШтрихКоды);
	
	Акции = АкцииПоНоменклатуре.Получить(СтрокаНоменклатуры.Номенклатура);
	
	Если Акции = Неопределено Тогда
		Акции = Новый Массив;
	КонецЕсли;
	
	Товар.Вставить("promos", Акции);
	
	Возврат Товар;

КонецФункции

Функция СформироватьКБЖУ(Ккал, Белки, Жиры, Углеводы)

		Возврат Null;
	КонецЕсли;

	Результат = Новый Структура;

	Если ЕстьЗначениеДляJSON(Ккал) Тогда
		Результат.Вставить("calories", ФорматDecimal(Ккал, 2));

		Результат.Вставить("proteins", ФорматDecimal(Белки, 2));

	Если ЕстьЗначениеДляJSON(Жиры) Тогда
		Результат.Вставить("fats", ФорматDecimal(Жиры, 2));
	КонецЕсли;

	Если ЕстьЗначениеДляJSON(Углеводы) Тогда
		Результат.Вставить("carbohydrates", ФорматDecimal(Углеводы, 2));
	КонецЕсли;

	КонецЕсли;

Функция СформироватьАкцию(СтрокаАкции)

	Если Не ЗначениеЗаполнено(СтрокаАкции.ТипАкции) Тогда
		Возврат Неопределено;
	КонецЕсли;

	Если Не ЕстьЗначениеДляJSON(СтрокаАкции.ЦенаАкции) Тогда
		Возврат Неопределено;
	КонецЕсли;

	Акция = Новый Структура;

	ТипАкции = СокрЛП(Строка(СтрокаАкции.ТипАкции));

	Если ТипАкции = "special_price" Тогда

		Акция.Вставить("type", "special_price");
		Акция.Вставить("promo_price", ФорматDecimal(СтрокаАкции.ЦенаАкции, 2));

		Если ЗначениеЗаполнено(СтрокаАкции.ДатаНачала) Тогда
			Акция.Вставить("starts_at", ФорматДатаJSON(СтрокаАкции.ДатаНачала));
		КонецЕсли;

		Если ЗначениеЗаполнено(СтрокаАкции.ДатаОкончания) Тогда
			Акция.Вставить("ends_at", ФорматДатаJSON(СтрокаАкции.ДатаОкончания));
		КонецЕсли;

	ИначеЕсли ТипАкции = "m_for_n" Тогда

		Акция.Вставить("type", "m_for_n");
		Акция.Вставить("promo_price", ФорматDecimal(СтрокаАкции.ЦенаАкции, 2));

		Если ЕстьЗначениеДляJSON(СтрокаАкции.Количество) Тогда
			Акция.Вставить("required_quantity", СтрокаАкции.Количество);
		КонецЕсли;

		Если ЗначениеЗаполнено(СтрокаАкции.ДатаНачала) Тогда
			Акция.Вставить("starts_at", ФорматДатаJSON(СтрокаАкции.ДатаНачала));
		КонецЕсли;

		Если ЗначениеЗаполнено(СтрокаАкции.ДатаОкончания) Тогда
			Акция.Вставить("ends_at", ФорматДатаJSON(СтрокаАкции.ДатаОкончания));
		КонецЕсли;

	Иначе
		Возврат Неопределено;
	КонецЕсли;

	Возврат Акция;

Функция ЕстьЗначениеДляJSON(Значение)

	Если Значение = Неопределено Тогда
		Возврат Ложь;
	КонецЕсли;

	Если ТипЗнч(Значение) = Тип("Null") Тогда
		Возврат Ложь;
	КонецЕсли;

	Если ТипЗнч(Значение) = Тип("Строка") Тогда
		Возврат НЕ ПустаяСтрока(СокрЛП(Значение));
	КонецЕсли;

	Возврат Истина;

КонецФункции

Функция ФорматDecimal(Значение, КоличествоЗнаков)

	Если Значение = 0 Тогда
		Если КоличествоЗнаков = 0 Тогда
			Возврат "0";
		Иначе
			Возврат "0." + Лев("000000000000000", КоличествоЗнаков);
		КонецЕсли;
	КонецЕсли;
	
	СтрокаФормата = "ЧДЦ=" + Строка(КоличествоЗнаков) + "; ЧГ=0";
	Результат = Формат(Значение, СтрокаФормата);
	Результат = СтрЗаменить(Результат, ",", ".");
	
	Возврат СокрЛП(Результат);  
	
КонецФункции

	Возврат Формат(Дата, "ДФ=yyyy-MM-dd") + "T" + Формат(Дата, "ДФ=HH:mm:ss") + "+07:00";
