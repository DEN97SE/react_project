// Параметры:
// ТипСообщения - строка
// ВходящееСообщение - строка JSON
// Возвращаемое значение - структура:
//	Результат - булево. признак успешного выполнения
//	ТекстОшибки - строка. заполняется, если результат не равен истина
Функция ЗагрузкаВходящегоСообщения(ТипСообщения, ВходящееСообщение) Экспорт
	
	РезультатВыполнения = Новый Структура("Результат, ТекстОшибки", Истина, "");
	
	Попытка
		
		ПоляДата = РеквизитыДатаВСообщении(ТипСообщения);
		Данные = ТекстJSONВОбъект(ВходящееСообщение, , ПоляДата);  
			
		Если ТипСообщения = "SendETRN" Тогда // Пришла эТрН
			ЗагрузкаЭТРН_Астор(Данные);
		ИначеЕсли ТипСообщения = "StatusETRN" Тогда // обновлен статус эТрН
			ЗагрузкаСтатусЭТРН(Данные);
		ИначеЕсли ТипСообщения = "ConformityGLN" Тогда
			ЗагрузкаСоответствияGLN(Данные);
		ИначеЕсли ТипСообщения = "ORDER" Тогда
			РезультатВыполнения = ЗагрузкаЗаказаПокупателя(Данные);
		ИначеЕсли ТипСообщения = "DESADV" Тогда
			РезультатВыполнения = ЗагрузкаУведомленияОбОтгрузке(Данные);
		ИначеЕсли ТипСообщения = "RECADV" Тогда
			РезультатВыполнения = ЗагрузкаУведомленияОПриемке(Данные);
		ИначеЕсли ТипСообщения = "ORDRSP" Тогда
			РезультатВыполнения = ЗагрузкаПодтвержденияЗаказа(Данные);
		ИначеЕсли ТипСообщения = "DataEDO" ИЛИ ТипСообщения = "InternalDataEDO" Тогда
			РезультатВыполнения = ЗагрузкаДанныхУПД(Данные);
		ИначеЕсли ТипСообщения = "ARMSupervisor" Тогда
			ЗагрузкаИзАРМСупервайзера(Данные);
		ИначеЕсли ТипСообщения = "Quickly" Тогда
			ЗагрузкаКТОизБыстродела(Данные);
		Иначе
			РезультатВыполнения.Результат = Ложь;
			РезультатВыполнения.ТекстОшибки = "Неизвестный тип сообщения";
		КонецЕсли;
		
	Исключение
		РезультатВыполнения.Результат = Ложь;
		РезультатВыполнения.ТекстОшибки = ОписаниеОшибки();
	КонецПопытки;
	
	Возврат РезультатВыполнения;
		
КонецФункции

// Получить из текста JSON структуру.
// 
// Параметры:
// 	ТекстJSON                    - Строка - Текст JSON.
// 	ПреобразовыватьВСоответствие - Булево - Признак преобразования в соответствие.
//  СвойстваДата 				 - Строка - перечисление реквизитов с типом дата через запятую, без пробелов
// Возвращаемое значение:
// 	Структура, Неопределено - Результат преобразования JSON.
Функция ТекстJSONВОбъект(ТекстJSON, ПреобразовыватьВСоответствие = Ложь, СвойстваДата = Неопределено) Экспорт
	
	Чтение = Новый ЧтениеJSON;
	Чтение.УстановитьСтроку(ТекстJSON);
	
	Попытка
		Если СвойстваДата = Неопределено Тогда
			РезультатРазбора = ПрочитатьJSON(Чтение, ПреобразовыватьВСоответствие);
		Иначе
			РезультатРазбора = ПрочитатьJSON(Чтение, ПреобразовыватьВСоответствие, СвойстваДата);			
		КонецЕсли;
	Исключение
		РезультатРазбора = Неопределено;
	КонецПопытки;
	
	Возврат РезультатРазбора;
	
КонецФункции

// Получить JSON из структуры
// Параметры:
// 	Данные - структура с заполненными полями
// Возвращаемое значение:
// 	строка - ТекстJSON
Функция ПреобразоватьВСтрокуJson(Данные)
	
	НастройкаСериализацииJSON = Новый НастройкиСериализацииJSON;
	НастройкаСериализацииJSON.ВариантЗаписиДаты = ВариантЗаписиДатыJSON.ЛокальнаяДата;
	НастройкаСериализацииJSON.ФорматСериализацииДаты = ФорматДатыJSON.ISO;
	
	тЗаписьJSON = Новый ЗаписьJSON;
	тПараметрыJSON = Новый ПараметрыЗаписиJSON(ПереносСтрокJSON.Нет, " ", Истина);  
	тЗаписьJSON.УстановитьСтроку(тПараметрыJSON); 
	ЗаписатьJSON(тЗаписьJSON, Данные, НастройкаСериализацииJSON);
	ДанныеJSON = тЗаписьJSON.Закрыть();
	
	Возврат ДанныеJSON;
	
КонецФункции

// Параметры:
// ТипСообщения - строка, тип входщего сообщения для загрузки
// Возвращаемое значение - строка с перечислением реквизитов-дата или Неопределено
Функция РеквизитыДатаВСообщении(ТипСообщения)
	
	МассивРеквизитовДата = МассивРеквизитыДатаПоТипу(ТипСообщения);
	
	Если МассивРеквизитовДата.Количество() = 0 Тогда
		РеквизитДата = Неопределено;
		
	Иначе
		РеквизитДата = "";
		Для Каждого ИмяРеквизита Из МассивРеквизитовДата Цикл 
			Если ЗначениеЗаполнено(РеквизитДата) Тогда
				РеквизитДата = РеквизитДата + ",";
			КонецЕсли;
			РеквизитДата = РеквизитДата + ИмяРеквизита;			
		КонецЦикла;
	КонецЕсли;
	
	Возврат РеквизитДата;
	
КонецФункции

// Параметры:
// ОбъектСсылка - ссылка, полученная по идентификатору
// возвращаемое значение - булево. ложь, если <Объект не найден>, иначе истина
Функция ОбъектСуществует(ОбъектСсылка) Экспорт
	
	Если СтрНайти(Строка(ОбъектСсылка), "<Объект не найден>") > 0 Тогда
		Возврат Ложь;
	Иначе
		Возврат Истина;
	КонецЕсли;
	
КонецФункции

// Параметры:
// ИмяСправочника - строка, имя справочника в метаданных
// ТекстЗначение - строка, по которой ищется элемент справочника: сначала предопределенный, затем по наименованию
// возвращает элемент указанного справочника
Функция ЗначениеСправочника(ИмяСправочника, ТекстЗначение) Экспорт
	
	Если Не ЗначениеЗаполнено(ТекстЗначение) Тогда
		Возврат Справочники[ИмяСправочника].ПустаяСсылка();
	КонецЕсли;
	
	Попытка // ищем предопределенное
		Значение = Справочники[ИмяСправочника][ТекстЗначение];
	Исключение
		Значение = Справочники[ИмяСправочника].НайтиПоНаименованию(ТекстЗначение, Истина);
	КонецПопытки;
	
	Возврат Значение;
		
КонецФункции

// Параметры:
// ИмяПеречисления - строка, имя перечисления в метаданных
// ТекстЗначение - строка, по которой ищется элемент перечисления: сначала по имени, затем по синониму
// возвращает значение указанного перечисления
Функция ЗначениеПеречисления(ИмяПеречисления, ТекстЗначение) Экспорт
	
	Если Не ЗначениеЗаполнено(ТекстЗначение) Тогда
		Возврат Перечисления[ИмяПеречисления].ПустаяСсылка();
	КонецЕсли;
	
	Попытка // ищем по имени метаданных
		Значение = Перечисления[ИмяПеречисления][ТекстЗначение];
	Исключение
		Для Каждого ЗначениеПеречисления Из Перечисления[ИмяПеречисления] Цикл
			Если Строка(ЗначениеПеречисления) = ТекстЗначение Тогда
				Значение = ЗначениеПеречисления;
				Прервать;
			КонецЕсли;
		КонецЦикла;
	КонецПопытки;
	
	Возврат Значение;                         
		
КонецФункции

// Параметры:
// ИНН - строка, ИНН организации
// КПП - строка, КПП организации
// Возвращаеиоме значение - ссылка на справочник Организации или Неопределено
Функция ОрганизацияПоИНН_КПП(ИНН, КПП) Экспорт
	
	ЗапросИННКПП = Новый Запрос;
	ЗапросИННКПП.Текст =
		"ВЫБРАТЬ
		|	Организации.Ссылка КАК Ссылка

		|	Справочник.Организации КАК Организации
		|ГДЕ
		|	Организации.ИНН = &ИНН

	Выборка = ЗапросИННКПП.Выполнить().Выбрать();
	Если Выборка.Следующий() Тогда
		Возврат Выборка.Ссылка;				
	Иначе
		Возврат Неопределено;
	КонецЕсли;
	
КонецФункции

Функция ОрганизацияКонтрагентПоСтруктуре(ОрганизацияСтруктура, ЭтоОрганизация = Ложь) Экспорт

	КПП = ОрганизацияСтруктура.КПП;

// Параметры:
// СтруктураФЛ - структура с полями справочника
// СоздаватьНовое - булево, если Истина и запрос ничего не вернул, создаем новый элемент
// Возвращаемое значение - ссылка на справочник Физические лица или Неопределено
Функция ФизическоеЛицоПоСтруктуре(СтруктураФЛ, СоздаватьНовое = Ложь) Экспорт
	
	ФизическоеЛицо = Неопределено;
	
	Если СтруктураФЛ.Свойство("ИНН") Тогда
		ИНН = СтруктураФЛ.ИНН;
	Иначе
		ИНН = "";
	КонецЕсли;
	
	Если СтруктураФЛ.Свойство("СтраховойНомерПФР") Тогда
		СНИЛС = СтруктураФЛ.СтраховойНомерПФР;
	Иначе
		СНИЛС = "";
	КонецЕсли;
	
	Если ЗначениеЗаполнено(ИНН) Тогда
		ФизическоеЛицо = Справочники.ФизическиеЛица.НайтиПоРеквизиту("ИНН", ИНН);
	КонецЕсли;
	
	Если Не ЗначениеЗаполнено(ФизическоеЛицо) И ЗначениеЗаполнено(СНИЛС) Тогда
		ФизическоеЛицо = Справочники.ФизическиеЛица.НайтиПоРеквизиту("СтраховойНомерПФР", СНИЛС);
	КонецЕсли;
	
	Если Не ЗначениеЗаполнено(ФизическоеЛицо) Тогда
		
		Запрос = Новый Запрос;
		Запрос.Текст =
			"ВЫБРАТЬ
			|	ФизическиеЛица.Ссылка КАК ФизическоеЛицо,
			|	ФизическиеЛица.ИНН КАК ИНН,
			|	ФизическиеЛица.СтраховойНомерПФР КАК СтраховойНомерПФР
			|ИЗ
			|	Справочник.ФизическиеЛица КАК ФизическиеЛица
			|ГДЕ
			|	ФизическиеЛица.Фамилия = &Фамилия
			|	И ФизическиеЛица.Имя = &Имя
			|	И ФизическиеЛица.Отчество = &Отчество
			|	И ФизическиеЛица.ДатаРождения = &ДатаРождения
			|
			|УПОРЯДОЧИТЬ ПО
			|	ФизическиеЛица.ПометкаУдаления
			|АВТОУПОРЯДОЧИВАНИЕ";
		Запрос.УстановитьПараметр("Фамилия", 		СтруктураФЛ.Фамилия);
		Запрос.УстановитьПараметр("Имя", 			СтруктураФЛ.Имя);
		Запрос.УстановитьПараметр("Отчество", 		СтруктураФЛ.Отчество);
		Запрос.УстановитьПараметр("ДатаРождения", 	СтруктураФЛ.ДатаРождения);
		Выборка = Запрос.Выполнить().Выбрать();
		
		Если Выборка.Следующий() Тогда
			ФизическоеЛицо = Выборка.ФизическоеЛицо;
			
		Иначе                                                                      
			
			Если СоздаватьНовое Тогда
				ФизическоеЛицоОбъект = Справочники.ФизическиеЛица.СоздатьЭлемент();
				ЗаполнитьЗначенияСвойств(ФизическоеЛицоОбъект, СтруктураФЛ);
				ФизическоеЛицоОбъект.Записать();
				ФизическоеЛицо = ФизическоеЛицоОбъект.Ссылка; 
			КонецЕсли;                                                             
			
		КонецЕсли;
		
	КонецЕсли;
	
	Возврат ФизическоеЛицо;
	
КонецФункции

		|ГДЕ
		|	ШтриховыеКоды.ШтрихКод В(&ШтрихКод)";

			Номенклатура = Результат.Номенклатура;

Функция СтруктураЗаписиРегистраСведений(СтрокаСтруктурыРеквизитов) Экспорт
	
	СтруктураРеквизитов = ЗначениеИзСтрокиВнутр(СтрокаСтруктурыРеквизитов);
	Если Не СтруктураРеквизитов.Свойство("ИмяРегистра") Тогда
		Возврат Неопределено;
	Иначе		
		ИмяРегистра = СтруктураРеквизитов.ИмяРегистра;
	КонецЕсли;
	
	НаборЗаписей = РегистрыСведений[ИмяРегистра].СоздатьНаборЗаписей();
	Для Каждого ЭлементОтбор Из СтруктураРеквизитов Цикл
		Измерение = ЭлементОтбор.Ключ;
		Если Измерение = "ИмяРегистра" Тогда
			Продолжить;
		КонецЕсли;
		
		НаборЗаписей.Отбор[Измерение].Значение = ЭлементОтбор.Значение;
		НаборЗаписей.Отбор[Измерение].Использование = Истина;
	КонецЦикла;
	НаборЗаписей.Прочитать();
	
	Если ИмяРегистра = "ИсходящиеДокументыЭДО" ИЛИ ИмяРегистра = "СтатусыЭДОРасходнойНакладной" Тогда // исходящие документы
		Возврат ОбменДаннымиЭДО.СтруктураИсходящегоДокументаПоЗаписиРегистраСведений(ИмяРегистра, НаборЗаписей);
	КонецЕсли;

	ПараметрыВыгрузки = Неопределено;
	
	Если НаборЗаписей.Количество() Тогда
		Для Каждого ЗаписьРегистра Из НаборЗаписей Цикл
			
			Если ИмяРегистра = "СтатусыЭлектронныхТранспортныхНакладных" Тогда
				
				СтруктураЗаписи = Новый Структура;
				СтруктураЗаписи.Вставить("Период", 		ЗаписьРегистра.Период);
				СтруктураЗаписи.Вставить("Статус", 		Строка(ЗаписьРегистра.Статус));
				СтруктураЗаписи.Вставить("Комментарий", Строка(ЗаписьРегистра.Комментарий));
				
				ДанныеЭТРН = Новый Структура("НомерЭТРН, ДатаСозданияЭТРН");
				ЗаполнитьЗначенияСвойств(ДанныеЭТРН, ЗаписьРегистра.Документ);
				СтруктураЗаписи.Вставить("Документ", ДанныеЭТРН);
				
				ПараметрыВыгрузки = Новый Структура("ТипСообщения, Данные");
				ПараметрыВыгрузки.ТипСообщения = "StatusETRN";
				ПараметрыВыгрузки.Данные = СтруктураЗаписи;
				
			ИначеЕсли ИмяРегистра = "СоответствиеГоловногоИПодчиненногоGLN" Тогда
				
				СтруктураЗаписи = Новый Структура;
				СтруктураЗаписи.Вставить("Головной", ЗаписьРегистра.Головной.GLN);
				СтруктураЗаписи.Вставить("Подчиненный", ЗаписьРегистра.Подчиненный.GLN);
				СтруктураЗаписи.Вставить("ТолькоДляРЦ", ЗаписьРегистра.ТолькоДляРЦ);
				
				ПараметрыВыгрузки = Новый Структура("ТипСообщения, Данные");
				ПараметрыВыгрузки.ТипСообщения = "ConformityGLN";
				ПараметрыВыгрузки.Данные = СтруктураЗаписи;
				
			ИначеЕсли ИмяРегистра = "РегистрацияЗапросовЭДО" Тогда
				
				ОбъединенныйИдентификатор = ЗаписьРегистра.IdДокументаУПровайдера;
				ДлинаИдентификатораКонтур = 72;
				Если СтрДлина(ОбъединенныйИдентификатор) = ДлинаИдентификатораКонтур Тогда // контур
					ИдентификаторСообщения = Лев(ОбъединенныйИдентификатор, 36);
					ИдентификаторДокумента = Прав(ОбъединенныйИдентификатор, 36);
				Иначе
					ИдентификаторСообщения = "";
					ИдентификаторДокумента = ОбъединенныйИдентификатор;
				КонецЕсли;
				
				СтруктураЗаписи = Новый Структура;
				СтруктураЗаписи.Вставить("ИдентификаторДокумента", ИдентификаторДокумента);
				СтруктураЗаписи.Вставить("ИдентификаторСообщения", ИдентификаторСообщения);
				Если ЗаписьРегистра.РезультатСверкиСУПД = Перечисления.РезультатСверкиСУПД.ПринятоСРасхождениями Тогда
					ТекстРасхождения = ОбменДаннымиЭДО.ТекстТитулаПриемкиСРасхождением(ЗаписьРегистра.ДокументПриемки);
					СтруктураЗаписи.Вставить("ЕстьРасхождения", Истина);
					СтруктураЗаписи.Вставить("КомментарийПриемки", ТекстРасхождения);
				КонецЕсли;
				
				Если ЗначениеЗаполнено(ЗаписьРегистра.ДокументПриемки) Тогда
					СтруктураЗаписи.Вставить("НомерАкта", ЗаписьРегистра.ДокументПриемки.Номер);
					СтруктураЗаписи.Вставить("ДатаАкта", ЗаписьРегистра.ДокументПриемки.Дата);
				КонецЕсли;
				
				СтруктураЗаписи.Вставить("Период", 			ЗаписьРегистра.ДатаИзменения);
				СтруктураЗаписи.Вставить("СтатусДокумента",	ОбменДаннымиЭДО.СтатусЭлектронногоДокументаДляОбмена(ЗаписьРегистра.СтатусДокументаВЭДО));
				СтруктураЗаписи.Вставить("Комментарий", 	ЗаписьРегистра.Комментарий);
				СтруктураЗаписи.Вставить("Исходящий", 		Ложь);
				
				СтруктураЗаписи.Вставить("ТипДокументаЭДО", "УПД");
				
				ОтветственноеЛицо = ОбменДаннымиЭДО.ОтветственноеЛицо(ЗаписьРегистра.Организация, ЗаписьРегистра.Период, ЗаписьРегистра.СтруктурнаяЕдиница); 
				Если ЗначениеЗаполнено(ОтветственноеЛицо) Тогда
					СтруктураЗаписи.Вставить("ПодписантПокупателя", ОтветственноеЛицо);
				КонецЕсли;
				
				Если ЗначениеЗаполнено(ЗаписьРегистра.ДокументПриемки) Тогда
					Приемщик = ЗаписьРегистра.ДокументПриемки.Приемщик;
					Если ЗначениеЗаполнено(Приемщик) Тогда
						СтруктураДанныхПриемщика = ОбменДаннымиЭДО.СтруктураДанныхФизическогоЛица(Приемщик, ЗаписьРегистра.Период);
						СтруктураЗаписи.Вставить("Приемщик", СтруктураДанныхПриемщика);
					КонецЕсли;
				КонецЕсли;
				
				// {{ egaevsky - 10.07.2025 - #Перевод РН на УПД, ТЗ Малышевой Н.Б.
				Если ЗначениеЗаполнено(ЗаписьРегистра.КорректировкаРеализации) Тогда
					СтруктураЗаписи.Вставить("ТипДокументаЭДО", "УКД");
					СтруктураЗаписи.Удалить("Приемщик"); // в УКД нет данных приемщика
				КонецЕсли;
				Если ЗначениеЗаполнено(ЗаписьРегистра.ДокументПриемки) Тогда
					ЭтоРННаУПД = ОбменДаннымиЭДО.ЭтоРННаУПД(ЗаписьРегистра.ДокументПриемки.ДокОснование)
						ИЛИ ОбменДаннымиЭДО.ЭтоРННаУПД(ЗаписьРегистра.ДокументАстор);
					Если ЭтоРННаУПД Тогда
						СтруктураЗаписи.Вставить("Внутренний", Истина);
					КонецЕсли;		
				КонецЕсли;
				// }} egaevsky - 10.07.2025 - #Перевод РН на УПД, ТЗ Малышевой Н.Б.
				
				ПараметрыВыгрузки = Новый Структура("ТипСообщения, Данные");
				ПараметрыВыгрузки.ТипСообщения = "DataEDO";
				ПараметрыВыгрузки.Данные = СтруктураЗаписи;
				
			КонецЕсли;
			
		КонецЦикла;
		
	Иначе // удаление записей
		
		Если ИмяРегистра = "СоответствиеГоловногоИПодчиненногоGLN" Тогда
				
			СтруктураЗаписи = Новый Структура;
			СтруктураЗаписи.Вставить("Подчиненный", НаборЗаписей.Отбор.Подчиненный.Значение.GLN);
			СтруктураЗаписи.Вставить("ТолькоДляРЦ", НаборЗаписей.Отбор.ТолькоДляРЦ.Значение);
			СтруктураЗаписи.Вставить("УдалениеЗаписи", Истина);
			
			ПараметрыВыгрузки = Новый Структура("ТипСообщения, Данные");
			ПараметрыВыгрузки.ТипСообщения = "ConformityGLN";
			ПараметрыВыгрузки.Данные = СтруктураЗаписи;
			
		КонецЕсли;
		
	КонецЕсли;
	
	Возврат ПараметрыВыгрузки;

	КонецЕсли;
	
	Если ТипЗнч(ТочкаМаршрута) = Тип("СправочникСсылка.СтруктурныеЕдиницы") Тогда
		Владелец = ТочкаМаршрута;
	Иначе
		Владелец = ТочкаМаршрута.Владелец;
	КонецЕсли;
	
	Если Не ЗначениеЗаполнено(Владелец) 
		ИЛИ Владелец = Справочники.СтруктурныеЕдиницы.СобственноеПроизводство Тогда 
		
		Возврат Неопределено;
