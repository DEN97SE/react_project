	Если ОбщиеПроцедурыИФункцииПовтИсп.ЭтоБазаЦентральногоОфиса() Тогда
		Возврат;
	КонецЕсли;

	Если НЕ ЗначениеЗаполнено(СтруктурнаяЕдиница) Тогда
		Возврат;
	КонецЕсли;

	ЗарегистрироватьПолнуюВыгрузкуВОчередь(СтруктурнаяЕдиница);
	ОтправитьПолнуюВыгрузкуИзОчереди(СтруктурнаяЕдиница);

КонецПроцедуры

Процедура ЗарегистрироватьПолнуюВыгрузкуВОчередь(СтруктурнаяЕдиница) Экспорт

	Если НЕ ЗначениеЗаполнено(СтруктурнаяЕдиница) Тогда
		Возврат;
	КонецЕсли;

	СписокНоменклатуры = КХ_ДанныеPriceChekerOpera.ПолучитьНоменклатуруДляПолнойВыгрузкиPriceChecker(СтруктурнаяЕдиница);

	Если СписокНоменклатуры = Неопределено Или СписокНоменклатуры.Количество() = 0 Тогда
		Возврат;
	КонецЕсли;

	КоличествоВТранзакции = 0;
	НачатьТранзакцию();

	Попытка

		Для Каждого Номенклатура Из СписокНоменклатуры Цикл

			ЗарегистрироватьНоменклатуруВОчереди(Номенклатура, СтруктурнаяЕдиница);
			КоличествоВТранзакции = КоличествоВТранзакции + 1;

			Если КоличествоВТранзакции >= 5000 Тогда

				ЗафиксироватьТранзакцию();
				НачатьТранзакцию();
				КоличествоВТранзакции = 0;

			КонецЕсли;

		КонецЦикла;

		Если ТранзакцияАктивна() Тогда
			ЗафиксироватьТранзакцию();
		КонецЕсли;

	Исключение

		Если ТранзакцияАктивна() Тогда
			ОтменитьТранзакцию();
		КонецЕсли;

		ТекстОшибки = ПодробноеПредставлениеОшибки(ИнформацияОбОшибке());
		ВызватьИсключение ТекстОшибки;

	КонецПопытки;

КонецПроцедуры

Процедура ОтправитьПолнуюВыгрузкуИзОчереди(СтруктурнаяЕдиница) Экспорт

	КоличествоОжидающих = ПолучитьКоличествоОжидающихВОчереди(СтруктурнаяЕдиница);

	Если КоличествоОжидающих = 0 Тогда
		Возврат;
	КонецЕсли;

	РазмерПакета = 5000;
	КоличествоПакетов = Цел((КоличествоОжидающих + РазмерПакета - 1) / РазмерПакета);
	ИдентификаторСессии = Строка(Новый УникальныйИдентификатор);

	Для НомерПакета = 1 По КоличествоПакетов Цикл

		ТаблицаОчереди = ПолучитьПорциюОчередиPriceCheker(СтруктурнаяЕдиница, РазмерПакета);

		Если ТаблицаОчереди.Количество() = 0 Тогда
			Прервать;
		КонецЕсли;

		НоменклатураОтбор = ПолучитьМассивНоменклатуры(ТаблицаОчереди);

		Пакет = КХ_ДанныеPriceChekerOpera.СформироватьПакетПолнойВыгрузки(СтруктурнаяЕдиница, НоменклатураОтбор, ИдентификаторСессии, НомерПакета, КоличествоПакетов);

		Если Пакет = Неопределено Тогда
			УстановитьОшибкуНоменклатурамОчереди(СтруктурнаяЕдиница, НоменклатураОтбор, "Не удалось сформировать full-пакет");
			Продолжить;
		КонецЕсли;

		НоменклатураПакета = ПолучитьНоменклатуруПакета(Пакет, НоменклатураОтбор);
		НоменклатураСОшибкой = ПолучитьНоменклатуруНеПопавшуюВПакет(НоменклатураОтбор, НоменклатураПакета);

		Если НоменклатураСОшибкой.Количество() > 0 Тогда
			УстановитьОшибкуНоменклатурамОчереди(СтруктурнаяЕдиница, НоменклатураСОшибкой, "Не удалось сформировать карточку товара");
		КонецЕсли;

		JSON = ДанныеВJSONСтроку(Пакет);

		ОтправитьПакетPriceCheker(СтруктурнаяЕдиница, НоменклатураПакета, JSON);

	КонецЦикла;

	ВыполнитьОбработкуСервиса();
