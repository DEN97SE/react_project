Функция ОтправитьДанныеAPIВ2(Эндпоинт, Метод, ДанныеJSON = "", ОшибочныеИдентификаторы = Неопределено) Экспорт

	ДанныеАвторизации = ПолучитьДанныеАвторизацииAPIВ2();
	Если ДанныеАвторизации = Неопределено Тогда
		Возврат 0;
	КонецЕсли;

	Токен = ПолучитьТокенДоступаAPIВ2(ДанныеАвторизации);
	Если ПустаяСтрока(Токен) Тогда
		Возврат 0;
	КонецЕсли;

	Ответ = ВыполнитьHTTPЗапросAPIВ2(Эндпоинт, Метод, ДанныеJSON, Токен);

	Если Ответ <> Неопределено И (Ответ.КодСостояния = 401 Или Ответ.КодСостояния = 403) Тогда
		Токен = ПолучитьТокенДоступаAPIВ2(ДанныеАвторизации, Истина);
		Если Не ПустаяСтрока(Токен) Тогда
			Ответ = ВыполнитьHTTPЗапросAPIВ2(Эндпоинт, Метод, ДанныеJSON, Токен);
		КонецЕсли;
	КонецЕсли;

	Если Ответ = Неопределено Тогда
		Возврат 0;
	КонецЕсли;

	Если Ответ.КодСостояния <> 200 Тогда
		ОбработатьОшибкуОтветаAPIВ2(Ответ, Ответ.КодСостояния, ОшибочныеИдентификаторы);
	КонецЕсли;

	Возврат Ответ.КодСостояния;

Функция ПолучитьДанныеАвторизацииAPIВ2()

	ДанныеАвторизации = ОбщегоНазначения.ПрочитатьДанныеИзБезопасногоХранилища(
		"АдминистраторShelfSpaceCenter");

	Если ДанныеАвторизации = Неопределено Тогда
		ЗаписьЖурналаРегистрации("ShelfSpaceCenter.APIv2",
			УровеньЖурналаРегистрации.Ошибка, , ,
			"Не получены данные авторизации из безопасного хранилища");
	КонецЕсли;

	Возврат ДанныеАвторизации;

Функция ПолучитьТокенДоступаAPIВ2(ДанныеАвторизации, ОбновитьПринудительно = Ложь)

	Если Не ОбновитьПринудительно
		И ЗначениеЗаполнено(ТокенДоступаAPIВ2)
		И ЗначениеЗаполнено(МоментПолученияТокенаAPIВ2)
		И ТекущаяДатаСеанса() - МоментПолученияТокенаAPIВ2 < 20 * 60 * 60 Тогда
		Возврат ТокенДоступаAPIВ2;
	КонецЕсли;

	Логин = Строка(ДанныеАвторизации.Логин);
	Пароль = Строка(ДанныеАвторизации.Пароль);
	СтрокаАвторизации = Логин + ":" + Пароль;
	ДвоичныеДанныеАвторизации = ПолучитьДвоичныеДанныеИзСтроки(
		СтрокаАвторизации, КодировкаТекста.UTF8);

	HTTPЗапрос = Новый HTTPЗапрос("Center/ru/hs/auth/AccessToken");
	HTTPЗапрос.Заголовки.Вставить("Authorization", "Basic " + Base64Строка(ДвоичныеДанныеАвторизации));
	HTTPЗапрос.Заголовки.Вставить("Content-Type", "application/json");
	HTTPЗапрос.Заголовки.Вставить("Accept", "application/json");

	HTTPСоединение = Новый HTTPСоединение(
		Константы.БазовыйURL_ShelfSpaceCenter.Получить(), 8088, , , , 120, , );

	Попытка
		Ответ = HTTPСоединение.Получить(HTTPЗапрос);
	Исключение
		ЗаписьЖурналаРегистрации("ShelfSpaceCenter.APIv2",
			УровеньЖурналаРегистрации.Ошибка, , ,
			"Ошибка получения токена: " + ОписаниеОшибки());
		Возврат "";
	КонецПопытки;

	Если Ответ.КодСостояния <> 200 Тогда
		ОбработатьОшибкуОтветаAPIВ2(Ответ, Ответ.КодСостояния);
		Возврат "";
	КонецЕсли;

	Токен = Ответ.Заголовки.Получить("Access_token");
	Если Токен = Неопределено Тогда
		Токен = Ответ.Заголовки.Получить("Access-Token");
	КонецЕсли;

	Если ПустаяСтрока(Строка(Токен)) Тогда
		ЗаписатьВЖурналОшибокОбменаShelfSpace(, "AccessToken", "Авторизация",
			"Ответ 200 получен без заголовка Access_token");
		Возврат "";
	КонецЕсли;

	ТокенДоступаAPIВ2 = Строка(Токен);
	МоментПолученияТокенаAPIВ2 = ТекущаяДатаСеанса();

	Возврат ТокенДоступаAPIВ2;

Функция ВыполнитьHTTPЗапросAPIВ2(Эндпоинт, Метод, ДанныеJSON, Токен)

	URL = "Center/ru/hs/api/" + Эндпоинт;
	HTTPЗапрос = Новый HTTPЗапрос(URL);
	HTTPЗапрос.Заголовки.Вставить("Content-Type", "application/json");
	HTTPЗапрос.Заголовки.Вставить("Accept", "application/json");
	HTTPЗапрос.Заголовки.Вставить("Authorization", "Bearer " + Токен);

	Если ЗначениеЗаполнено(ДанныеJSON) Тогда
		HTTPЗапрос.УстановитьТелоИзСтроки(ДанныеJSON, КодировкаТекста.UTF8);
	КонецЕсли;

	HTTPСоединение = Новый HTTPСоединение(
		Константы.БазовыйURL_ShelfSpaceCenter.Получить(), 8088, , , , 120, , );

	Попытка
		Если Метод = "POST" Тогда
			Возврат HTTPСоединение.ОтправитьДляОбработки(HTTPЗапрос);
		ИначеЕсли Метод = "DELETE" Тогда
			Возврат HTTPСоединение.Удалить(HTTPЗапрос);
		ИначеЕсли Метод = "GET" Тогда
			Возврат HTTPСоединение.Получить(HTTPЗапрос);
		Иначе
			ЗаписатьВЖурналОшибокОбменаShelfSpace(, Эндпоинт, "HTTP",
				"Неподдерживаемый HTTP-метод: " + Метод);
			Возврат Неопределено;
		КонецЕсли;
	Исключение
		ЗаписьЖурналаРегистрации("ShelfSpaceCenter.APIv2",
			УровеньЖурналаРегистрации.Ошибка, , ,
			"Ошибка соединения с методом " + Эндпоинт + ": " + ОписаниеОшибки());
		Возврат Неопределено;
	КонецПопытки;

Процедура ОбработатьОшибкуОтветаAPIВ2(ОтветОтСервера, КодСостояния,
	ОшибочныеИдентификаторы = Неопределено)

	ТелоСтрокой = ОтветОтСервера.ПолучитьТелоКакСтроку();
	Если ПустаяСтрока(ТелоСтрокой) Тогда
		ЗаписатьВЖурналОшибокОбменаShelfSpace(, , ,
			"HTTP " + Строка(КодСостояния) + ". Тело ответа пустое");
		Возврат;
	КонецЕсли;

	Попытка
		Ответ = ИнтерфейсМОТПСлужебный.ТекстJSONВОбъект(ТелоСтрокой);
	Исключение
		ЗаписатьВЖурналОшибокОбменаShelfSpace(, , ,
			"HTTP " + Строка(КодСостояния) + ". " + ТелоСтрокой);
		Возврат;
	КонецПопытки;

	Сообщение = Неопределено;
	Если ТипЗнч(Ответ) = Тип("Структура") Тогда
		Ответ.Свойство("message", Сообщение);
	КонецЕсли;

	Если ТипЗнч(Сообщение) = Тип("Строка") Тогда
		Попытка
			СообщениеJSON = ИнтерфейсМОТПСлужебный.ТекстJSONВОбъект(Сообщение);
			Если ТипЗнч(СообщениеJSON) = Тип("Массив") Тогда
				Сообщение = СообщениеJSON;
			Иначе
				ЗаписатьВЖурналОшибокОбменаShelfSpace(, , , Сообщение);
				Возврат;
			КонецЕсли;
		Исключение
			ЗаписатьВЖурналОшибокОбменаShelfSpace(, , , Сообщение);
			Возврат;
		КонецПопытки;
	КонецЕсли;

	Если ТипЗнч(Сообщение) <> Тип("Массив") Тогда
		ЗаписатьВЖурналОшибокОбменаShelfSpace(, , ,
			"HTTP " + Строка(КодСостояния) + ". " + ТелоСтрокой);
		Возврат;
	КонецЕсли;

	Для Каждого Ошибка Из Сообщение Цикл
		Если ТипЗнч(Ошибка) <> Тип("Структура") Тогда
			Продолжить;
		КонецЕсли;

		Идентификатор = ?(Ошибка.Свойство("ID"), Строка(Ошибка.ID), "");
		Описание = ?(Ошибка.Свойство("Description"), Строка(Ошибка.Description), "");

		ЗаписатьВЖурналОшибокОбменаShelfSpace(
			Идентификатор,
			?(Ошибка.Свойство("Object"), Ошибка.Object, ""),
			?(Ошибка.Свойство("Type"), Ошибка.Type, ""),
			Описание);

		Если ОшибочныеИдентификаторы <> Неопределено
			И Не ПустаяСтрока(Идентификатор)
			И ОшибочныеИдентификаторы.Найти(НРег(Идентификатор)) = Неопределено Тогда
			ОшибочныеИдентификаторы.Добавить(НРег(Идентификатор));
		КонецЕсли;

		ДобавитьОтсутствующуюНоменклатуруИзОшибки(Описание, ОшибочныеИдентификаторы);
	КонецЦикла;

Процедура ДобавитьОтсутствующуюНоменклатуруИзОшибки(Описание,
	ОшибочныеИдентификаторы = Неопределено)

	Если СтрНайти(НРег(Описание), "'item'") = 0
		И СтрНайти(НРег(Описание), "номенклатур") = 0 Тогда
		Возврат;
	КонецЕсли;

	Идентификатор = ИзвлечьGUIDИзОшибкиAPIВ2(Описание);
	Если ПустаяСтрока(Идентификатор) Тогда
		Возврат;
	КонецЕсли;

	Если ОшибочныеИдентификаторы <> Неопределено
		И ОшибочныеИдентификаторы.Найти(Идентификатор) = Неопределено Тогда
		ОшибочныеИдентификаторы.Добавить(Идентификатор);
	КонецЕсли;

	Попытка
		СсылкаНоменклатуры = Справочники.Номенклатура.ПолучитьСсылку(
			Новый УникальныйИдентификатор(Идентификатор));
	Исключение
		Возврат;
	КонецПопытки;

	Если Не ЗначениеЗаполнено(СсылкаНоменклатуры)
		Или НоменклатураУжеВОчередиAPIВ2(СсылкаНоменклатуры) Тогда
		Возврат;
	КонецЕсли;

	ЗарегистрироватьИсточник(СсылкаНоменклатуры, ТекущаяДатаСеанса());

Функция ИзвлечьGUIDИзОшибкиAPIВ2(Описание)

	Маркер = "(ID = ";
	Позиция = СтрНайти(Описание, Маркер);
	Если Позиция > 0 Тогда
		Возврат НРег(СокрЛП(Сред(Описание, Позиция + СтрДлина(Маркер), 36)));
	КонецЕсли;

	Маркер = "Не найден ";
	Позиция = СтрНайти(Описание, Маркер);
	Если Позиция > 0 Тогда
		Возврат НРег(СокрЛП(Сред(Описание, Позиция + СтрДлина(Маркер), 36)));
	КонецЕсли;

	Возврат "";

Функция НоменклатураУжеВОчередиAPIВ2(Номенклатура)

	Запрос = Новый Запрос;
	Запрос.Текст =
	"ВЫБРАТЬ ПЕРВЫЕ 1
	|	Очередь.ИдентификаторСообщения КАК ИдентификаторСообщения
	|ИЗ
	|	РегистрСведений.ОчередьВыгрузкиДанныхShelfSpaceCenter КАК Очередь
	|ГДЕ
	|	Очередь.ВыгружаемыеДанные = &Номенклатура
	|	И Очередь.СтатусОтправки В
	|		(ЗНАЧЕНИЕ(Перечисление.пит_СтатусыСообщений.ОжиданиеОтправки),
	|		 ЗНАЧЕНИЕ(Перечисление.пит_СтатусыСообщений.Ошибка))";
	Запрос.УстановитьПараметр("Номенклатура", Номенклатура);

	Возврат Не Запрос.Выполнить().Пустой();

Процедура ОбработатьОтветAPIВ2(СсылкаНаОбъекты, ПолеСтруктуры, ИдентификаторОбъекта, Ответ, ДатаОтправки)

	СсылкаНаОбъект = СсылкаНаОбъекты.НайтиСтроки(
		Новый Структура(ПолеСтруктуры, ИдентификаторОбъекта));

	Если Ответ = 200 Тогда
		СтатусОтправкиОбъекта(СсылкаНаОбъект, ДатаОтправки);

		ОбъектОшибки = СсылкаНаОбъект[0];
		ЗаписатьВЖурналОшибокОбменаShelfSpace(
			Строка(ОбъектОшибки.ИдентификаторСообщения),
			Строка(ИдентификаторОбъекта), ,
			"КодОтвета = " + Строка(Ответ));
		СтатусОтправкиОбъекта(
			СсылкаНаОбъект, , Перечисления.пит_СтатусыСообщений.Ошибка);
	КонецЕсли;

	Если Прав(КаталогКартинок, 1) <> "\" Тогда

		Попытка

			ИмяФайла = Строка(Выборка.Код) + "_" + Строка(Выборка.БазоваяЕдиницаИзмерения) + ".BMP";
			ПутьККартинкам = КаталогКартинок + ИмяФайла;

			// Преобразование в BMP (если требуется) и запись на диск
			Если Картинка.Формат() <> ФорматКартинки.BMP Тогда
				Попытка
					Картинка.Преобразовать(ФорматКартинки.BMP);
				Исключение
					// Если преобразование не поддерживается платформой для этого типа файла, пишем как есть
				КонецПопытки;
			КонецЕсли;

			Картинка.Записать(ПутьККартинкам);

			УспешноОчищенФон = УдалитьФонКартинки(ПутьККартинкам);

				//ЗаписьЖурналаРегистрации("ShelfSpace.ОбработкаКартинок", УровеньЖурналаРегистрации.Предупреждение,,,
				//	"Не удалось очистить фон для файла: " + ИмяФайла);
			КонецЕсли;

			СтруктураКартинки.Вставить("UOM", Строка(Выборка.ГУИД_ЕдиницыИзмерения));
			СтруктураКартинки.Вставить("FileName", ИмяФайла);
			МассивКартинок.Добавить(СтруктураКартинки);

			JSON = Новый Структура;

		Исключение
			// Логируем любые критические ошибки (отсутствие прав на диск, падение сети и т.д.)
			ЗаписьЖурналаРегистрации("ShelfSpace.ОшибкаВыгрузкиКартинок", УровеньЖурналаРегистрации.Ошибка,,,
			"Ошибка обработки товара с кодом " + Выборка.Код + ": " + ОписаниеОшибки());

// Вспомогательный метод, чтобы не дублировать код формирования JSON
Процедура ОтправитьПорциюДанных(ДанныеПорции)

	КоличествоЭлементов = ДанныеПорции.Количество();

	Если КоличествоЭлементов = 0 Тогда
		Возврат;
	КонецЕсли;

	JSON = Новый Структура;
	JSON.Вставить("DateUploaded", ФорматДатыAPI());
	JSON.Вставить("ItemStock", ДанныеПорции);

	СтрокаJSON = ИнтерфейсМОТП.ОбъектВСтрокуJSON(JSON);
	Ответ = ОтправитьДанныеAPIВ2("ItemStock", "POST", СтрокаJSON);

	Если Ответ = 200 Тогда
		Возврат;
	КонецЕсли;


	Если Ответ = 0 Тогда

		Если КоличествоЭлементов > 1 Тогда

			// Большой пакет мог не пройти из-за таймаута или разрыва соединения.
			// Пробуем повторить меньшими частями.

		Иначе

			ПроблемныйОстаток = ДанныеПорции[0];

			ЗаписатьВЖурналОшибокОбменаShelfSpace("",
			СтрШаблон("Item: %1; SalesFloor: %2",
			ПроблемныйОстаток.Item,
			ПроблемныйОстаток.SalesFloor),
			"ItemStock",
			СтрШаблон("Запрос завершился без HTTP-ответа. Quantity: %1",
			ПроблемныйОстаток.Quantity));

			Возврат;

		КонецЕсли;

	ИначеЕсли Ответ <> 500 Тогда
		Возврат;
	КонецЕсли;

	Если КоличествоЭлементов = 1 Тогда

		ПроблемныйОстаток = ДанныеПорции[0];

		ЗаписатьВЖурналОшибокОбменаShelfSpace("",
		СтрШаблон("Item: %1; SalesFloor: %2",
		ПроблемныйОстаток.Item,
		ПроблемныйОстаток.SalesFloor),
		"ItemStock",
		СтрШаблон("Не удалось выгрузить остаток. Quantity: %1",
		ПроблемныйОстаток.Quantity));

		Возврат;

	КонецЕсли;

	ГраницаРазделения = Цел(КоличествоЭлементов / 2);

	ПерваяЧасть = Новый Массив;
	ВтораяЧасть = Новый Массив;

	Для Индекс = 0 По КоличествоЭлементов - 1 Цикл

		Если Индекс < ГраницаРазделения Тогда
			ПерваяЧасть.Добавить(ДанныеПорции[Индекс]);
		Иначе
			ВтораяЧасть.Добавить(ДанныеПорции[Индекс]);
		КонецЕсли;

	КонецЦикла;

	ОтправитьПорциюДанных(ПерваяЧасть);
	ОтправитьПорциюДанных(ВтораяЧасть);

				УровеньЖурналаРегистрации.Ошибка, ТекстОшибки);
				Продолжить;
			КонецПопытки;

		КонецЦикла;

		УстановитьПривилегированныйРежим(Ложь);

	КонецЕсли;

КонецПроцедуры

Функция ПолучитьСтруктуруДляЗаписиВРегистр(Источник, ИдентификаторСообщения = Неопределено, ДатаРегистрации = Неопределено,
	ДатаОтправки = Неопределено, СтатусОтправки = Неопределено)

	Если ИдентификаторСообщения = Неопределено Тогда
		ИдентификаторСообщения = Строка(Новый УникальныйИдентификатор());
	КонецЕсли;

	Если ДатаРегистрации = Неопределено Тогда
		ДатаРегистрации = ТекущаяДатаСеанса();
	КонецЕсли;

	Если ДатаОтправки = Неопределено Тогда
		ДатаОтправки = Дата(1, 1, 1);
	КонецЕсли;

	Если СтатусОтправки = Неопределено Тогда
		СтатусОтправки = Перечисления.пит_СтатусыСообщений.ОжиданиеОтправки;
	КонецЕсли;

	ИдентификаторОбъекта = Неопределено;
	ВыгружаемыеДанные = Неопределено;
	КлючОтбора = "";

	ОбъектМетаданных = Метаданные.НайтиПоТипу(ТипЗнч(Источник));

	Если пит_ИнтеграцияRMQСервер.ЭтоРегистр(ОбъектМетаданных) Тогда

		ВыгружаемыеДанные = Строка(ТипЗнч(Источник));

		Если ТипЗнч(Источник) = Тип("РегистрСведенийНаборЗаписей.ЗначенияСвойствОбъектов") Или
			ТипЗнч(Источник) = Тип("РегистрСведенийНаборЗаписей.ЗначенияСвойствТоваров") Тогда

			Если ТипЗнч(Источник.Отбор.Объект.Значение) = Тип("СправочникСсылка.Номенклатура")
				И Источник.Отбор.Объект.Значение.ЭтоГруппа = Ложь Тогда

				ИдентификаторОбъекта = Источник.Отбор.Объект.Значение;

			КонецЕсли;

		ИначеЕсли ТипЗнч(Источник) = Тип("РегистрСведенийНаборЗаписей.АссортиментныеМикроматрицы") Тогда

				ИдентификаторОбъекта = Источник.Отбор.Номенклатура.Значение;
				КлючОтбора = Источник.Отбор.Микроматрица.Значение;

		ИначеЕсли ТипЗнч(Источник) = Тип("РегистрСведенийНаборЗаписей.ШтриховыеКоды") Тогда

			Если Источник.Отбор.Номенклатура.Значение.ЭтоГруппа = Ложь Тогда
				ИдентификаторОбъекта = Источник.Отбор.Номенклатура.Значение;
			КонецЕсли;

		ИначеЕсли ТипЗнч(Источник) = Тип("РегистрСведенийНаборЗаписей.ЦеныНоменклатуры") Тогда

			ИдентификаторОбъекта = Источник.Отбор.Регистратор.Значение;

		КонецЕсли;

	Иначе

		Если ТипЗнч(Источник) = Тип("СправочникСсылка.ЕдиницыИзмерения") Тогда

			Если Источник.Владелец.ЭтоГруппа = Ложь Тогда
				ИдентификаторОбъекта = Источник.Владелец;
				ВыгружаемыеДанные = ТипЗнч(Источник);
			КонецЕсли;

		ИначеЕсли ТипЗнч(Источник) = Тип("СправочникСсылка.ХранилищеИнформации") Тогда

			Если ТипЗнч(Источник.Объект) = Тип("СправочникСсылка.Номенклатура")
				И Источник.ВидДанных = Перечисления.ВидыДополнительнойИнформацииОбъектов.Изображение Тогда

				ИдентификаторОбъекта = Источник.Объект;
				ВыгружаемыеДанные = ТипЗнч(Источник);

			КонецЕсли;

		Иначе
			ИдентификаторОбъекта = Источник.УникальныйИдентификатор();
			ВыгружаемыеДанные = Источник;
		КонецЕсли;

	КонецЕсли;

	Если ИдентификаторОбъекта = Неопределено Тогда
		Возврат ИдентификаторОбъекта;
	КонецЕсли;

	МассивОчередиОбмена = Новый Массив;

	Если ТипЗнч(ИдентификаторОбъекта) = Тип("ДокументСсылка.Переоценка") Тогда

		КлючОтбора = ИдентификаторОбъекта.СтруктурнаяЕдиницаНазначения;

		Для Каждого ТекСтрокаСостава Из ИдентификаторОбъекта.Состав Цикл
			ИдентификаторСообщения = Строка(Новый УникальныйИдентификатор());
			МассивОчередиОбмена.Добавить(
			СоздатьБазовуюСтруктуруОчереди(
			ИдентификаторСообщения,
			ТекСтрокаСостава.Номенклатура,
			ВыгружаемыеДанные,
			ДатаРегистрации,
			ДатаОтправки,
			СтатусОтправки,
			КлючОтбора));
		КонецЦикла;

	Иначе

		МассивОчередиОбмена.Добавить(
		СоздатьБазовуюСтруктуруОчереди(
		ИдентификаторСообщения,
		ИдентификаторОбъекта,
		ВыгружаемыеДанные,
		ДатаРегистрации,
		ДатаОтправки,
		СтатусОтправки,
		КлючОтбора));

	КонецЕсли;

	Возврат МассивОчередиОбмена;

Процедура СтатусОтправкиОбъекта(СсылкаНаОбъект, ДатаОтправки, СтатусОтправки = Неопределено)

	Если СсылкаНаОбъект.Количество() = 0 Тогда
		Возврат;
	КонецЕсли;

	Для Каждого СтрокаДанных Из СсылкаНаОбъект Цикл

		ИдентификаторСообщения = СтрокаДанных.ИдентификаторСообщения;
		ИдентификаторОбъекта = СтрокаДанных.ИдентификаторОбъекта;
		ВыгружаемыеДанные = СтрокаДанных.ВыгружаемыеДанные;
		ДатаРегистрации = СтрокаДанных.ДатаРегистрации;

		НаборЗаписей = РегистрыСведений.ОчередьВыгрузкиДанныхShelfSpaceCenter.СоздатьНаборЗаписей();
		НаборЗаписей.Отбор.ИдентификаторСообщения.Установить(ИдентификаторСообщения);
		НаборЗаписей.Прочитать();

		Если СтатусОтправки = Неопределено Тогда

			Если НаборЗаписей.Количество() > 0 Тогда

				УстановитьПривилегированныйРежим(Истина);
				НаборЗаписей.Очистить();

				Попытка
					НаборЗаписей.Записать();
				Исключение
					ТекстОшибки = ПодробноеПредставлениеОшибки(ИнформацияОбОшибке());
					ЗаписьЖурналаРегистрации("СтатусОтправки_ОчередьВыгрузкиДанныхShelfSpaceCenter", УровеньЖурналаРегистрации.Ошибка, , , ТекстОшибки);
					ВызватьИсключение;
				КонецПопытки;

				УстановитьПривилегированныйРежим(Ложь);

			КонецЕсли;

			Продолжить;

		КонецЕсли;

		Если НаборЗаписей.Количество() > 0 Тогда

			ЗаписьДляОбновления = НаборЗаписей[0];

			Если ЗаписьДляОбновления.СтатусОтправки = СтатусОтправки Тогда
				Продолжить;
			КонецЕсли;

			ЗаписьДляОбновления.СтатусОтправки = СтатусОтправки;
			ЗаписьДляОбновления.ДатаОтправки = ДатаОтправки;

			УстановитьПривилегированныйРежим(Истина);

			Попытка
				НаборЗаписей.Записать(Истина);
			Исключение
				ТекстОшибки = ПодробноеПредставлениеОшибки(ИнформацияОбОшибке());
				ЗаписьЖурналаРегистрации("СтатусОтправки_ОчередьВыгрузкиДанныхShelfSpaceCenter", УровеньЖурналаРегистрации.Ошибка, , , ТекстОшибки);
				ВызватьИсключение;
			КонецПопытки;

			УстановитьПривилегированныйРежим(Ложь);

		Иначе

			НоваяЗапись = НаборЗаписей.Добавить();
			НоваяЗапись.ИдентификаторСообщения = ИдентификаторСообщения;
			НоваяЗапись.ИдентификаторОбъекта = ИдентификаторОбъекта;
			НоваяЗапись.ВыгружаемыеДанные = ВыгружаемыеДанные;
			НоваяЗапись.ДатаРегистрации = ДатаРегистрации;
			НоваяЗапись.ДатаОтправки = ДатаОтправки;
			НоваяЗапись.СтатусОтправки = СтатусОтправки;
			НоваяЗапись.ВидОшибки = Перечисления.пит_ВидыОшибок.ПустаяСсылка();

			УстановитьПривилегированныйРежим(Истина);

			Попытка
				НаборЗаписей.Записать();
			Исключение
				ТекстОшибки = ПодробноеПредставлениеОшибки(ИнформацияОбОшибке());
				ЗаписьЖурналаРегистрации("СтатусОтправки_ОчередьВыгрузкиДанныхShelfSpaceCenter", УровеньЖурналаРегистрации.Ошибка, , , ТекстОшибки);
				ВызватьИсключение;
			КонецПопытки;

			УстановитьПривилегированныйРежим(Ложь);

		КонецЕсли;

	КонецЦикла;

// Формирует записи в очереди выгрузки для номенклатуры, которая ранее выгружалась с ошибкой.
// Необходимость возникает при создании номенклатуры без сохранения (подписка на запись не срабатывает).
Процедура ЗаписатьВОчередьНеПрогруженнуюНоменклатуру()

	Запрос = Новый Запрос;
	Запрос.Текст =
	"ВЫБРАТЬ
	|	ОчередьВыгрузкиДанныхShelfSpaceCenter.ИдентификаторОбъекта КАК ИдентификаторОбъекта,
	|	ОчередьВыгрузкиДанныхShelfSpaceCenter.ИдентификаторСообщения КАК ИдентификаторСообщения
	|ИЗ
	|	РегистрСведений.ОчередьВыгрузкиДанныхShelfSpaceCenter КАК ОчередьВыгрузкиДанныхShelfSpaceCenter
	|ГДЕ
	|	ОчередьВыгрузкиДанныхShelfSpaceCenter.СтатусОтправки = ЗНАЧЕНИЕ(Перечисление.пит_СтатусыСообщений.Ошибка)
	|
	|СГРУППИРОВАТЬ ПО
	|	ОчередьВыгрузкиДанныхShelfSpaceCenter.ИдентификаторОбъекта,
	|	ОчередьВыгрузкиДанныхShelfSpaceCenter.ИдентификаторСообщения";

	РезультатЗапроса = Запрос.Выполнить();

	Выборка = РезультатЗапроса.Выбрать();

	Счетчик = 0;
	НачатьТранзакцию();
	Пакет = 100;

	Пока Выборка.Следующий() Цикл

		Если ТипЗнч(Выборка.ИдентификаторОбъекта) <> Тип("СправочникСсылка.Номенклатура")
			Или Выборка.ИдентификаторОбъекта.ЭтоГруппа Тогда

			Продолжить;

		КонецЕсли;

		Запись = РегистрыСведений.ОчередьВыгрузкиДанныхShelfSpaceCenter.СоздатьМенеджерЗаписи();
		Запись.Приоритет = 1;
		Запись.ВыгружаемыеДанные = Выборка.ИдентификаторОбъекта;
		Запись.ИдентификаторОбъекта = Выборка.ИдентификаторОбъекта.УникальныйИдентификатор();
		Запись.ДатаРегистрации = ТекущаяДатаСеанса();
		Запись.СтатусОтправки = Перечисления.пит_СтатусыСообщений.ОжиданиеОтправки;
		Запись.ИдентификаторСообщения = Новый УникальныйИдентификатор();
		Запись.Записать();

		УдалитьОшибочное = РегистрыСведений.ОчередьВыгрузкиДанныхShelfSpaceCenter.СоздатьМенеджерЗаписи();
		УдалитьОшибочное.ИдентификаторСообщения = Выборка.ИдентификаторСообщения;
		УдалитьОшибочное.Прочитать();
		УдалитьОшибочное.Удалить();

		Счетчик = Счетчик + 1;

		Если Счетчик % Пакет = 0 Тогда

			ЗафиксироватьТранзакцию();
			НачатьТранзакцию();

		КонецЕсли;

	КонецЦикла;

	Если ТранзакцияАктивна() Тогда
		ЗафиксироватьТранзакцию();
	КонецЕсли;
