Функция ПолучитьНаборРодителей() Экспорт 
	
	// Шаг 1: Получаем все группы и их цепочки
	НовТЗСоВсемиРодителями = Новый ТаблицаЗначений;
	ТипНоменклатура = Новый ОписаниеТипов("СправочникСсылка.Номенклатура");
	
	НовТЗСоВсемиРодителями.Колонки.Добавить("group", ТипНоменклатура);
	НовТЗСоВсемиРодителями.Колонки.Добавить("subcategory", ТипНоменклатура);
	НовТЗСоВсемиРодителями.Колонки.Добавить("code", Новый ОписаниеТипов("Число"));
	НовТЗСоВсемиРодителями.Колонки.Добавить("guid", Новый ОписаниеТипов("Строка"));
	НовТЗСоВсемиРодителями.Колонки.Добавить("name", Новый ОписаниеТипов("Строка"));
	НовТЗСоВсемиРодителями.Колонки.Добавить("parent_guid", Новый ОписаниеТипов("Строка"));
	НовТЗСоВсемиРодителями.Колонки.Добавить("Наименование", Новый ОписаниеТипов("Строка"));

	// Получаем все группы
	Запрос = Новый Запрос;
	Запрос.Текст =
	"ВЫБРАТЬ РАЗЛИЧНЫЕ
	|	Номенклатура.Ссылка КАК Ссылка,
	|	Номенклатура.Наименование КАК Наименование
	|ИЗ
	|	РегистрСведений.РаспределениеКМПоКатегориям.СрезПоследних(&Период, Формат В (&Направления)) КАК Регистр
	|		ЛЕВОЕ СОЕДИНЕНИЕ Справочник.Номенклатура КАК Номенклатура
	|		ПО Регистр.Категория = Номенклатура.Ссылка
	|ГДЕ
	|	Номенклатура.ПометкаУдаления = ЛОЖЬ
	|	И Номенклатура.ЭтоГруппа = ИСТИНА
	// Новое условие отбора по иерархии
	|	И Номенклатура.Ссылка В ИЕРАРХИИ(&ГруппаНоменклатуры)
	|
	|СГРУППИРОВАТЬ ПО
	|	Номенклатура.Ссылка,
	|	Номенклатура.Наименование";
	
	Запрос.УстановитьПараметр("Период", ТекущаяДатаСеанса());
	Запрос.УстановитьПараметр("Направления", ПолучитьСписокНаправлений());
	Запрос.УстановитьПараметр("ГруппаНоменклатуры", ПолучитьГруппыНоменклатуры());
	
	Результат = Запрос.Выполнить().Выгрузить();

	Для Каждого ТекСтр Из Результат Цикл
		
		Группа = ТекСтр.Ссылка;
		НаименованиеГруппы = ТекСтр.Наименование;
		
		Цепочка = ПолучитьЦепочкуРодителей(Группа); // [Группа, Родитель, ..., Корень]
		ДлинаЦепочки = Цепочка.Количество();
		
		Для Индекс = 0 По ДлинаЦепочки - 1 Цикл
			
			ТекущийРодитель = Цепочка[Индекс];
			
			// Получаем данные
			guidРодителя = ТекущийРодитель.УникальныйИдентификатор();
			ОбъектРодителя = ТекущийРодитель.ПолучитьОбъект();
			НаименованиеРодителя = ?(ОбъектРодителя = Неопределено, "", ОбъектРодителя.Наименование);
			
			Стр = НовТЗСоВсемиРодителями.Добавить();
			Стр.group = Группа;
			Стр.subcategory = ТекущийРодитель;
			Стр.code = ДлинаЦепочки - 1 - Индекс; // 0 = корень
			Если ЗначениеЗаполнено(ТекущийРодитель.Родитель) Тогда 
				Стр.parent_guid = ТекущийРодитель.Родитель.СтроковыйИдентификатор; 
			Иначе
				Стр.parent_guid = "null";
			КонецЕсли;
			Стр.guid = guidРодителя;
			Стр.Наименование = НаименованиеРодителя;
			
		КонецЦикла;
		
	КонецЦикла;

	// Шаг 2: Строим дерево
	Иерархия = ПостроитьИерархиюИзТаблицы(НовТЗСоВсемиРодителями);
	
	Возврат Иерархия;
	
КонецФункции

Функция ПолучитьЦепочкуРодителей(Группа)
	
	Цепочка = Новый Массив();
	Текущий = Группа.Ссылка;
	
	Пока Не Текущий.Пустая() Цикл
		Цепочка.Вставить(0, Текущий); // Всегда в начало → получаем [Корень, ..., Группа]
		Текущий = Текущий.Родитель;
	КонецЦикла;
		
	Возврат Цепочка;
	
КонецФункции

Функция ПостроитьИерархиюИзТаблицы(ТаблицаРодителей) Экспорт
	
	// 1. Создаём карту: GUID → узел
	Карта = Новый Соответствие();
	
	Для Каждого Стр Из ТаблицаРодителей Цикл
		
		Если Карта.Получить(Стр.guid) <> Неопределено Тогда
			Продолжить;
		КонецЕсли;
		
		Узел = Новый Структура;
		Узел.Вставить("guid", Стр.guid);
		Узел.Вставить("name", Стр.Наименование);
		Узел.Вставить("parent_guid", Стр.parent_guid);
		Узел.Вставить("level", Стр.code);
		Узел.Вставить("children", Новый Массив());
		
		Карта.Вставить(Стр.guid, Узел);
		
	КонецЦикла;
	
	// 2. Строим связи: для каждой строки определяем родителя
	Для Каждого Стр Из ТаблицаРодителей Цикл
		
		ТекGUID = Стр.guid;
		ТекLevel = Стр.code;
		
		Если ТекLevel = 0 Тогда
			// Это корень — ничего не делаем
			Продолжить;
		КонецЕсли;
		
	КонецЦикла;
	
	УникальныеГруппы = Новый Соответствие();
	Для Каждого Стр Из ТаблицаРодителей Цикл
		УникальныеГруппы.Вставить(Стр.group, Истина);
	КонецЦикла;
	
	Корни = Новый Массив();
	КартаGUID = Новый Соответствие(); // guid → узел
	
	// Для каждой группы строим путь и вставляем в дерево
	Для Каждого Ключ Из УникальныеГруппы Цикл
		
		// Получаем цепочку для этой группы
		Цепочка = ПолучитьЦепочкуРодителей(Ключ.Ключ);
		//Цепочка.Развернуть(); // Теперь: [Корень, ..., Группа]
		
		ТекущийРодитель = Неопределено;
		
		Для Индекс = 0 По Цепочка.Количество() - 1 Цикл
			
			Элемент = Цепочка[Индекс];
			guid = Элемент.УникальныйИдентификатор();
			Объект = Элемент.Ссылка; 
			parent_guid = Элемент.Родитель.УникальныйИдентификатор();
			Наименование = ?(Объект = Неопределено, "", Объект.Наименование);
			
			Если КартаGUID.Получить(guid) <> Неопределено Тогда
				Узел = КартаGUID[guid];
			Иначе
				Узел = Новый Структура;
				Узел.Вставить("guid", Строка(guid));
				Узел.Вставить("name", Строка(Наименование));
				Узел.Вставить("parent_guid", Строка(parent_guid));
				Узел.Вставить("level", Строка(Индекс));
				Узел.Вставить("children", Новый Массив());
				КартаGUID.Вставить(guid, Узел);
				
				Если ТекущийРодитель = Неопределено Тогда
					Корни.Добавить(Узел);
				Иначе
					ТекущийРодитель.children.Добавить(Узел);
				КонецЕсли;
			КонецЕсли;
			
			ТекущийРодитель = Узел;
			
		КонецЦикла;
		
	КонецЦикла;
	
	Результат = Корни;
	
	Возврат Результат;
	
КонецФункции
