// АСТОР. Модуль управляемой формы внешней обработки КХ_КП_Проверка.epf.
// Реквизиты формы:
//   Ревизия: Число 15,0, начальное значение 1;
//   ТекстПакета: Строка неограниченной длины;
//   РезультатПроверок: Строка неограниченной длины.
// Дополнительные реквизиты: Включен (Булево), СервисИнтеграции, КаналИсходящий,
// УзелПолучатель, КодОтправителя (Строка 150), МаксимальныйРазмерСообщения (Число 15,0).
// Команды: ПроверитьФормат, СформироватьПакет, ЗаписатьНастройки,
// ЗарегистрироватьИзменения, ОтправитьТолькоКабинет.
// Последние три команды изменяют настройки/очередь и выполняют реальную отправку.

&НаСервере
Процедура ПриСозданииНаСервере(Отказ, СтандартнаяОбработка)
    Настройки = КХ_КП_ОбменАстор.ПолучитьНастройки();
    Включен = Настройки.Включен;
    СервисИнтеграции = Настройки.СервисИнтеграции;
    КаналИсходящий = Настройки.КаналИсходящий;
    УзелПолучатель = Настройки.УзелПолучатель;
    КодОтправителя = Настройки.КодОтправителя;
    МаксимальныйРазмерСообщения = Настройки.МаксимальныйРазмерСообщения;
    Ревизия = 1;
КонецПроцедуры

&НаКлиенте
Процедура ПроверитьФормат(Команда)
    РезультатПроверок = ПроверитьФорматНаСервере();
КонецПроцедуры

&НаСервереБезКонтекста
Функция ПроверитьФорматНаСервере()
    Возврат КХ_КП_ТестыОбмена.ВыполнитьПроверки();
КонецФункции

&НаКлиенте
Процедура СформироватьПакет(Команда)
    Пакет = СформироватьПакетНаСервере(Ревизия);
    ТекстПакета = Пакет.ТелоJSON;
    РезультатПроверок = "Пакет сформирован и проверен в памяти. "
        + "В ОчередьШины не записан. Идентификатор: " + Пакет.ИдентификаторСообщения;
КонецПроцедуры

&НаСервереБезКонтекста
Функция СформироватьПакетНаСервере(НомерРевизии)
    Возврат КХ_КП_ОбменАстор.СформироватьПакетДляШины(НомерРевизии);
КонецФункции

&НаКлиенте
Процедура ЗаписатьНастройки(Команда)
    Настройки = Новый Структура;
    Настройки.Вставить("Включен", Включен);
    Настройки.Вставить("СервисИнтеграции", СервисИнтеграции);
    Настройки.Вставить("КаналИсходящий", КаналИсходящий);
    Настройки.Вставить("УзелПолучатель", УзелПолучатель);
    Настройки.Вставить("КодОтправителя", КодОтправителя);
    Настройки.Вставить("МаксимальныйРазмерСообщения", МаксимальныйРазмерСообщения);
    СохранитьНастройкиНаСервере(Настройки);
    РезультатПроверок = "Настройки сохранены.";
КонецПроцедуры

&НаСервереБезКонтекста
Процедура СохранитьНастройкиНаСервере(Настройки)
    КХ_КП_ОбменАстор.СохранитьНастройки(Настройки);
КонецПроцедуры

&НаКлиенте
Процедура ЗарегистрироватьИзменения(Команда)
    РезультатПроверок = ЗарегистрироватьНаСервере();
КонецПроцедуры

&НаСервереБезКонтекста
Функция ЗарегистрироватьНаСервере()
    Результат = КХ_КП_ОбменАстор.ПодготовитьИзменения();
    Возврат КХ_КП_СлужебныйОбмен.JSON(Результат);
КонецФункции

&НаКлиенте
Процедура ОтправитьТолькоКабинет(Команда)
    ОтправитьНаСервере();
    РезультатПроверок = "Обработка очереди KP_SUPPLIER завершена. Проверьте статусы записей ОчередьШины.";
КонецПроцедуры

&НаСервереБезКонтекста
Процедура ОтправитьНаСервере()
    Интеграция1СШина.ОтправитьСообщенияКабинета();
КонецПроцедуры
