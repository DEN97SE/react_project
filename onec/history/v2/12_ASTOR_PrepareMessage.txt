Процедура ПроверитьНастройки(Настройки, ТребоватьВключения = Истина) Экспорт
    Включен = КХ_КП_СлужебныйОбмен.Поле(Настройки, "Включен");
    Если ТипЗнч(Включен) <> Тип("Булево") Тогда
        ВызватьИсключение "Включен должен иметь тип Булево.";
    КонецЕсли;
    Если ТребоватьВключения И Не Включен Тогда
        ВызватьИсключение "Обмен кабинета выключен в настройках.";
    КонецЕсли;
    Для Каждого Имя Из СтрРазделить("СервисИнтеграции,КаналИсходящий,УзелПолучатель,КодОтправителя", ",") Цикл
        ЗначениеПоля = КХ_КП_СлужебныйОбмен.Поле(Настройки, Имя);
        Если ТипЗнч(ЗначениеПоля) <> Тип("Строка") Тогда
            ВызватьИсключение "Настройка должна быть строкой: " + Имя;
        КонецЕсли;
        Если ПустаяСтрока(ЗначениеПоля) Тогда
            ВызватьИсключение "Не заполнена настройка: " + Имя;
        КонецЕсли;
        Если ЗначениеПоля <> СокрЛП(ЗначениеПоля) Тогда
            ВызватьИсключение "Удалите пробелы по краям настройки: " + Имя;
        КонецЕсли;
    КонецЦикла;
    Максимум = КХ_КП_СлужебныйОбмен.Поле(Настройки, "МаксимальныйРазмерСообщения");
    Если ТипЗнч(Максимум) <> Тип("Число") Тогда
        ВызватьИсключение "МаксимальныйРазмерСообщения должен быть числом байт.";
    КонецЕсли;
    Если Максимум <= 0 Или Максимум <> Цел(Максимум) Тогда
        ВызватьИсключение "Размер должен быть положительным целым числом байт.";
    КонецЕсли;
КонецПроцедуры

Функция СформироватьПакетДляШины(Ревизия) Экспорт
    // Предпросмотр: очередь и состояние не меняются.
    Если ТипЗнч(Ревизия) <> Тип("Число") Тогда
        ВызватьИсключение "Ревизия должна быть числом.";
    КонецЕсли;
    Если Ревизия <= 0 Или Ревизия <> Цел(Ревизия) Тогда
        ВызватьИсключение "Ревизия должна быть положительным целым числом.";
    КонецЕсли;
    Данные = КХ_КП_ДанныеАстора.ПолучитьДанныеДляОбмена();
    Пакет = КХ_КП_СлужебныйОбмен.СоздатьПакет(Данные, Ревизия, "astor-master", "astor.reference.snapshot");
    КХ_КП_СлужебныйОбмен.ПроверитьСнимок(Пакет, "astor.reference.snapshot", "astor-master");
    Возврат Новый Структура("ИдентификаторСообщения,Ревизия,Хеш,ТелоJSON",
        Пакет.message_id, Пакет.revision, Пакет.content_hash, КХ_КП_СлужебныйОбмен.JSON(Пакет));
КонецФункции

Функция ПодготовитьИзменения(Принудительно = Ложь) Экспорт
    Настройки = ПолучитьНастройки();
    ПроверитьНастройки(Настройки);
    ПрежняяРевизия = ПолучитьСостояниеОбмена().Ревизия;
    Данные = КХ_КП_ДанныеАстора.ПолучитьДанныеДляОбмена();
    JSONДанных = КХ_КП_СлужебныйОбмен.JSON(Данные);
    ХешДанных = КХ_КП_СлужебныйОбмен.Хеш(JSONДанных);
    Маршрут = Новый Структура("service,channel,receiver,sender",
        Настройки.СервисИнтеграции, Настройки.КаналИсходящий,
        Настройки.УзелПолучатель, Настройки.КодОтправителя);
    ХешНастроек = КХ_КП_СлужебныйОбмен.Хеш(КХ_КП_СлужебныйОбмен.JSON(Маршрут));
    НачатьТранзакцию();
    Попытка
        Блокировка = Новый БлокировкаДанных;
        Элемент = Блокировка.Добавить("РегистрСведений.КП_СостояниеСправочников");
        Элемент.Режим = РежимБлокировкиДанных.Исключительный;
        Элемент.УстановитьЗначение("Источник", "astor-master");
        Блокировка.Заблокировать();
        СостояниеОбмена = ПолучитьСостояниеОбмена();
        Если СостояниеОбмена.Ревизия <> ПрежняяРевизия Тогда
            ВызватьИсключение "Другой сборщик обновил состояние. Повторите запуск.";
        КонецЕсли;
        Если Не Принудительно И СостояниеОбмена.Хеш = ХешДанных
            И СостояниеОбмена.ХешНастроек = ХешНастроек Тогда
            ЗафиксироватьТранзакцию();
            Возврат Новый Структура("ЕстьИзменения,Ревизия,ИдентификаторСообщения",
                Ложь, СостояниеОбмена.Ревизия, СостояниеОбмена.ИдентификаторПакета);
        КонецЕсли;
        Пакет = КХ_КП_СлужебныйОбмен.СоздатьПакет(Данные,
            СостояниеОбмена.Ревизия + 1, "astor-master", "astor.reference.snapshot");
        КХ_КП_СлужебныйОбмен.ПроверитьСнимок(Пакет, "astor.reference.snapshot", "astor-master");
        ТелоJSON = КХ_КП_СлужебныйОбмен.JSON(Пакет);
        Байты = ПолучитьДвоичныеДанныеИзСтроки(ТелоJSON, КодировкаТекста.UTF8, Ложь);
        Если Байты.Размер() > Настройки.МаксимальныйРазмерСообщения Тогда
            ВызватьИсключение "Снимок превышает лимит. Проверьте лимиты или согласуйте разбиение.";
        КонецЕсли;
        // Новый снимок -> новый объект очереди. Постоянный ID здесь затёр бы старый снимок.
        ПрежнийРежим = ПривилегированныйРежим();
        Попытка
            Интеграция1СШина.ЗарегистрироватьОбъектВОчереди(
                ТелоJSON, Настройки.УзелПолучатель, Настройки.КаналИсходящий,
                Новый УникальныйИдентификатор(Пакет.message_id), КодВнешнейСистемы());
        Исключение
            УстановитьПривилегированныйРежим(ПрежнийРежим);
            ВызватьИсключение;
        КонецПопытки;
        УстановитьПривилегированныйРежим(ПрежнийРежим);
        СостояниеОбмена.Источник = "astor-master";
        СостояниеОбмена.Ревизия = Пакет.revision;
        СостояниеОбмена.Хеш = ХешДанных;
        СостояниеОбмена.ХешНастроек = ХешНастроек;
        СостояниеОбмена.JSONДанных = JSONДанных;
        СостояниеОбмена.ТелоПакета = ТелоJSON;
        СостояниеОбмена.ИдентификаторПакета = Пакет.message_id;
        СостояниеОбмена.Обновлено = ТекущаяДатаСеанса();
        СостояниеОбмена.Записать();
        ЗафиксироватьТранзакцию();
        Возврат Новый Структура("ЕстьИзменения,Ревизия,ИдентификаторСообщения",
            Истина, Пакет.revision, Пакет.message_id);
    Исключение
        ОтменитьТранзакцию();
        ВызватьИсключение;
    КонецПопытки;
