Функция JSON(Значение) Экспорт
    // Ключи объектов сортируются; порядок массивов сохраняется.
    // Только JSON-совместимые значения. Даты заранее преобразуются в ISO-строки.
    ТипЗначения = ТипЗнч(Значение);
    Если ТипЗначения = Тип("Структура") Тогда
        Таблица = Новый ТаблицаЗначений;
        Таблица.Колонки.Добавить("Ключ", Новый ОписаниеТипов("Строка"));
        Таблица.Колонки.Добавить("Значение");
        Для Каждого Элемент Из Значение Цикл
            СтрокаТаблицы = Таблица.Добавить();
            СтрокаТаблицы.Ключ = Элемент.Ключ;
            СтрокаТаблицы.Значение = Элемент.Значение;
        КонецЦикла;
        Таблица.Сортировать("Ключ Возр");
        Части = Новый Массив;
        Для Каждого СтрокаТаблицы Из Таблица Цикл
            Части.Добавить(JSONПримитива(СтрокаТаблицы.Ключ) + ":" + JSON(СтрокаТаблицы.Значение));
        КонецЦикла;
        Возврат "{" + СтрСоединить(Части, ",") + "}";
    ИначеЕсли ТипЗначения = Тип("Массив") Тогда
        Части = Новый Массив;
        Для Каждого Элемент Из Значение Цикл
            Части.Добавить(JSON(Элемент));
        КонецЦикла;
        Возврат "[" + СтрСоединить(Части, ",") + "]";
    ИначеЕсли ТипЗначения = Тип("Строка")
        Или ТипЗначения = Тип("Число")
        Или ТипЗначения = Тип("Булево")
        Или ТипЗначения = Тип("Null") Тогда
        Возврат JSONПримитива(Значение);
    КонецЕсли;
    ВызватьИсключение "Недопустимый тип в JSON: " + Строка(ТипЗначения);
КонецФункции

Функция JSONПримитива(Значение)
    Запись = Новый ЗаписьJSON;
    Запись.УстановитьСтроку();
    ЗаписатьJSON(Запись, Значение);
    Возврат Запись.Закрыть();
КонецФункции

Функция ИзJSON(Текст) Экспорт
    Чтение = Новый ЧтениеJSON;
    Чтение.УстановитьСтроку(Текст);
    Попытка
        Результат = ПрочитатьJSON(Чтение, Ложь);
    Исключение
        Чтение.Закрыть();
        ВызватьИсключение;
    КонецПопытки;
    Чтение.Закрыть();
    Возврат Результат;
КонецФункции

Функция Хеш(Текст) Экспорт
    Данные = ПолучитьДвоичныеДанныеИзСтроки(Текст, КодировкаТекста.UTF8, Ложь);
    Вычислитель = Новый ХешированиеДанных(ХешФункция.SHA256);
    Вычислитель.Добавить(Данные);
    Возврат НРег(ПолучитьHexСтрокуИзДвоичныхДанных(Вычислитель.ХешСумма));
КонецФункции

Функция СоздатьПакет(Данные, Ревизия, Источник, ТипСообщения) Экспорт
    Пакет = Новый Структура;
    Пакет.Вставить("schema_version", 1);
    Пакет.Вставить("message_id", Строка(Новый УникальныйИдентификатор));
    Пакет.Вставить("message_type", ТипСообщения);
    Пакет.Вставить("source_system", Источник);
    Пакет.Вставить("dataset", "supplier-reference");
    Пакет.Вставить("scope", "configured-formats");
    Пакет.Вставить("revision", Ревизия);
    Пакет.Вставить("mode", "snapshot");
    Пакет.Вставить("generated_at", Формат(ТекущаяУниверсальнаяДата(), "ДФ=yyyy-MM-dd'T'HH:mm:ss") + "Z");
    Пакет.Вставить("content_hash", Хеш(JSON(Данные)));
    Пакет.Вставить("payload", Данные);
    Возврат Пакет;
КонецФункции

Функция Поле(Объект, Имя) Экспорт
    Если ТипЗнч(Объект) <> Тип("Структура") Тогда
        ВызватьИсключение "Ожидался JSON-объект.";
    КонецЕсли;
    ЗначениеПоля = Неопределено;
    Если Не Объект.Свойство(Имя, ЗначениеПоля) Тогда
        ВызватьИсключение "Не заполнено поле: " + Имя;
    КонецЕсли;
    Возврат ЗначениеПоля;
КонецФункции

Процедура ПроверитьСнимок(Пакет, ОжидаемыйТип, ОжидаемыйИсточник) Экспорт
    Если Поле(Пакет, "schema_version") <> 1
        Или Поле(Пакет, "message_type") <> ОжидаемыйТип
        Или Поле(Пакет, "source_system") <> ОжидаемыйИсточник
        Или Поле(Пакет, "dataset") <> "supplier-reference"
        Или Поле(Пакет, "scope") <> "configured-formats"
        Или Поле(Пакет, "mode") <> "snapshot" Тогда
        ВызватьИсключение "Неподдерживаемый тип, источник или область пакета.";
    КонецЕсли;
    Идентификатор = Поле(Пакет, "message_id");
    Если ТипЗнч(Идентификатор) <> Тип("Строка") Тогда
        ВызватьИсключение "message_id должен быть строкой UUID.";
    КонецЕсли;
    ПроверкаUUID = Новый УникальныйИдентификатор(Идентификатор);
    Если ПроверкаUUID = Новый УникальныйИдентификатор("00000000-0000-0000-0000-000000000000") Тогда
        ВызватьИсключение "Пустой message_id недопустим.";
    КонецЕсли;
    Ревизия = Поле(Пакет, "revision");
    Если ТипЗнч(Ревизия) <> Тип("Число") Тогда
        ВызватьИсключение "revision должна быть числом.";
    КонецЕсли;
    Если Ревизия <= 0 Или Ревизия <> Цел(Ревизия) Тогда
        ВызватьИсключение "revision должна быть положительным целым числом.";
    КонецЕсли;
    Данные = Поле(Пакет, "payload");
    Если ТипЗнч(Данные) <> Тип("Структура") Тогда
        ВызватьИсключение "payload должен быть объектом.";
    КонецЕсли;
    ИменаМассивов = "categories,formats,territories,structural_units";
    Если ОжидаемыйТип = "astor.reference.snapshot" Тогда
        ИменаМассивов = ИменаМассивов + ",category_format_assignments";
    КонецЕсли;
    Для Каждого ИмяМассива Из СтрРазделить(ИменаМассивов, ",") Цикл
        Если ТипЗнч(Поле(Данные, ИмяМассива)) <> Тип("Массив") Тогда
            ВызватьИсключение "Ожидался массив: " + ИмяМассива;
        КонецЕсли;
    КонецЦикла;
    Если Данные.categories.Количество() = 0 Тогда
        ВызватьИсключение "Пустой список категорий: требуется отдельная проверка полноты.";
    КонецЕсли;
    Если Поле(Пакет, "content_hash") <> Хеш(JSON(Данные)) Тогда
        ВызватьИсключение "Контрольная сумма данных не совпадает.";
    КонецЕсли;
КонецПроцедуры
