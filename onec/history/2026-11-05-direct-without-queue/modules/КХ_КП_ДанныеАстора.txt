Функция ПолучитьНаборРодителей(МоментСреза) Экспорт

    Запрос.Текст = "ВЫБРАТЬ РАЗЛИЧНЫЕ Регистр.Категория КАК Категория ИЗ РегистрСведений.РаспределениеКМПоКатегориям.СрезПоследних(&Период, Формат В (&Направления)) КАК Регистр ГДЕ Регистр.Категория.ПометкаУдаления = ЛОЖЬ И Регистр.Категория.ЭтоГруппа = ИСТИНА И Регистр.Категория В ИЕРАРХИИ(&Корень) УПОРЯДОЧИТЬ ПО Регистр.Категория";
    Корень = ПолучитьГруппыНоменклатуры()[0];
    Запрос.УстановитьПараметр("Период", МоментСреза);
    Запрос.УстановитьПараметр("Направления", ПолучитьСписокНаправлений());
    Запрос.УстановитьПараметр("Корень", Корень);
    Выборка = Запрос.Выполнить().Выбрать();
    Корни = Новый Массив;
    Узлы = Новый Соответствие;
    Пока Выборка.Следующий() Цикл
        Цепочка = ПолучитьЦепочкуРодителей(Выборка.Категория, Корень);
        РодительУзел = Неопределено;
        Для Индекс = 0 По Цепочка.Количество() - 1 Цикл
            Ссылка = Цепочка[Индекс];
            КлючУзла = Строка(Ссылка.УникальныйИдентификатор());
            Узел = Узлы.Получить(КлючУзла);
            Если Узел = Неопределено Тогда
                Узел = Новый Структура;
                Узел.Вставить("guid", КлючУзла);
                Узел.Вставить("name", Ссылка.Наименование);
                Если Индекс = 0 Тогда
                    Узел.Вставить("parent_guid", Null);
                Иначе
                    Узел.Вставить("parent_guid", Строка(Цепочка[Индекс - 1].УникальныйИдентификатор()));
                КонецЕсли;
                Узел.Вставить("level", Индекс + 1);
                Узел.Вставить("children", Новый Массив);
                Узлы.Вставить(КлючУзла, Узел);
                Если РодительУзел = Неопределено Тогда
                    Корни.Добавить(Узел);
                Иначе
                    РодительУзел.children.Добавить(Узел);
                КонецЕсли;
            КонецЕсли;
            РодительУзел = Узел;
        КонецЦикла;
    КонецЦикла;
    Возврат Корни;

Функция ПолучитьЦепочкуРодителей(Группа, Корень)

    Посещенные = Новый Соответствие;
    Текущий = Группа;
    Пока ЗначениеЗаполнено(Текущий) Цикл
        Если Посещенные.Получить(Текущий) <> Неопределено Тогда
            ВызватьИсключение "Обнаружен цикл в иерархии номенклатуры.";
        КонецЕсли;
        Если Текущий.ПометкаУдаления Тогда
            ВызватьИсключение "В цепочке категории есть помеченная на удаление группа.";
        КонецЕсли;
        Посещенные.Вставить(Текущий, Истина);
        Цепочка.Вставить(0, Текущий);
        Если Текущий = Корень Тогда
            Возврат Цепочка;
        КонецЕсли;
        Текущий = Текущий.Родитель;
    КонецЦикла;
    ВызватьИсключение "Категория находится вне настроенного корня.";

// Ответственные берутся из того же распределения, а не из отдельной ручной базы.
Функция ПолучитьОтветственныхДляОбмена(Распределение) Экспорт
    Пользователи = Новый Соответствие;
    Для Каждого Назначение Из Распределение Цикл
        ДобавитьОтветственного(Пользователи, Назначение.km_guid, Назначение.km, Назначение.km_personnel_number, "category_manager");
        ДобавитьОтветственного(Пользователи, Назначение.bayer_guid, Назначение.bayer, Назначение.bayer_personnel_number, "buyer");
    КонецЦикла;
    Таблица = Новый ТаблицаЗначений;
    Таблица.Колонки.Добавить("guid", Новый ОписаниеТипов("Строка"));
    Таблица.Колонки.Добавить("Описание");
    Для Каждого Элемент Из Пользователи Цикл
        СтрокаТаблицы = Таблица.Добавить();
        СтрокаТаблицы.guid = Элемент.Ключ;
        СтрокаТаблицы.Описание = Элемент.Значение;
    КонецЦикла;
    Таблица.Сортировать("guid Возр");
    Результат = Новый Массив;
    Для Каждого СтрокаТаблицы Из Таблица Цикл
        Описание = СтрокаТаблицы.Описание;
        Роли = Новый Массив;
        Если Описание.Байер Тогда
            Роли.Добавить("buyer");
        КонецЕсли;
        Если Описание.КМ Тогда
            Роли.Добавить("category_manager");
        КонецЕсли;
        Пользователь = Новый Структура("guid,name,personnel_number,roles", СтрокаТаблицы.guid, Описание.Имя, Описание.ТабельныйНомер, Роли);
        Результат.Добавить(Пользователь);
    КонецЦикла;
    Возврат Результат;
КонецФункции

Процедура ДобавитьОтветственного(Пользователи, Идентификатор, Имя, ТабельныйНомер, Роль)
    Если ТипЗнч(Идентификатор) = Тип("Null") Тогда
        Возврат;
    КонецЕсли;
    Если ТипЗнч(Идентификатор) <> Тип("Строка") Тогда
        ВызватьИсключение "GUID ответственного должен быть строкой или null.";
    КонецЕсли;
    Если ПустаяСтрока(Идентификатор) Или Идентификатор = "00000000-0000-0000-0000-000000000000" Тогда
        Возврат;
    КонецЕсли;
    Описание = Пользователи.Получить(Идентификатор);
    Если Описание = Неопределено Тогда
        Описание = Новый Структура("Имя,ТабельныйНомер,КМ,Байер", Имя, ТабельныйНомер, Ложь, Ложь);
        Пользователи.Вставить(Идентификатор, Описание);
    Иначе
        Если Описание.Имя <> Имя Или Описание.ТабельныйНомер <> ТабельныйНомер Тогда
            ВызватьИсключение "Разные данные одного ответственного в снимке: " + Идентификатор;
        КонецЕсли;
    КонецЕсли;
    Если Роль = "category_manager" Тогда
        Описание.КМ = Истина;
    ИначеЕсли Роль = "buyer" Тогда
        Описание.Байер = Истина;
    КонецЕсли;
