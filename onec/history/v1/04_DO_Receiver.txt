Функция ПринятьСнимокИзАстора(ТекстJSON) Экспорт
    Пакет = КХ_КП_СлужебныйОбмен.ИзJSON(ТекстJSON);
    КХ_КП_СлужебныйОбмен.ПроверитьСнимок(Пакет, "astor.reference.snapshot", "astor-master");
    КаноническоеТело = КХ_КП_СлужебныйОбмен.JSON(Пакет);
    НачатьТранзакцию();
    Попытка
        КХ_КП_СлужебныйОбмен.ЗаблокироватьСостояние(Пакет.source_system);
        Входящее = РегистрыСведений.КП_СообщенияОбмена.СоздатьМенеджерЗаписи();
        Входящее.Направление = "ИзАстора";
        Входящее.Идентификатор = Пакет.message_id;
        Входящее.Прочитать();
        Если Входящее.Выбран() Тогда
            Если Входящее.ТелоJSON <> КаноническоеТело Тогда
                ВызватьИсключение "Повтор message_id с другим содержимым.";
            КонецЕсли;
            ЗафиксироватьТранзакцию();
            Возврат Новый Структура("status,message_id,revision", "duplicate", Пакет.message_id, Пакет.revision);
        КонецЕсли;
        Состояние = КХ_КП_СлужебныйОбмен.ПолучитьСостояниеОбмена(Пакет.source_system);
        Если Пакет.revision < Состояние.Ревизия Тогда
            КХ_КП_СлужебныйОбмен.ЗаписатьСообщение("ИзАстора", Пакет, "Устаревший");
            ЗафиксироватьТранзакцию();
            Возврат Новый Структура("status,message_id,revision", "stale", Пакет.message_id, Пакет.revision);
        КонецЕсли;
        Если Пакет.revision = Состояние.Ревизия Тогда
            Если Пакет.content_hash <> Состояние.Хеш Тогда
                ВызватьИсключение "Одна ревизия получена с разным содержимым.";
            КонецЕсли;
            КХ_КП_СлужебныйОбмен.ЗаписатьСообщение("ИзАстора", Пакет, "ПовторРевизии");
            ЗафиксироватьТранзакцию();
            Возврат Новый Структура("status,message_id,revision", "duplicate_revision", Пакет.message_id, Пакет.revision);
        КонецЕсли;
        // Сохраняем снимок. Это не назначение исполнителей и не обновление задач ДО.
        Состояние.Источник = Пакет.source_system;
        Состояние.Ревизия = Пакет.revision;
        Состояние.Хеш = Пакет.content_hash;
        Состояние.JSONДанных = КХ_КП_СлужебныйОбмен.JSON(Пакет.payload);
        Состояние.Обновлено = ТекущаяДатаСеанса();
        Состояние.Записать();
        КХ_КП_СлужебныйОбмен.ЗаписатьСообщение("ИзАстора", Пакет, "СнимокСохранен");

        // Внешний состав — предложение контракта, согласовать с разработчиком кабинета.
        // Внутреннее распределение КМ/байеров остаётся в снимке ДО.
        ПубличныеДанные = Новый Структура;
        ПубличныеДанные.Вставить("categories", Пакет.payload.categories);
        ПубличныеДанные.Вставить("formats", Пакет.payload.formats);
        ПубличныеДанные.Вставить("territories", Пакет.payload.territories);
        ПубличныеДанные.Вставить("structural_units", Пакет.payload.structural_units);
        Исходящее = КХ_КП_СлужебныйОбмен.СоздатьПакет(ПубличныеДанные, Пакет.revision, "do-master", "supplier.reference.snapshot");
        Исходящее.Вставить("causation_id", Пакет.message_id);
        КХ_КП_СлужебныйОбмен.ПроверитьСнимок(Исходящее, "supplier.reference.snapshot", "do-master");
        КХ_КП_СлужебныйОбмен.ЗаписатьСообщение("НаБэкенд", Исходящее, "КОтправке");
        ЗафиксироватьТранзакцию();
        Возврат Новый Структура("status,message_id,revision", "snapshot_stored", Пакет.message_id, Пакет.revision);
    Исключение
        ОтменитьТранзакцию();
        ВызватьИсключение;
    КонецПопытки;
КонецФункции
