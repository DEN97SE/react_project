Функция ПодготовитьИзменения() Экспорт
    Источник = "astor-master";
    // Оптимистический контроль двух одновременных сборщиков.
    ПрежняяРевизия = КХ_КП_СлужебныйОбмен.ПолучитьСостояниеОбмена(Источник).Ревизия;
    Данные = КХ_КП_ДанныеАстора.ПолучитьДанныеДляОбмена();
    JSONДанных = КХ_КП_СлужебныйОбмен.JSON(Данные);
    ХешДанных = КХ_КП_СлужебныйОбмен.Хеш(JSONДанных);
    НачатьТранзакцию();
    Попытка
        КХ_КП_СлужебныйОбмен.ЗаблокироватьСостояние(Источник);
        Состояние = КХ_КП_СлужебныйОбмен.ПолучитьСостояниеОбмена(Источник);
        Если Состояние.Ревизия <> ПрежняяРевизия Тогда
            ВызватьИсключение "Другой процесс обновил ревизию во время сборки. Повторите сборку.";
        КонецЕсли;
        Если Состояние.Хеш = ХешДанных Тогда
            ЗафиксироватьТранзакцию();
            Возврат Новый Структура("ЕстьИзменения,Ревизия,Идентификатор", Ложь, Состояние.Ревизия, "");
        КонецЕсли;
        Пакет = КХ_КП_СлужебныйОбмен.СоздатьПакет(Данные, Состояние.Ревизия + 1, Источник, "astor.reference.snapshot");
        КХ_КП_СлужебныйОбмен.ПроверитьСнимок(Пакет, "astor.reference.snapshot", Источник);
        КХ_КП_СлужебныйОбмен.ЗаписатьСообщение("ВШину", Пакет, "КОтправке");
        Состояние.Источник = Источник;
        Состояние.Ревизия = Пакет.revision;
        Состояние.Хеш = ХешДанных;
        Состояние.JSONДанных = JSONДанных;
        Состояние.Обновлено = ТекущаяДатаСеанса();
        Состояние.Записать();
        ЗафиксироватьТранзакцию();
        Возврат Новый Структура("ЕстьИзменения,Ревизия,Идентификатор", Истина, Пакет.revision, Пакет.message_id);
    Исключение

        ВызватьИсключение;
    КонецПопытки;
