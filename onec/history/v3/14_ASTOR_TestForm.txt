// Управляемая форма внешней обработки. Никаких новых регистров/констант.
// Реквизиты: Ревизия (Число 15,0, начальное 1), ТекстПакета,
// РезультатПроверок (оба Строка неограниченной длины).
// Команды: ПроверитьФормат, СформироватьПакет, ЗарегистрироватьИзменения,
// ОтправитьТолькоКабинет, ПоказатьМаршрут.
// Настройки задаются в ПолучитьНастройки() общего модуля, кнопки сохранения нет.

&НаКлиенте
Процедура ПроверитьФормат(Команда)
    РезультатПроверок = ПроверитьФорматНаСервере();
КонецПроцедуры

&НаСервереБезКонтекста
Функция ПроверитьФорматНаСервере()
    Возврат КХ_КП_ТестыОбмена.ВыполнитьПроверки();
КонецФункции

&НаКлиенте
Процедура СформироватьПакет(Команда)
    Пакет = СформироватьПакетНаСервере(Ревизия);
    ТекстПакета = Пакет.ТелоJSON;
    РезультатПроверок = "Предпросмотр без записи в очередь. " + Пакет.ИдентификаторСообщения;
КонецПроцедуры

&НаСервереБезКонтекста
Функция СформироватьПакетНаСервере(НомерРевизии)
    Возврат КХ_КП_ОбменАстор.СформироватьПакетДляШины(НомерРевизии);
КонецФункции

&НаКлиенте
Процедура ЗарегистрироватьИзменения(Команда)
    РезультатПроверок = ЗарегистрироватьНаСервере();
КонецПроцедуры

&НаСервереБезКонтекста
Функция ЗарегистрироватьНаСервере()
    Результат = КХ_КП_ОбменАстор.ПодготовитьИзменения();
    Возврат КХ_КП_СлужебныйОбмен.JSON(Результат);
КонецФункции

&НаКлиенте
Процедура ОтправитьТолькоКабинет(Команда)
    ОтправитьНаСервере();
    РезультатПроверок = "Обработка KP_SUPPLIER завершена. Проверьте штатные статусы ОчередьШины.";
КонецПроцедуры

&НаСервереБезКонтекста
Процедура ОтправитьНаСервере()
    Интеграция1СШина.ОтправитьСообщенияКабинета();
КонецПроцедуры

&НаКлиенте
Процедура ПоказатьМаршрут(Команда)
    РезультатПроверок = МаршрутНаСервере();
КонецПроцедуры

&НаСервереБезКонтекста
Функция МаршрутНаСервере()
    Возврат КХ_КП_СлужебныйОбмен.JSON(КХ_КП_ОбменАстор.ПолучитьНастройки());
КонецФункции
