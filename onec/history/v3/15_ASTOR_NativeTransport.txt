// АСТОР. Общий модуль КХ_КП_ТранспортШины.
// Сервер = Да; оба клиентских флага = Нет; Глобальный = Нет;
// Вызов сервера = Нет; Привилегированный = Нет.
// Код использует API сервисов интеграции из присланного действующего модуля.
// Компиляция и отправка в вашей 1С ещё не выполнялись.
//
// Назначение: отправить уже сохранённый JSON из ОчередьШины для KP_SUPPLIER.
// Объект повторно не сериализуется; ANSI-отправитель других обменов не вызывается.
// Доставка по AMQP 1.0 к бэкенду выполняется далее в Шине, не этим BSL-модулем.

Функция ОтправитьЗаписьОчереди(ЗаписьОчереди) Экспорт
    Статус = Новый Структура;
    ФайловыйПоток = Неопределено;
    ТелоСообщения = Неопределено;
    ИмяВременногоФайла = "";

    Попытка
        Если ЗаписьОчереди.ВнешняяСистема <> "KP_SUPPLIER" Тогда
            ВызватьИсключение "Этот отправитель обслуживает только внешнюю систему KP_SUPPLIER.";
        КонецЕсли;

        Настройки = КХ_КП_ОбменАстор.ПолучитьНастройки();
        Если Не Настройки.Включен Тогда
            // Административное выключение не превращаем в ошибку доставки.
            Возврат Новый Структура("ЕстьОшибки,Пропустить", Ложь, Истина);
        КонецЕсли;

        Если ПустаяСтрока(Настройки.СервисИнтеграции)
            Или ПустаяСтрока(Настройки.КодОтправителя)
            Или ПустаяСтрока(Настройки.КаналИсходящий)
            Или ПустаяСтрока(Настройки.УзелПолучатель) Тогда
            ВызватьИсключение "Не заполнены сервис, отправитель, канал или получатель обмена кабинета.";
        КонецЕсли;

        // Канал и получатель были зафиксированы при регистрации сообщения.
        // Нельзя незаметно отправить старое сообщение по вновь изменённому маршруту.
        Если СокрЛП(ЗаписьОчереди.КаналСервисаИнтеграции)
                <> СокрЛП(Настройки.КаналИсходящий)
            Или СокрЛП(ЗаписьОчереди.УзелПолучатель)
                <> СокрЛП(Настройки.УзелПолучатель) Тогда
            ВызватьИсключение "Маршрут записи очереди отличается от текущих настроек. Требуется явное решение о переносе ожидающих сообщений.";
        КонецЕсли;

        ТекстСообщения = ЗаписьОчереди.СтроковыйОбъект;
        Если ТипЗнч(ТекстСообщения) <> Тип("Строка") Тогда
            ВызватьИсключение "СтроковыйОбъект сообщения KP_SUPPLIER должен содержать строку JSON.";
        КонецЕсли;
        Если ПустаяСтрока(ТекстСообщения) Тогда
            ВызватьИсключение "СтроковыйОбъект сообщения KP_SUPPLIER пуст.";
        КонецЕсли;

        Пакет = КХ_КП_СлужебныйОбмен.ИзJSON(ТекстСообщения);
        КХ_КП_СлужебныйОбмен.ПроверитьСнимок(
            Пакет, "astor.reference.snapshot", "astor-master");

        ИсходящееСообщение = СервисыИнтеграции[
            СокрЛП(Настройки.СервисИнтеграции)].СоздатьСообщение();
        ИсходящееСообщение.КодОтправителя = Настройки.КодОтправителя;
        ИсходящееСообщение.КодПолучателя = ЗаписьОчереди.УзелПолучатель;
        ИсходящееСообщение.Параметры.Вставить("ТипСообщения", Пакет.message_type);
        ИсходящееСообщение.Параметры.Вставить("ИдентификаторПакета", Пакет.message_id);
        ИсходящееСообщение.Параметры.Вставить("РевизияПакета", Пакет.revision);
        ИсходящееСообщение.Параметры.Вставить("ContentType", "application/json; charset=utf-8");
        ИсходящееСообщение.Параметры.Вставить("Кодировка", "UTF-8");

        // Сохраняем точные байты зафиксированного JSON: UTF-8 без BOM и без
        // дополнительного перевода строки, который добавлял бы ЗаписатьСтроку().
        БайтыСообщения = ПолучитьДвоичныеДанныеИзСтроки(
            ТекстСообщения, КодировкаТекста.UTF8, Ложь);
        ИмяВременногоФайла = ПолучитьИмяВременногоФайла("json");
        БайтыСообщения.Записать(ИмяВременногоФайла);
        ФайловыйПоток = ФайловыеПотоки.Открыть(
            ИмяВременногоФайла, РежимОткрытияФайла.Открыть, ДоступКФайлу.Чтение);
        РазмерСообщения = ФайловыйПоток.Размер();
        ТелоСообщения = ИсходящееСообщение.ПолучитьТелоКакПоток();
        ФайловыйПоток.КопироватьВ(ТелоСообщения);
        ТелоСообщения.СброситьБуферы();

        // Закрываем оба потока и удаляем наш временный файл до вызова отправки.
        ФайловыйПоток.Закрыть();
        ФайловыйПоток = Неопределено;
        ТелоСообщения.Закрыть();
        ТелоСообщения = Неопределено;
        УдалитьФайлы(ИмяВременногоФайла);
        ИмяВременногоФайла = "";

        ИсходящееСообщение.Параметры.Вставить("РазмерСообщения", РазмерСообщения);
        СервисыИнтеграции[СокрЛП(Настройки.СервисИнтеграции)][
            СокрЛП(ЗаписьОчереди.КаналСервисаИнтеграции)].ОтправитьСообщение(
                ИсходящееСообщение);

        // Это идентификатор транспорта. Он отличается от стабильного
        // application message_id в JSON. Повторная доставка сохраняет тело JSON.
        Статус.Вставить("ЕстьОшибки", Ложь);
        Статус.Вставить("ИдентификаторСообщения", ИсходящееСообщение.Идентификатор);

    Исключение
        ТекстОшибки = ПодробноеПредставлениеОшибки(ИнформацияОбОшибке());
        ЗакрытьПотокБезИсключения(ФайловыйПоток);
        ЗакрытьПотокБезИсключения(ТелоСообщения);
        УдалитьВременныйФайлБезИсключения(ИмяВременногоФайла);
        Статус = Новый Структура("ЕстьОшибки,ТекстОшибки", Истина, ТекстОшибки);
    КонецПопытки;

    // Дату, статус и транспортный идентификатор записи сохраняет штатный
    // ЗафиксироватьИзмененияВОчередиШины() из модуля Интеграция1СШина.
    Возврат Статус;
КонецФункции

Процедура ЗакрытьПотокБезИсключения(Поток)
    Если Поток = Неопределено Тогда
        Возврат;
    КонецЕсли;
    Попытка
        Поток.Закрыть();
    Исключение
        // При очистке ресурсов сохраняем первичную ошибку отправки.
    КонецПопытки;
КонецПроцедуры

Процедура УдалитьВременныйФайлБезИсключения(ИмяФайла)
    Если ПустаяСтрока(ИмяФайла) Тогда
        Возврат;
    КонецЕсли;
    Попытка
        УдалитьФайлы(ИмяФайла);
    Исключение
        // Удаляется только созданный этой функцией временный файл.
        // Ошибка очистки не подменяет первичную ошибку отправки.
    КонецПопытки;
КонецПроцедуры
