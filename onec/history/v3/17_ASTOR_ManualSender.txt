// Добавить целиком в существующий общий модуль Интеграция1СШина.
// Использует его приватные ОчередьДляВыгрузки и ЗафиксироватьИзмененияВОчередиШины.
// Для первого ручного испытания: отправляет только KP_SUPPLIER.
// Не запускайте одновременно с другим обработчиком той же очереди.

Процедура ОтправитьСообщенияКабинета() Экспорт
    ОчередьВыгрузки = ОчередьДляВыгрузки();
    Для Каждого Запись Из ОчередьВыгрузки Цикл
        Если Запись.ВнешняяСистема <> "KP_SUPPLIER" Тогда
            Продолжить;
        КонецЕсли;
        ПрежнийРежим = ПривилегированныйРежим();
        Попытка
            Статус = КХ_КП_ТранспортШины.ОтправитьЗаписьОчереди(Запись);
            Пропустить = Ложь;
            Статус.Свойство("Пропустить", Пропустить);
            Если Не Пропустить Тогда
                ЗафиксироватьИзмененияВОчередиШины(Запись, Статус);
            КонецЕсли;
        Исключение
            УстановитьПривилегированныйРежим(ПрежнийРежим);
            ВызватьИсключение;
        КонецПопытки;
        УстановитьПривилегированныйРежим(ПрежнийРежим);
    КонецЦикла;
КонецПроцедуры
