Функция ВыполнитьПроверки() Экспорт
    Отчет = Новый Массив;

    ПроверитьРавенство(КХ_КП_СлужебныйОбмен.Хеш(""), "e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855", "SHA-256 пустой строки");
    Отчет.Добавить("PASS: SHA-256 пустой строки");

    ПроверитьРавенство(КХ_КП_СлужебныйОбмен.Хеш("abc"), "ba7816bf8f01cfea414140de5dae2223b00361a396177a9cb410ff61f20015ad", "SHA-256 abc");
    Отчет.Добавить("PASS: SHA-256 abc");

    ПроверитьРавенство(КХ_КП_СлужебныйОбмен.Хеш("Привет"), "dd679c0b9fd408a04148aa7d30c9df393f67b7227f65693fffe0ed6d0f0ade59", "SHA-256 кириллицы UTF-8");
    Отчет.Добавить("PASS: SHA-256 кириллицы UTF-8");

    Первый = Новый Структура;
    Первый.Вставить("b", Ложь);
    Первый.Вставить("a", Null);
    Второй = Новый Структура;
    Второй.Вставить("a", Null);
    Второй.Вставить("b", Ложь);
    ПроверитьРавенство(КХ_КП_СлужебныйОбмен.JSON(Первый), КХ_КП_СлужебныйОбмен.JSON(Второй), "Порядок ключей JSON");
    Отчет.Добавить("PASS: порядок вставки ключей не влияет на JSON");

    ПроверкаТипов = КХ_КП_СлужебныйОбмен.ИзJSON( КХ_КП_СлужебныйОбмен.JSON(Первый));
    Если ТипЗнч(ПроверкаТипов.a) <> Тип("Null") Или ТипЗнч(ПроверкаТипов.b) <> Тип("Булево") Тогда
        ВызватьИсключение "FAIL: null/boolean изменили тип после JSON-преобразования.";
    КонецЕсли;
    Отчет.Добавить("PASS: null и boolean сохраняют типы");

    Категория = Новый Структура;
    Категория.Вставить("guid", "11111111-1111-1111-1111-111111111111");
    Категория.Вставить("name", "Тестовая категория");
    Категория.Вставить("parent_guid", Null);
    Категория.Вставить("level", 1);
    Категория.Вставить("children", Новый Массив);
    Категории = Новый Массив;
    Категории.Добавить(Категория);
    Данные = Новый Структура;
    Данные.Вставить("categories", Категории);
    Данные.Вставить("formats", Новый Массив);
    Данные.Вставить("territories", Новый Массив);
    Данные.Вставить("structural_units", Новый Массив);
    Данные.Вставить("category_format_assignments", Новый Массив);
    Данные.Вставить("responsible_users", Новый Массив);

    ТекстJSON = КХ_КП_СлужебныйОбмен.JSON(Данные);
    КХ_КП_ТранспортШины.ПроверитьТелоJSON(ТекстJSON);
    Копия = КХ_КП_СлужебныйОбмен.ИзJSON(ТекстJSON);
    Если Копия.categories[0].name <> "Тестовая категория" Или ТипЗнч(Копия.responsible_users) <> Тип("Массив") Тогда
        ВызватьИсключение "FAIL: содержимое справочников изменилось после JSON-преобразования.";
    КонецЕсли;
    Отчет.Добавить("PASS: шесть наборов передаются без дополнительной оболочки");
    Для Каждого НедопустимыйJSON Из СтрРазделить("null|true|123|""текст""|{", "|") Цикл
        ОшибкаПоймана = Ложь;
        Попытка
            КХ_КП_ТранспортШины.ПроверитьТелоJSON(НедопустимыйJSON);
        Исключение
            ОшибкаПоймана = Истина;
        КонецПопытки;
        Если Не ОшибкаПоймана Тогда
            ВызватьИсключение "FAIL: недопустимое тело сообщения принято.";
        КонецЕсли;
    КонецЦикла;
    Отчет.Добавить("PASS: примитивы и повреждённый JSON не отправляются");
    Распределение = Новый Массив;
    Назначение = Новый Структура;
    Назначение.Вставить("km_guid", "11111111-1111-1111-1111-111111111111");
    Назначение.Вставить("km", "Иванов");
    Назначение.Вставить("km_personnel_number", "001");
    Назначение.Вставить("bayer_guid", "22222222-2222-2222-2222-222222222222");
    Назначение.Вставить("bayer", "Петров");
    Назначение.Вставить("bayer_personnel_number", "002");
    Распределение.Добавить(Назначение);
    ВтороеНазначение = КХ_КП_СлужебныйОбмен.ИзJSON(КХ_КП_СлужебныйОбмен.JSON(Назначение));
    ВтороеНазначение.bayer_guid = Назначение.km_guid;
    ВтороеНазначение.bayer = Назначение.km;
    ВтороеНазначение.bayer_personnel_number = Назначение.km_personnel_number;
    Распределение.Добавить(ВтороеНазначение);
    Ответственные = КХ_КП_ДанныеАстора.ПолучитьОтветственныхДляОбмена(Распределение);
    Если Ответственные.Количество() <> 2 Или Ответственные[0].roles.Количество() <> 2 Тогда
        ВызватьИсключение "FAIL: ответственный с двумя ролями продублирован или потерял роль.";
    КонецЕсли;
    Отчет.Добавить("PASS: ответственные из распределения объединяются по GUID и сохраняют обе роли");
    НачалоUTC = Дата(2026, 11, 5, 12, 0, 0);
    ПроверитьРавенство(КХ_КП_ОчередьКабинета.СрокЖизниСекунд(), 259200, "TTL = 72 часа");
    СрокUTC = НачалоUTC + КХ_КП_ОчередьКабинета.СрокЖизниСекунд();
    ПроверитьРавенство(КХ_КП_ОчередьКабинета.СрокИстек(СрокUTC, СрокUTC - 1), Ложь, "TTL до границы");
    ПроверитьРавенство(КХ_КП_ОчередьКабинета.СрокИстек(СрокUTC, СрокUTC), Истина, "TTL на границе");
    ПроверитьРавенство(КХ_КП_ОчередьКабинета.СрокИстек(СрокUTC, СрокUTC + 1), Истина, "TTL после границы");
    ПроверитьРавенство(КХ_КП_ОчередьКабинета.ВременныйHTTPКод(503), Истина, "503 требует повтора");
    ПроверитьРавенство(КХ_КП_ОчередьКабинета.ВременныйHTTPКод(429), Истина, "429 требует паузы");
    ПроверитьРавенство(КХ_КП_ОчередьКабинета.ВременныйHTTPКод(400), Ложь, "400 не повторяется автоматически");
    ПроверитьРавенство(КХ_КП_ОчередьКабинета.ВременныйHTTPКод(202), Ложь, "202 не подтверждает применение");
    Отчет.Добавить("PASS: границы 72 часов и коды временных ошибок");
    Возврат СтрСоединить(Отчет, Символы.ПС);
КонецФункции

Процедура ПроверитьРавенство(Фактическое, Ожидаемое, Название)
    Если Фактическое <> Ожидаемое Тогда
        ВызватьИсключение "FAIL: " + Название + ". Получено: " + Строка(Фактическое);
    КонецЕсли;
