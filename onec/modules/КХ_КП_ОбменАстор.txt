Функция ПолучитьНастройки() Экспорт
    Настройки = Новый Структура;
    Настройки.Вставить("Включен", Ложь);
    Настройки.Вставить("СервисИнтеграции", "");
    Настройки.Вставить("КаналИсходящий", "");
    Настройки.Вставить("КодОтправителя", "");
    Настройки.Вставить("КодПолучателя", "");
    Настройки.Вставить("МаксимальныйРазмерСообщения", 5242880);
    Настройки.Вставить("ИнтервалПовтораСекунд", 60);
    Настройки.Вставить("ТаймаутОжиданияОтветаСекунд", 900);
    Возврат Настройки;
КонецФункции

Процедура ПроверитьНастройки(Настройки) Экспорт
    Если Не Настройки.Включен Тогда
        ВызватьИсключение "Обмен кабинета выключен в ПолучитьНастройки().";
    КонецЕсли;
    Для Каждого Имя Из СтрРазделить("СервисИнтеграции,КаналИсходящий,КодОтправителя,КодПолучателя", ",") Цикл
        ЗначениеПоля = КХ_КП_СлужебныйОбмен.Поле(Настройки, Имя);
        Если ТипЗнч(ЗначениеПоля) <> Тип("Строка") Тогда
            ВызватьИсключение "Настройка должна быть строкой: " + Имя;
        КонецЕсли;
        Если ПустаяСтрока(ЗначениеПоля) Или ЗначениеПоля <> СокрЛП(ЗначениеПоля) Тогда
            ВызватьИсключение "Заполните настройку без крайних пробелов: " + Имя;
        КонецЕсли;
    КонецЦикла;
    Если ТипЗнч(Настройки.МаксимальныйРазмерСообщения) <> Тип("Число") Тогда
        ВызватьИсключение "Лимит размера должен быть числом байт.";
    КонецЕсли;
    Если Настройки.МаксимальныйРазмерСообщения <= 0 Или Настройки.МаксимальныйРазмерСообщения <> Цел(Настройки.МаксимальныйРазмерСообщения) Тогда
        ВызватьИсключение "Лимит размера должен быть положительным целым числом.";
    КонецЕсли;
    Для Каждого Имя Из СтрРазделить("ИнтервалПовтораСекунд,ТаймаутОжиданияОтветаСекунд", ",") Цикл
        Интервал = КХ_КП_СлужебныйОбмен.Поле(Настройки, Имя);
        Если ТипЗнч(Интервал) <> Тип("Число") Тогда
            ВызватьИсключение "Интервал должен быть числом: " + Имя;
        КонецЕсли;
        Если Интервал <= 0 Или Интервал <> Цел(Интервал) Тогда
            ВызватьИсключение "Интервал должен быть положительным целым: " + Имя;
        КонецЕсли;
    КонецЦикла;
КонецПроцедуры

Функция СформироватьJSON() Экспорт
    Данные = КХ_КП_ДанныеАстора.ПолучитьДанныеДляОбмена();
    Возврат КХ_КП_СлужебныйОбмен.JSON(Данные);
КонецФункции

Функция ОтправитьПолныйСнимок() Экспорт
    // Шесть наборов текущего сборщика. Совместимость полей с бэком сверить отдельно.
    // Метод теперь только регистрирует полный JSON; отправка выполняется отдельным заданием.
    Настройки = ПолучитьНастройки();
    ПроверитьНастройки(Настройки);
    Возврат КХ_КП_ОчередьКабинета.ПоставитьJSON(СформироватьJSON());
КонецФункции
