Функция ОбработатьОтчетШины(ТекстJSON) Экспорт
    Ответ = КХ_КП_СлужебныйОбмен.ИзJSON(ТекстJSON);
    Если КХ_КП_СлужебныйОбмен.Поле(Ответ, "message_type") <> "supplier.http.result" Тогда
        ВызватьИсключение "Неизвестный тип отчёта Шины.";
    КонецЕсли;
    IDПакета = КХ_КП_СлужебныйОбмен.Поле(Ответ, "packet_id");
    IDПопытки = КХ_КП_СлужебныйОбмен.Поле(Ответ, "attempt_id");
    КодHTTP = КХ_КП_СлужебныйОбмен.Поле(Ответ, "http_status");
    Применено = КХ_КП_СлужебныйОбмен.Поле(Ответ, "applied");
    ПовторЧерез = 0;
    Ответ.Свойство("retry_after_seconds", ПовторЧерез);
    Возврат КХ_КП_ОчередьКабинета.ОбработатьРезультатHTTP(IDПакета, IDПопытки, КодHTTP, Применено, ПовторЧерез);
КонецФункции
