Функция СистемыСобственнойОчереди() Экспорт
    // Общий перечень для исключения систем из штатной обработки.
    // Каждый собственный отправитель выбирает только свою группу через СистемыОбмена().
    Системы = Новый Массив;
    Для Каждого Описание Из ОписанияСистем() Цикл
        Если Описание.Обмен = "KP_SUPPLIER" И Не ВключенСобственныйОбработчикКабинета() Тогда
            Продолжить;
        КонецЕсли;
        Если Системы.Найти(Описание.Метка) = Неопределено Тогда
            Системы.Добавить(Описание.Метка);
        КонецЕсли;
    КонецЦикла;
    Возврат Системы;
КонецФункции

Функция СистемыОбмена(ИмяОбмена) Экспорт
    // Группа меток для конкретного отправителя. Это не признак включения обработчика.
    Системы = Новый Массив;
    Для Каждого Описание Из ОписанияСистем() Цикл
        Если Описание.Обмен = ИмяОбмена Тогда
            Системы.Добавить(Описание.Метка);
        КонецЕсли;
    КонецЦикла;
    Если Системы.Количество() = 0 Тогда
        ВызватьИсключение "Неизвестный обмен: " + ИмяОбмена;
    КонецЕсли;
    Возврат Системы;
КонецФункции

Функция ЭтоСистемаСобственнойОчереди(ВнешняяСистема) Экспорт
    Возврат СистемыСобственнойОчереди().Найти(ВнешняяСистема) <> Неопределено;
КонецФункции
