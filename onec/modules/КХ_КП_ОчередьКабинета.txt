Функция ПоставитьJSON(ТекстJSON) Экспорт
    Настройки = КХ_КП_ОбменАстор.ПолучитьНастройки();

    ПроверитьИзоляцию();

    Если ПолучитьДвоичныеДанныеИзСтроки(ТекстJSON, КодировкаТекста.UTF8, Ложь).Размер() > Настройки.МаксимальныйРазмерСообщения Тогда
        ВызватьИсключение "JSON превышает лимит сообщения.";
    КонецЕсли;
    UUIDПакета = Новый УникальныйИдентификатор;
    Идентификатор = XMLСтрока(UUIDПакета);
    ХешТела = КХ_КП_СлужебныйОбмен.Хеш(ТекстJSON);
    ПрежнийРежим = ПривилегированныйРежим();
    НачатьТранзакцию();
    Попытка
        Заблокировать(Идентификатор, Настройки.КодПолучателя);
        // Штатный регистратор заменяет строки по ID+узлу: не допускаем совпадения с чужой записью.
        Проверка = Новый Запрос("ВЫБРАТЬ ПЕРВЫЕ 1 Очередь.ИдентификаторОбъекта ИЗ РегистрСведений.ОчередьШины КАК Очередь ГДЕ Очередь.ИдентификаторОбъекта = &ID И Очередь.УзелПолучатель = &Узел");

            ВызватьИсключение "Идентификатор уже занят; регистрация без перезаписи существующих строк остановлена.";
        КонецЕсли;
        Интеграция1СШина.ЗарегистрироватьОбъектВОчереди(ТекстJSON, Настройки.КодПолучателя, Настройки.КаналИсходящий, UUIDПакета, "KP_SUPPLIER");
        УстановитьПривилегированныйРежим(ПрежнийРежим);
        Запись = ПрочитатьЗапись(Идентификатор);
        Если Запись = Неопределено Тогда

        СохранитьЗапись(Запись);
        ЗафиксироватьТранзакцию();
    Исключение
        УстановитьПривилегированныйРежим(ПрежнийРежим);
        ОтменитьТранзакцию();
        ВызватьИсключение;
    КонецПопытки;
    Возврат Новый Структура("ПоставленоВОчередь,ИдентификаторПакета,СрокUTC", Истина, Строка(UUIDПакета), Маркер.СрокUTC);

        ЭлементID.УстановитьЗначение("УзелПолучатель", Узел);

Функция ПрочитатьЗапись(Идентификатор)
    Запрос = Новый Запрос("ВЫБРАТЬ Очередь.ИдентификаторОбъекта КАК ИдентификаторОбъекта, Очередь.ИдентификаторСообщения КАК ИдентификаторСообщения, Очередь.КаналСервисаИнтеграции КАК КаналСервисаИнтеграции, Очередь.УзелПолучатель КАК УзелПолучатель, Очередь.ВнешняяСистема КАК ВнешняяСистема, Очередь.Объект КАК Объект, Очередь.ДатаРегистрации КАК ДатаРегистрации, Очередь.ДатаОтправки КАК ДатаОтправки, Очередь.СтатусСообщения КАК СтатусСообщения, Очередь.ТекстОшибки КАК ТекстОшибки, Очередь.ДатаЗагрузки КАК ДатаЗагрузки, Очередь.СтроковыйОбъект КАК СтроковыйОбъект ИЗ РегистрСведений.ОчередьШины КАК Очередь ГДЕ Очередь.ВнешняяСистема = &Система И Очередь.ИдентификаторОбъекта = &ID");
    Запрос.УстановитьПараметр("Система", "KP_SUPPLIER");
    Запрос.УстановитьПараметр("ID", Идентификатор);
    Выборка = Запрос.Выполнить().Выбрать();
    Если Не Выборка.Следующий() Тогда
        Возврат Неопределено;
    КонецЕсли;
    Запись = Новый Структура("ИдентификаторОбъекта,ИдентификаторСообщения,КаналСервисаИнтеграции,УзелПолучатель,ВнешняяСистема,Объект,ДатаРегистрации,ДатаОтправки,СтатусСообщения,ТекстОшибки,ДатаЗагрузки,СтроковыйОбъект");
    ЗаполнитьЗначенияСвойств(Запись, Выборка);
    Если Выборка.Следующий() Тогда
        ВызватьИсключение "У одного ID кабинета несколько записей: автоматическое изменение остановлено.";
    КонецЕсли;
    Возврат Запись;

Функция ПрочитатьМаркер(Запись)
    Если Запись = Неопределено Тогда
        Возврат Неопределено;
    КонецЕсли;
    Попытка
        Маркер = ЗначениеИзСтрокиВнутр(Запись.Объект);
    Исключение
        Возврат Неопределено;
    КонецПопытки;
    Формат = "";
    Если ТипЗнч(Маркер) <> Тип("Структура") Тогда
        Возврат Неопределено;
    КонецЕсли;
    Если Не Маркер.Свойство("Формат", Формат) Или Формат <> "KP_HTTP_BUFFER_V1" Тогда
        Возврат Неопределено;
    КонецЕсли;
    Возврат Маркер;

Функция ПодготовитьПопытку(Идентификатор, Настройки)
    НачатьТранзакцию();
    Попытка
        Заблокировать();
        Запись = ПрочитатьЗапись(Идентификатор);
        Маркер = ПрочитатьМаркер(Запись);
        Если Маркер = Неопределено Тогда
            ЗафиксироватьТранзакцию();
            Возврат Неопределено;
        КонецЕсли;
        СейчасUTC = ТекущаяУниверсальнаяДата();
        Если СрокИстек(Маркер.СрокUTC, СейчасUTC) Тогда
            УдалитьЗапись(Запись, Истина);
            ЗафиксироватьТранзакцию();
            Возврат Неопределено;
        КонецЕсли;
        Разрешено = Запись.СтатусСообщения = Перечисления.СтатусыСообщенийШины.ОжидаетОтправки Или Запись.СтатусСообщения = Перечисления.СтатусыСообщенийШины.Ошибка Или Запись.СтатусСообщения = Перечисления.СтатусыСообщенийШины.Отправлено;
        Если Не Разрешено Или СейчасUTC < Маркер.СледующаяПопыткаUTC Тогда
            ЗафиксироватьТранзакцию();
            Возврат Неопределено;
        КонецЕсли;
        Если Маркер.Сервис <> Настройки.СервисИнтеграции Или Маркер.Отправитель <> Настройки.КодОтправителя Или Запись.КаналСервисаИнтеграции <> Настройки.КаналИсходящий Или Запись.УзелПолучатель <> Настройки.КодПолучателя Тогда
            ВызватьИсключение "Нельзя отправить старый пакет по изменённому маршруту.";
        КонецЕсли;
        Если Маркер.Хеш <> КХ_КП_СлужебныйОбмен.Хеш(Запись.СтроковыйОбъект) Тогда
            ВызватьИсключение "Тело сохранённого пакета изменено.";
        КонецЕсли;
        Маркер.ИдентификаторПопытки = Строка(Новый УникальныйИдентификатор);
        Маркер.РезультатПопытки = "";
        Маркер.СледующаяПопыткаUTC = СейчасUTC + Настройки.ТаймаутОжиданияОтветаСекунд;
        Запись.СтатусСообщения = Перечисления.СтатусыСообщенийШины.Отправлено;
        Запись.ДатаОтправки = ТекущаяДатаСеанса();
        Запись.ТекстОшибки = "";
        Запись.Объект = ЗначениеВСтрокуВнутр(Маркер);
        СохранитьЗапись(Запись);
        ЗафиксироватьТранзакцию();
        Возврат Новый Структура("ИдентификаторПакета,ИдентификаторПопытки,ТекстJSON,СрокUTC", Маркер.ИдентификаторПакета, Маркер.ИдентификаторПопытки, Запись.СтроковыйОбъект, Маркер.СрокUTC);
    Исключение
        ОтменитьТранзакцию();
        ВызватьИсключение;
    КонецПопытки;
КонецФункции

    Настройки = КХ_КП_ОбменАстор.ПолучитьНастройки();

            ЗаписьЖурналаРегистрации("КП.ОбработкаОчереди", УровеньЖурналаРегистрации.Ошибка, , , "ID: " + Выборка.ID + "; " + ОписаниеОшибки());
        КонецПопытки;

Процедура ЗавершитьОтправку(Идентификатор, ПопыткаОтправки, Результат, ТекстОшибки, Настройки)
    НачатьТранзакцию();
    Попытка
        Заблокировать();
        Запись = ПрочитатьЗапись(Идентификатор);
        Маркер = ПрочитатьМаркер(Запись);
        // Ранний успех бэка мог уже удалить строку. Не создаём её заново.
        Если Маркер = Неопределено Тогда
            ЗафиксироватьТранзакцию();
            Возврат;
        КонецЕсли;
        Если Маркер.ИдентификаторПопытки <> ПопыткаОтправки Тогда
            ЗафиксироватьТранзакцию();
            Возврат;
        КонецЕсли;
        Если Результат <> Неопределено Тогда
            Маркер.ТранспортID = Строка(Результат.ИдентификаторСообщения);
        ИначеЕсли Маркер.РезультатПопытки <> ПопыткаОтправки Тогда
            Запись.СтатусСообщения = Перечисления.СтатусыСообщенийШины.Ошибка;
            Запись.ТекстОшибки = ТекстОшибки;
            Маркер.СледующаяПопыткаUTC = ТекущаяУниверсальнаяДата() + Настройки.ИнтервалПовтораСекунд;
        КонецЕсли;
        Запись.Объект = ЗначениеВСтрокуВнутр(Маркер);
        СохранитьЗапись(Запись);
        ЗафиксироватьТранзакцию();
    Исключение
        ОтменитьТранзакцию();
        ВызватьИсключение;
    КонецПопытки;
КонецПроцедуры

Функция ОбработатьРезультатHTTP(ИдентификаторПакета, ИдентификаторПопытки, КодHTTP, Применено = Ложь, ПовторЧерезСекунд = 0) Экспорт
    // Только после проверки канала/отправителя. Применено выставляет разбор согласованного ответа бэка.
    ПроверитьИзоляцию();
    Идентификатор = XMLСтрока(Новый УникальныйИдентификатор(ИдентификаторПакета));
    ПопыткаОтвета = Строка(Новый УникальныйИдентификатор(ИдентификаторПопытки));
    Если ПопыткаОтвета = "00000000-0000-0000-0000-000000000000" Или Идентификатор = XMLСтрока(Новый УникальныйИдентификатор("00000000-0000-0000-0000-000000000000")) Тогда
        ВызватьИсключение "Пустые идентификаторы результата недопустимы.";
    КонецЕсли;
    Если ТипЗнч(КодHTTP) <> Тип("Число") Или ТипЗнч(Применено) <> Тип("Булево") Или ТипЗнч(ПовторЧерезСекунд) <> Тип("Число") Тогда
        ВызватьИсключение "Недопустимые типы результата HTTP.";
    КонецЕсли;
    Если КодHTTP <> Цел(КодHTTP) Или (КодHTTP <> 0 И (КодHTTP < 100 Или КодHTTP > 599)) Или ПовторЧерезСекунд < 0 Или ПовторЧерезСекунд > 31536000 Или ПовторЧерезСекунд <> Цел(ПовторЧерезСекунд) Тогда
        ВызватьИсключение "Недопустимый код HTTP или Retry-After.";
    КонецЕсли;
    Если Применено И КодHTTP <> 200 Тогда
        ВызватьИсключение "Код успеха требует согласования: в текущем контракте только 200.";
    КонецЕсли;
    Настройки = КХ_КП_ОбменАстор.ПолучитьНастройки();
    НачатьТранзакцию();
    Попытка
        Заблокировать();
        Запись = ПрочитатьЗапись(Идентификатор);
        Маркер = ПрочитатьМаркер(Запись);
        Если Маркер = Неопределено Тогда
            ЗафиксироватьТранзакцию();
            Возврат Истина;
        КонецЕсли;
        Если Маркер.ИдентификаторПопытки = "" Тогда
            ВызватьИсключение "Пакет ещё не отправлялся.";
        КонецЕсли;
        Если Маркер.Хеш <> КХ_КП_СлужебныйОбмен.Хеш(Запись.СтроковыйОбъект) Тогда
            ВызватьИсключение "Содержимое пакета изменено после регистрации.";
        КонецЕсли;
        Если Применено Тогда
            ЗаписьЖурналаРегистрации("КП.ПодтвержденоПрименение", УровеньЖурналаРегистрации.Информация, , , "Бэк подтвердил применение. ID: " + Маркер.ИдентификаторПакета + "; попытка: " + ПопыткаОтвета);
            УдалитьЗапись(Запись);
        ИначеЕсли Маркер.ИдентификаторПопытки = ПопыткаОтвета И Маркер.РезультатПопытки <> ПопыткаОтвета И Запись.СтатусСообщения <> Перечисления.СтатусыСообщенийШины.Отменено Тогда
            Маркер.РезультатПопытки = ПопыткаОтвета;
            Запись.ТекстОшибки = "HTTP " + СокрЛП(Строка(КодHTTP)) + ": применение не подтверждено";
            Если ВременныйHTTPКод(КодHTTP) Тогда
                Запись.СтатусСообщения = Перечисления.СтатусыСообщенийШины.Ошибка;
                Маркер.СледующаяПопыткаUTC = ТекущаяУниверсальнаяДата() + Макс(Настройки.ИнтервалПовтораСекунд, ПовторЧерезСекунд);
            Иначе
                Запись.СтатусСообщения = Перечисления.СтатусыСообщенийШины.Отменено;
            КонецЕсли;
            Запись.Объект = ЗначениеВСтрокуВнутр(Маркер);
            СохранитьЗапись(Запись);
        КонецЕсли;
        ЗафиксироватьТранзакцию();
        Возврат Истина;
    Исключение
        ОтменитьТранзакцию();
        ВызватьИсключение;
    КонецПопытки;
