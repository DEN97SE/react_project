// Собственный обработчик KP_SUPPLIER подготовлен текстом. Штатный модуль не меняется; в 1С код ещё не проверялся.
// Интеграция1СШина остаётся исходным. Прежние вставки в него не применять.
// АСТОР. Общий модуль КХ_КП_ТранспортШины.
// Сервер = Да; оба клиентских флага = Нет; Глобальный = Нет;
// Вызов сервера = Нет; Привилегированный = Нет.
// Код использует API сервисов интеграции из присланного действующего модуля.
// Компиляция и отправка в вашей 1С ещё не выполнялись.
//
// Назначение: отправить уже сохранённый JSON из ОчередьШины для KP_SUPPLIER.
// Объект повторно не сериализуется; ANSI-отправитель других обменов не вызывается.
// HTTP к бэкенду выполняет Шина. Здесь только её штатный сервис интеграции.
// Новые HTTP-попытки и сроки повторов хранятся в существующем поле Объект.

Функция ОтправитьЗаписьОчереди(ЗаписьОчереди) Экспорт
    Статус = Новый Структура;
    ФайловыйПоток = Неопределено;
    ТелоСообщения = Неопределено;
    ИмяВременногоФайла = "";
    ИдентификаторПопытки = "";

    Попытка
        ПроверитьИзоляциюОчереди();
        Если ЗаписьОчереди.ВнешняяСистема <> "KP_SUPPLIER" Тогда
            ВызватьИсключение "Этот отправитель обслуживает только внешнюю систему KP_SUPPLIER.";
        КонецЕсли;

        Настройки = КХ_КП_ОбменАстор.ПолучитьНастройки();
        Если Не Настройки.Включен Тогда
            // Административное выключение не превращаем в ошибку доставки.
            Возврат Новый Структура("ЕстьОшибки,Пропустить", Ложь, Истина);
        КонецЕсли;

        Если ПустаяСтрока(Настройки.СервисИнтеграции) Или ПустаяСтрока(Настройки.КодОтправителя) Или ПустаяСтрока(Настройки.КаналИсходящий) Или ПустаяСтрока(Настройки.УзелПолучатель) Тогда
            ВызватьИсключение "Не заполнены сервис, отправитель, канал или получатель обмена кабинета.";
        КонецЕсли;

        КХ_КП_ОбменАстор.ПроверитьНастройки(Настройки);

        // Канал и получатель были зафиксированы при регистрации сообщения.
        // Нельзя незаметно отправить старое сообщение по вновь изменённому маршруту.
        Если СокрЛП(ЗаписьОчереди.КаналСервисаИнтеграции) <> СокрЛП(Настройки.КаналИсходящий) Или СокрЛП(ЗаписьОчереди.УзелПолучатель) <> СокрЛП(Настройки.УзелПолучатель) Тогда
            ВызватьИсключение "Маршрут записи очереди отличается от текущих настроек. Требуется явное решение о переносе ожидающих сообщений.";
        КонецЕсли;

        ТекстСообщения = ЗаписьОчереди.СтроковыйОбъект;
        Если ТипЗнч(ТекстСообщения) <> Тип("Строка") Тогда
            ВызватьИсключение "СтроковыйОбъект сообщения KP_SUPPLIER должен содержать строку JSON.";
        КонецЕсли;
        Если ПустаяСтрока(ТекстСообщения) Тогда
            ВызватьИсключение "СтроковыйОбъект сообщения KP_SUPPLIER пуст.";
        КонецЕсли;

        Пакет = КХ_КП_СлужебныйОбмен.ИзJSON(ТекстСообщения);
        КХ_КП_СлужебныйОбмен.ПроверитьСнимок( Пакет, "astor.reference.snapshot", "astor-master");
        Если КХ_КП_СлужебныйОбмен.Поле(Пакет, "route_hash") <> КХ_КП_ОбменАстор.ХешМаршрута(Настройки) Тогда
            ВызватьИсключение "Пакет создан для другого маршрута. Автоматический перенос на HTTP запрещён.";
        КонецЕсли;

        ИдентификаторПопытки = Строка(Новый УникальныйИдентификатор);
        Если Не НачатьHTTPПопытку(ЗаписьОчереди.ИдентификаторОбъекта, ИдентификаторПопытки, Настройки) Тогда
            Возврат Новый Структура("ЕстьОшибки,Пропустить", Ложь, Истина);
        КонецЕсли;
        ИсходящееСообщение = СервисыИнтеграции[СокрЛП(Настройки.СервисИнтеграции)].СоздатьСообщение();
        ИсходящееСообщение.КодОтправителя = Настройки.КодОтправителя;
        ИсходящееСообщение.КодПолучателя = ЗаписьОчереди.УзелПолучатель;
        ИсходящееСообщение.Параметры.Вставить("ТипСообщения", Пакет.message_type);
        ИсходящееСообщение.Параметры.Вставить("ИдентификаторПопыткиHTTP", ИдентификаторПопытки);
        ИсходящееСообщение.Параметры.Вставить("ВерсияHTTPМаршрута", Настройки.ВерсияHTTPМаршрута);
        ИсходящееСообщение.Параметры.Вставить("ИдентификаторПакета", Пакет.message_id);
        ИсходящееСообщение.Параметры.Вставить("РевизияПакета", Пакет.revision);
        ИсходящееСообщение.Параметры.Вставить("ContentType", "application/json; charset=utf-8");
        ИсходящееСообщение.Параметры.Вставить("Кодировка", "UTF-8");

        // Сохраняем точные байты зафиксированного JSON: UTF-8 без BOM и без
        // дополнительного перевода строки, который добавлял бы ЗаписатьСтроку().
        БайтыСообщения = ПолучитьДвоичныеДанныеИзСтроки( ТекстСообщения, КодировкаТекста.UTF8, Ложь);
        ИмяВременногоФайла = ПолучитьИмяВременногоФайла("json");
        БайтыСообщения.Записать(ИмяВременногоФайла);
        ФайловыйПоток = ФайловыеПотоки.Открыть( ИмяВременногоФайла, РежимОткрытияФайла.Открыть, ДоступКФайлу.Чтение);
        РазмерСообщения = ФайловыйПоток.Размер();
        ТелоСообщения = ИсходящееСообщение.ПолучитьТелоКакПоток();
        ФайловыйПоток.КопироватьВ(ТелоСообщения);
        ТелоСообщения.СброситьБуферы();

        // Закрываем оба потока и удаляем наш временный файл до вызова отправки.
        ФайловыйПоток.Закрыть();
        ФайловыйПоток = Неопределено;
        ТелоСообщения.Закрыть();
        ТелоСообщения = Неопределено;
        УдалитьФайлы(ИмяВременногоФайла);
        ИмяВременногоФайла = "";

        ИсходящееСообщение.Параметры.Вставить("РазмерСообщения", РазмерСообщения);
        СервисыИнтеграции[СокрЛП(Настройки.СервисИнтеграции)][ СокрЛП(ЗаписьОчереди.КаналСервисаИнтеграции)].ОтправитьСообщение( ИсходящееСообщение);

        // Это идентификатор транспорта. Он отличается от стабильного
        // application message_id в JSON. Повторная доставка сохраняет тело JSON.
        Статус.Вставить("ЕстьОшибки", Ложь);
        Статус.Вставить("ИдентификаторСообщения", ИсходящееСообщение.Идентификатор);

    Исключение
        ТекстОшибки = ПодробноеПредставлениеОшибки(ИнформацияОбОшибке());
        ЗакрытьПотокБезИсключения(ФайловыйПоток);
        ЗакрытьПотокБезИсключения(ТелоСообщения);
        УдалитьВременныйФайлБезИсключения(ИмяВременногоФайла);
        Статус = Новый Структура("ЕстьОшибки,ТекстОшибки", Истина, ТекстОшибки);
    КонецПопытки;

    // Результат должен фиксировать наш обработчик собственной очереди.
    // Приватные процедуры Интеграция1СШина отсюда не вызываются.
    Статус.Вставить("ИдентификаторПопытки", ИдентификаторПопытки);
    Возврат Статус;
КонецФункции

Процедура ЗакрытьПотокБезИсключения(Поток)
    Если Поток = Неопределено Тогда
        Возврат;
    КонецЕсли;
    Попытка
        Поток.Закрыть();
    Исключение
        // При очистке ресурсов сохраняем первичную ошибку отправки.
    КонецПопытки;
КонецПроцедуры

Процедура УдалитьВременныйФайлБезИсключения(ИмяФайла)
    Если ПустаяСтрока(ИмяФайла) Тогда
        Возврат;
    КонецЕсли;
    Попытка
        УдалитьФайлы(ИмяФайла);
    Исключение
        // Удаляется только созданный этой функцией временный файл.
        // Ошибка очистки не подменяет первичную ошибку отправки.
    КонецПопытки;
КонецПроцедуры


Процедура ПроверитьИзоляциюОчереди() Экспорт
    // Штатный код уже использует этот перечень для исключения собственных систем.
    Если Не КХ_ОбменШиныОбщий.ЭтоСистемаСобственнойОчереди("KP_SUPPLIER") Тогда
        ВызватьИсключение "Собственный обработчик кабинета ещё не включён в общем реестре.";
    КонецЕсли;
    // Проверяем также фактическую старую точку входа, которую вызывает штатный модуль.
    Системы = КХ_ОбменШиныОпера.СистемыСобственнойОчереди();
    Включена = Ложь;
    Если ТипЗнч(Системы) = Тип("Массив") Тогда
        Включена = Системы.Найти("KP_SUPPLIER") <> Неопределено;
    ИначеЕсли ТипЗнч(Системы) = Тип("СписокЗначений") Тогда
        Включена = Системы.НайтиПоЗначению("KP_SUPPLIER") <> Неопределено;
    Иначе
        ВызватьИсключение "Неизвестный формат списка собственных систем. Сначала проверить реализацию СистемыСобственнойОчереди().";
    КонецЕсли;
    Если Не Включена Тогда
        ВызватьИсключение "KP_SUPPLIER ещё не подключён к механизму собственной очереди. Регистрация и отправка заблокированы, чтобы общий отправитель не обработал то же сообщение.";
    КонецЕсли;

// Отдельная обработка существующей ОчередьШины, только для KP_SUPPLIER.
// Интеграция1СШина и её приватные методы здесь не меняются и не вызываются.
Процедура ОтправитьСообщенияКабинета() Экспорт
    ПроверитьИзоляциюОчереди();
    Настройки = КХ_КП_ОбменАстор.ПолучитьНастройки();
    Если Не Настройки.Включен Тогда
        Возврат;
    КонецЕсли;
    Таймаут = 900;
    Настройки.Свойство("ТаймаутКвитанцииСекунд", Таймаут);
    Если ТипЗнч(Таймаут) <> Тип("Число") Тогда
        ВызватьИсключение "Таймаут квитанции должен быть числом секунд.";
    КонецЕсли;
    Если Таймаут <= 0 Тогда
        ВызватьИсключение "Таймаут квитанции должен быть положительным.";
    КонецЕсли;
    КХ_КП_ОбменАстор.ПроверитьНастройки(Настройки);
    ПорогПовтора = ТекущаяДатаСеанса() - Настройки.ИнтервалПовтораСекунд;
    ПорогКвитанции = ТекущаяДатаСеанса() - Таймаут;
    СистемыКабинета = КХ_ОбменШиныОбщий.СистемыОбмена("KP_SUPPLIER");
    Статусы = Новый Массив;
    Статусы.Добавить(Перечисления.СтатусыСообщенийШины.ОжидаетОтправки);
    Статусы.Добавить(Перечисления.СтатусыСообщенийШины.Ошибка);
    Запрос = Новый Запрос("ВЫБРАТЬ Очередь.ИдентификаторОбъекта КАК ИдентификаторОбъекта ИЗ РегистрСведений.ОчередьШины КАК Очередь ГДЕ Очередь.ВнешняяСистема В (&Системы) И ((Очередь.СтатусСообщения В (&Статусы) И Очередь.ДатаОтправки <= &ПорогПовтора) ИЛИ (Очередь.СтатусСообщения = &Отправлено И Очередь.ДатаОтправки <= &ПорогКвитанции)) УПОРЯДОЧИТЬ ПО Очередь.ДатаОтправки, Очередь.ДатаРегистрации, Очередь.ИдентификаторОбъекта");
    Запрос.УстановитьПараметр("Системы", СистемыКабинета);
    Запрос.УстановитьПараметр("Статусы", Статусы);
    Запрос.УстановитьПараметр("Отправлено", Перечисления.СтатусыСообщенийШины.Отправлено);
    Запрос.УстановитьПараметр("ПорогКвитанции", ПорогКвитанции);
    Запрос.УстановитьПараметр("ПорогПовтора", ПорогПовтора);
    Выборка = Запрос.Выполнить().Выбрать();
    КоличествоОтправок = 0;
    Пока Выборка.Следующий() Цикл
        Текущая = ПрочитатьНашуЗапись(Выборка.ИдентификаторОбъекта);
        Если Текущая = Неопределено Тогда
            Продолжить;
        КонецЕсли;
        ОжидаетКвитанцию = Текущая.СтатусСообщения = Перечисления.СтатусыСообщенийШины.Отправлено И Текущая.ДатаОтправки <= ПорогКвитанции;
        Если Текущая.СтатусСообщения <> Перечисления.СтатусыСообщенийШины.ОжидаетОтправки И Текущая.СтатусСообщения <> Перечисления.СтатусыСообщенийШины.Ошибка И Не ОжидаетКвитанцию Тогда
            Продолжить;
        КонецЕсли;
        РезультатОтправки = ОтправитьЗаписьОчереди(Текущая);
        Пропустить = Ложь;
        РезультатОтправки.Свойство("Пропустить", Пропустить);
        Если Не Пропустить Тогда
            ЗафиксироватьРезультатНашейОтправки(Текущая.ИдентификаторОбъекта, РезультатОтправки);
            КоличествоОтправок = КоличествоОтправок + 1;
            Если КоличествоОтправок >= 100 Тогда
                Прервать;
            КонецЕсли;
        КонецЕсли;
    КонецЦикла;
КонецПроцедуры

Функция ПрочитатьНашуЗапись(ИдентификаторОбъекта)
    Запрос = Новый Запрос("ВЫБРАТЬ Очередь.ИдентификаторОбъекта КАК ИдентификаторОбъекта, Очередь.ИдентификаторСообщения КАК ИдентификаторСообщения, Очередь.КаналСервисаИнтеграции КАК КаналСервисаИнтеграции, Очередь.УзелПолучатель КАК УзелПолучатель, Очередь.ВнешняяСистема КАК ВнешняяСистема, Очередь.Объект КАК Объект, Очередь.ДатаРегистрации КАК ДатаРегистрации, Очередь.ДатаОтправки КАК ДатаОтправки, Очередь.СтатусСообщения КАК СтатусСообщения, Очередь.ТекстОшибки КАК ТекстОшибки, Очередь.ДатаЗагрузки КАК ДатаЗагрузки, Очередь.СтроковыйОбъект КАК СтроковыйОбъект ИЗ РегистрСведений.ОчередьШины КАК Очередь ГДЕ Очередь.ВнешняяСистема = &Система И Очередь.ИдентификаторОбъекта = &ИдентификаторОбъекта");
    Запрос.УстановитьПараметр("Система", "KP_SUPPLIER");
    Запрос.УстановитьПараметр("ИдентификаторОбъекта", ИдентификаторОбъекта);
    Выборка = Запрос.Выполнить().Выбрать();
    Результат = Неопределено;
    Пока Выборка.Следующий() Цикл
        Запись = Новый Структура("ИдентификаторОбъекта,ИдентификаторСообщения,КаналСервисаИнтеграции,УзелПолучатель,ВнешняяСистема,Объект,ДатаРегистрации,ДатаОтправки,СтатусСообщения,ТекстОшибки,ДатаЗагрузки,СтроковыйОбъект");
        ЗаполнитьЗначенияСвойств(Запись, Выборка);
        Если Результат <> Неопределено Тогда
            Если Результат.УзелПолучатель <> Запись.УзелПолучатель Или Результат.КаналСервисаИнтеграции <> Запись.КаналСервисаИнтеграции Тогда
                ВызватьИсключение "Один ID пакета KP_SUPPLIER найден на разных маршрутах.";
            КонецЕсли;
            Если Результат.СтатусСообщения = Перечисления.СтатусыСообщенийШины.ДоставкаПодтверждена Тогда
                Продолжить;
            КонецЕсли;
        КонецЕсли;
        Результат = Запись;
    КонецЦикла;
    Возврат Результат;
КонецФункции

        Запись = ПрочитатьНашуЗапись(ИдентификаторОбъекта);
        Если Запись = Неопределено Тогда
            ЗафиксироватьТранзакцию();
            Возврат Ложь;
        КонецЕсли;
        Сейчас = ТекущаяДатаСеанса();
        Разрешено = Запись.СтатусСообщения = Перечисления.СтатусыСообщенийШины.ОжидаетОтправки;
        Разрешено = Разрешено Или (Запись.СтатусСообщения = Перечисления.СтатусыСообщенийШины.Ошибка И Запись.ДатаОтправки <= Сейчас - Настройки.ИнтервалПовтораСекунд);
        Разрешено = Разрешено Или (Запись.СтатусСообщения = Перечисления.СтатусыСообщенийШины.Отправлено И Запись.ДатаОтправки <= Сейчас - Настройки.ТаймаутКвитанцииСекунд);
        Маркер = МаркерHTTP(Запись);
        НеРанее = Дата(1, 1, 1);
        Маркер.Свойство("HTTPНеРанее", НеРанее);
        Если Не Разрешено Или Сейчас < НеРанее Тогда
            ЗафиксироватьТранзакцию();
            Возврат Ложь;
        КонецЕсли;
        Маркер.Вставить("HTTPТекущаяПопытка", ПопыткаHTTP);
        Маркер.Вставить("HTTPРезультатПопытки", "");
        Запись.Объект = ЗначениеВСтрокуВнутр(Маркер);
        // Резервируем попытку до отправки, сеть вызывается после фиксации.
        // При падении до отправки запись повторится после таймаута квитанции.
        Запись.СтатусСообщения = Перечисления.СтатусыСообщенийШины.Отправлено;
        Запись.ДатаОтправки = Сейчас;
        Запись.ТекстОшибки = "";
        ЗаписатьНашуЗапись(Запись);
        ЗафиксироватьТранзакцию();
        Возврат Истина;
    Исключение
        ОтменитьТранзакцию();
        ВызватьИсключение;
    КонецПопытки;

Процедура ЗафиксироватьРезультатНашейОтправки(ИдентификаторОбъекта, РезультатОтправки)

    КонецПопытки;

Функция ОбработатьРезультатКабинета(ТекстJSON) Экспорт
    // Только внутренний результат HTTP-адаптера Шины из выделенного входящего канала.
    // Обработчик события должен проверить отправителя и канал до этого вызова.
    ПроверитьИзоляциюОчереди();
    Настройки = КХ_КП_ОбменАстор.ПолучитьНастройки();
    // При оперативной паузе квитанции продолжаем принимать.
    Если Не Настройки.HTTPКонтрактПодтвержден Тогда
        ВызватьИсключение "HTTP-контракт ещё не подтверждён.";
    КонецЕсли;
    РезультатШины = КХ_КП_СлужебныйОбмен.ИзJSON(ТекстJSON);
    ИдентификаторПакета = КХ_КП_СлужебныйОбмен.Поле(РезультатШины, "correlation_id");
    ИдентификаторОбъекта = XMLСтрока(Новый УникальныйИдентификатор(ИдентификаторПакета));
    НачатьТранзакцию();
    Попытка
        ЗаблокироватьНашиЗаписи();
        Запись = ПрочитатьНашуЗапись(ИдентификаторОбъекта);
        Если Запись = Неопределено Тогда
            ЗафиксироватьТранзакцию();
            Возврат Ложь;
        КонецЕсли;
        Пакет = КХ_КП_СлужебныйОбмен.ИзJSON(Запись.СтроковыйОбъект);
        Результат = КХ_КП_HTTPКонтракт.РазобратьРезультатHTTP(РезультатШины, Пакет);
        Маркер = МаркерHTTP(Запись);
        ТекущаяПопытка = "";
        ПолученРезультат = "";
        Маркер.Свойство("HTTPТекущаяПопытка", ТекущаяПопытка);
        Маркер.Свойство("HTTPРезультатПопытки", ПолученРезультат);
        // Поздний ответ предыдущей попытки не отменяет результат новой.
        // Поздний 200 для неизменного пакета доказывает применение этого же снимка.
        Если ПустаяСтрока(ТекущаяПопытка) Тогда
            ВызватьИсключение "Пакет ещё не отправлялся по HTTP-маршруту.";
        КонецЕсли;
        Если Результат.ИдентификаторПопытки <> ТекущаяПопытка И Результат.Статус <> "applied" Тогда
            ЗафиксироватьТранзакцию();
            Возврат Истина;
        КонецЕсли;
        Если Запись.СтатусСообщения = Перечисления.СтатусыСообщенийШины.ДоставкаПодтверждена Тогда
            ЗафиксироватьТранзакцию();
            Возврат Истина;
        КонецЕсли;
        Если (ПолученРезультат = Результат.ИдентификаторПопытки Или Запись.СтатусСообщения = Перечисления.СтатусыСообщенийШины.Отменено) И Результат.Статус <> "applied" Тогда
            ЗафиксироватьТранзакцию();
            Возврат Истина;
        КонецЕсли;
        Маркер.Вставить("HTTPРезультатПопытки", Результат.ИдентификаторПопытки);
        Если Результат.Статус = "applied" Тогда
            Запись.СтатусСообщения = Перечисления.СтатусыСообщенийШины.ДоставкаПодтверждена;
            Запись.ДатаЗагрузки = ТекущаяДатаСеанса();
            Запись.ТекстОшибки = "";
        ИначеЕсли Результат.Статус = "retry" Тогда
            Запись.СтатусСообщения = Перечисления.СтатусыСообщенийШины.Ошибка;
            Запись.ТекстОшибки = Результат.Описание;
            Задержка = Макс(Настройки.ИнтервалПовтораСекунд, Результат.Задержка);
            Маркер.Вставить("HTTPНеРанее", ТекущаяДатаСеанса() + Задержка);
        Иначе
            // rejected: данные/размер; blocked: маршрут/доступ/неподдерживаемый ответ.
            // Автоматически не повторяем. После исправления нужен явный перезапуск.
            Запись.СтатусСообщения = Перечисления.СтатусыСообщенийШины.Отменено;
            Запись.ТекстОшибки = Результат.Описание;
        КонецЕсли;
        Запись.Объект = ЗначениеВСтрокуВнутр(Маркер);
        ЗаписатьНашуЗапись(Запись);
        ЗафиксироватьТранзакцию();
        Возврат Истина;
    Исключение
        ОтменитьТранзакцию();
        ВызватьИсключение;
    КонецПопытки;
