// АСТОР. Общий модуль: Сервер = Да, остальные контексты и привилегированный = Нет.
// Прямая отправка через сервис интеграции. Код не выполняет HTTP-запрос.
// Тело передаётся без новой JSON-оболочки, UTF-8 без BOM и добавочного перевода строки.
// Успешный возврат означает отправку в Шину, а не сохранение данных бэком.

Функция ОтправитьJSON(ТекстJSON, ИдентификаторПакета = "", ИдентификаторПопытки = "", СрокUTC = Неопределено) Экспорт
    Настройки = КХ_КП_ОбменАстор.ПолучитьНастройки();
    КХ_КП_ОбменАстор.ПроверитьНастройки(Настройки);
    ПроверитьТелоJSON(ТекстJSON);
    Если ТипЗнч(ИдентификаторПакета) <> Тип("Строка") Тогда
        ВызватьИсключение "Идентификатор пакета должен быть строкой UUID.";
    КонецЕсли;
    Если ПустаяСтрока(ИдентификаторПакета) Тогда
        ИдентификаторПакета = Строка(Новый УникальныйИдентификатор);
    КонецЕсли;
    ПроверкаUUID = Новый УникальныйИдентификатор(ИдентификаторПакета);
    Если ПроверкаUUID = Новый УникальныйИдентификатор("00000000-0000-0000-0000-000000000000") Тогда
        ВызватьИсключение "Пустой UUID пакета недопустим.";
    КонецЕсли;
    Если ПустаяСтрока(ИдентификаторПопытки) Тогда
        ИдентификаторПопытки = Строка(Новый УникальныйИдентификатор);
    КонецЕсли;
    ПроверкаПопытки = Новый УникальныйИдентификатор(ИдентификаторПопытки);
    Если ПроверкаПопытки = Новый УникальныйИдентификатор("00000000-0000-0000-0000-000000000000") Тогда
        ВызватьИсключение "Пустой UUID попытки недопустим.";
    КонецЕсли;
    БайтыСообщения = ПолучитьДвоичныеДанныеИзСтроки(ТекстJSON, КодировкаТекста.UTF8, Ложь);
    Если БайтыСообщения.Размер() > Настройки.МаксимальныйРазмерСообщения Тогда
        ВызватьИсключение "JSON превышает настроенный лимит сообщения.";
    КонецЕсли;

    ФайловыйПоток = Неопределено;
    ТелоСообщения = Неопределено;
    ИмяФайла = "";
    Попытка
        Сообщение = СервисыИнтеграции[Настройки.СервисИнтеграции].СоздатьСообщение();
        Сообщение.КодОтправителя = Настройки.КодОтправителя;
        Сообщение.КодПолучателя = Настройки.КодПолучателя;
        Сообщение.Параметры.Вставить("configuration", "Astor");
        Сообщение.Параметры.Вставить("ТипСообщения", "astor.reference.snapshot");
        Сообщение.Параметры.Вставить("ИдентификаторПакета", ИдентификаторПакета);
        Сообщение.Параметры.Вставить("ИдентификаторПопытки", ИдентификаторПопытки);
        Сообщение.Параметры.Вставить("ContentType", "application/json; charset=utf-8");
        Сообщение.Параметры.Вставить("Кодировка", "UTF-8");
        Сообщение.Параметры.Вставить("РазмерСообщения", БайтыСообщения.Размер());

        ИмяФайла = ПолучитьИмяВременногоФайла("json");
        БайтыСообщения.Записать(ИмяФайла);
        ФайловыйПоток = ФайловыеПотоки.Открыть(ИмяФайла, РежимОткрытияФайла.Открыть, ДоступКФайлу.Чтение);
        ТелоСообщения = Сообщение.ПолучитьТелоКакПоток();
        ФайловыйПоток.КопироватьВ(ТелоСообщения);
        ТелоСообщения.СброситьБуферы();
        ФайловыйПоток.Закрыть();
        ФайловыйПоток = Неопределено;
        ТелоСообщения.Закрыть();
        ТелоСообщения = Неопределено;
        УдалитьФайлы(ИмяФайла);
        ИмяФайла = "";

        Если СрокUTC <> Неопределено Тогда
            Если ТекущаяУниверсальнаяДата() >= СрокUTC Тогда
                ВызватьИсключение "Истёк срок жизни пакета до передачи в Шину.";
            КонецЕсли;
        КонецЕсли;
        СервисыИнтеграции[Настройки.СервисИнтеграции][Настройки.КаналИсходящий].ОтправитьСообщение(Сообщение);
    Исключение
        ЗакрытьПотокБезИсключения(ФайловыйПоток);
        ЗакрытьПотокБезИсключения(ТелоСообщения);
        УдалитьФайлБезИсключения(ИмяФайла);
        ВызватьИсключение;
    КонецПопытки;
    Возврат Новый Структура("ОтправленоВШину,ИдентификаторСообщения,ИдентификаторПакета", Истина, Сообщение.Идентификатор, ИдентификаторПакета);
КонецФункции

Процедура ПроверитьТелоJSON(ТекстJSON) Экспорт
    Если ТипЗнч(ТекстJSON) <> Тип("Строка") Тогда
        ВызватьИсключение "Тело сообщения должно быть строкой JSON.";
    КонецЕсли;
    Если ПустаяСтрока(ТекстJSON) Тогда
        ВызватьИсключение "Пустое тело сообщения недопустимо.";
    КонецЕсли;
    Данные = КХ_КП_СлужебныйОбмен.ИзJSON(ТекстJSON);
    Если ТипЗнч(Данные) <> Тип("Структура") И ТипЗнч(Данные) <> Тип("Массив") Тогда
        ВызватьИсключение "Корнем JSON должен быть объект или массив.";
    КонецЕсли;
КонецПроцедуры

Процедура ЗакрытьПотокБезИсключения(Поток)
    Если Поток = Неопределено Тогда
        Возврат;
    КонецЕсли;
    Попытка
        Поток.Закрыть();
    Исключение
        // Сохраняем первичную ошибку отправки.
    КонецПопытки;
КонецПроцедуры

Процедура УдалитьФайлБезИсключения(ИмяФайла)
    Если ПустаяСтрока(ИмяФайла) Тогда
        Возврат;
    КонецЕсли;
    Попытка
        УдалитьФайлы(ИмяФайла);
    Исключение
        // Сохраняем первичную ошибку отправки.
    КонецПопытки;
КонецПроцедуры
