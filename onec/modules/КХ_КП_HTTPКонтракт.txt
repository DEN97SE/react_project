Функция КлассифицироватьКодHTTP(КодHTTP) Экспорт
    Если ТипЗнч(КодHTTP) <> Тип("Число") Тогда
        ВызватьИсключение "http_status должен быть числом.";
    КонецЕсли;
    Если КодHTTP <> Цел(КодHTTP) Или (КодHTTP <> 0 И (КодHTTP < 100 Или КодHTTP > 599)) Тогда
        ВызватьИсключение "Недопустимый HTTP-код.";
    КонецЕсли;
    Если КодHTTP = 200 Тогда
        Возврат "applied";
    ИначеЕсли КодHTTP = 0 Или КодHTTP = 408 Или КодHTTP = 425 Или КодHTTP = 429 Или КодHTTP >= 500 Тогда
        Возврат "retry";
    ИначеЕсли КодHTTP = 400 Или КодHTTP = 413 Или КодHTTP = 415 Или КодHTTP = 422 Тогда
        Возврат "rejected";
    КонецЕсли;
    // Включая 202/204, перенаправления, 401/403/404/405/409.
    // Другие коды успеха и конфликт ревизии нельзя молча считать применением.
    Возврат "blocked";
КонецФункции

Функция РазобратьРезультатHTTP(РезультатШины, Пакет) Экспорт
    Если КХ_КП_СлужебныйОбмен.Поле(РезультатШины, "message_type") <> "supplier.reference.http-result" Тогда
        ВызватьИсключение "Ожидался внутренний HTTP-результат Шины.";
    КонецЕсли;
    Для Каждого Имя Из СтрРазделить("correlation_id,content_hash,attempt_id", ",") Цикл
        ЗначениеПоля = КХ_КП_СлужебныйОбмен.Поле(РезультатШины, Имя);
        Если ТипЗнч(ЗначениеПоля) <> Тип("Строка") Тогда
            ВызватьИсключение "Поле HTTP-результата должно быть строкой: " + Имя;
        КонецЕсли;
    КонецЦикла;
    ПроверкаПопытки = Новый УникальныйИдентификатор(РезультатШины.attempt_id);
    Если ПроверкаПопытки = Новый УникальныйИдентификатор("00000000-0000-0000-0000-000000000000") Тогда
        ВызватьИсключение "Пустой attempt_id недопустим.";
    КонецЕсли;
    Ревизия = КХ_КП_СлужебныйОбмен.Поле(РезультатШины, "revision");
    Если РезультатШины.correlation_id <> Пакет.message_id Или Ревизия <> Пакет.revision Или РезультатШины.content_hash <> Пакет.content_hash Тогда
        ВызватьИсключение "HTTP-результат относится к другому пакету, ревизии или содержимому.";
    КонецЕсли;
    КодHTTP = КХ_КП_СлужебныйОбмен.Поле(РезультатШины, "http_status");
    Статус = КлассифицироватьКодHTTP(КодHTTP);
    Задержка = 0;
    РезультатШины.Свойство("retry_after_seconds", Задержка);
    Если ТипЗнч(Задержка) <> Тип("Число") Тогда
        ВызватьИсключение "retry_after_seconds должен быть числом.";
    КонецЕсли;
    Если Задержка < 0 Или Задержка <> Цел(Задержка) Или Задержка > 31536000 Тогда
        ВызватьИсключение "Недопустимый интервал Retry-After: требуется ручная проверка.";
    КонецЕсли;
    Описание = "HTTP " + СокрЛП(Строка(КодHTTP)) + ": " + Статус;
    // Не переносим в очередь тело ответа, HTML шлюза, ключи и персональные данные.
    Возврат Новый Структура("Статус,Описание,Задержка,ИдентификаторПопытки", Статус, Описание, Задержка, РезультатШины.attempt_id);
КонецФункции
