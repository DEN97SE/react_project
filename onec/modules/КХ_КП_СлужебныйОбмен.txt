Функция JSON(Значение) Экспорт
    // Ключи объектов сортируются; порядок массивов сохраняется.
    // Только JSON-совместимые значения. Даты заранее преобразуются в ISO-строки.
    ТипЗначения = ТипЗнч(Значение);
    Если ТипЗначения = Тип("Структура") Тогда
        Таблица = Новый ТаблицаЗначений;
        Таблица.Колонки.Добавить("Ключ", Новый ОписаниеТипов("Строка"));
        Таблица.Колонки.Добавить("Значение");
        Для Каждого Элемент Из Значение Цикл
            СтрокаТаблицы = Таблица.Добавить();
            СтрокаТаблицы.Ключ = Элемент.Ключ;
            СтрокаТаблицы.Значение = Элемент.Значение;
        КонецЦикла;
        Таблица.Сортировать("Ключ Возр");
        Части = Новый Массив;
        Для Каждого СтрокаТаблицы Из Таблица Цикл
            Части.Добавить(JSONПримитива(СтрокаТаблицы.Ключ) + ":" + JSON(СтрокаТаблицы.Значение));
        КонецЦикла;
        Возврат "{" + СтрСоединить(Части, ",") + "}";
    ИначеЕсли ТипЗначения = Тип("Массив") Тогда
        Части = Новый Массив;
        Для Каждого Элемент Из Значение Цикл
            Части.Добавить(JSON(Элемент));
        КонецЦикла;
        Возврат "[" + СтрСоединить(Части, ",") + "]";
    ИначеЕсли ТипЗначения = Тип("Строка") Или ТипЗначения = Тип("Число") Или ТипЗначения = Тип("Булево") Или ТипЗначения = Тип("Null") Тогда
        Возврат JSONПримитива(Значение);
    КонецЕсли;
    ВызватьИсключение "Недопустимый тип в JSON: " + Строка(ТипЗначения);
КонецФункции

Функция JSONПримитива(Значение)
    Запись = Новый ЗаписьJSON;
    Запись.УстановитьСтроку();
    ЗаписатьJSON(Запись, Значение);
    Возврат Запись.Закрыть();
КонецФункции

Функция ИзJSON(Текст) Экспорт
    Чтение = Новый ЧтениеJSON;
    Чтение.УстановитьСтроку(Текст);
    Попытка
        Результат = ПрочитатьJSON(Чтение, Ложь);
    Исключение
        Чтение.Закрыть();
        ВызватьИсключение;
    КонецПопытки;
    Чтение.Закрыть();
    Возврат Результат;
КонецФункции

Функция Хеш(Текст) Экспорт
    Данные = ПолучитьДвоичныеДанныеИзСтроки(Текст, КодировкаТекста.UTF8, Ложь);
    Вычислитель = Новый ХешированиеДанных(ХешФункция.SHA256);
    Вычислитель.Добавить(Данные);
    Возврат НРег(ПолучитьHexСтрокуИзДвоичныхДанных(Вычислитель.ХешСумма));
КонецФункции

Функция Поле(Объект, Имя) Экспорт
    Если ТипЗнч(Объект) <> Тип("Структура") Тогда
        ВызватьИсключение "Ожидался JSON-объект.";
    КонецЕсли;
    ЗначениеПоля = Неопределено;
    Если Не Объект.Свойство(Имя, ЗначениеПоля) Тогда
        ВызватьИсключение "Не заполнено поле: " + Имя;
