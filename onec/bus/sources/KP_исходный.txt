метод ПолучитьКПGETОбработкаЗапроса(Запрос: HttpСервисЗапрос)
    ПроверитьТипСообщения(Запрос)
;

метод ПроверитьТипСообщения(Запрос: HttpСервисЗапрос)
    знч ЯвляетсяJSON = Запрос.Заголовки.ПолучитьПервый("content-type")
    если ЯвляетсяJSON == Неопределено
        ОтправитьОтвет(Запрос, Ложь, "Нет обязательного заголовка Сontent-type")
    иначе
        если ЯвляетсяJSON.ВНижнийРегистр().Содержит("application/json")
            знч ДанныеТела = Запрос.Тело.ПрочитатьКакСтроку(Кодировка.Utf8).ВБайты()
            знч ПотокСДанными = ПотокЧтения.ИзБайтов(ДанныеТела)
            знч Сообщение = новый СообщениеИнтеграции({
                    "IsJSON": ЯвляетсяJSON,
                    "SizeMessage": ДанныеТела.Длина().ВСтроку()
                }, ПотокСДанными)
            Integration.ОтправитьСообщениеВУзлы(Сообщение, Integration.Схема.Узлы.message_kp)
            ОтправитьОтвет(Запрос, Истина, "Проверка в ДО!")
        иначе
            ОтправитьОтвет(Запрос, Ложь, "В Сontent-type передан неопознанный тип %ЯвляетсяJSON. Используйте application/json")
        ;
    ;
    возврат
;

метод ОтправитьОтвет(Запрос: HttpСервисЗапрос, Result: Булево, ТекстОшибки: Строка)
    пер ТекстОтвета = {
        "Result": Result,
        "Message": ТекстОшибки
    }
    пер ТелоОтвета = СериализацияJson.ЗаписатьОбъект(ТекстОтвета)
    Запрос.Ответ.Заголовки.Установить("Content-Type", "application/json")
    Запрос.Ответ.УстановитьТело(ТелоОтвета, "UTF-8")
;
